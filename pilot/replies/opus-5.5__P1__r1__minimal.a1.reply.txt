<script>
  // Sticky nav border
  const nav = document.getElementById('nav');
  addEventListener('scroll', () => nav.classList.toggle('scrolled', scrollY > 8), { passive: true });

  // Year
  document.getElementById('year').textContent = new Date().getFullYear();

  // Kanban demo: drag & drop + keyboard (←/→ to move focused card)
  const board = document.getElementById('board');
  const cols = [...board.querySelectorAll('.col')];
  let dragged = null;

  function updateBoard() {
    let total = 0, done = 0;
    cols.forEach(c => {
      const n = c.querySelectorAll('.card').length;
      c.querySelector('.count').textContent = n;
      total += n;
      if (c.dataset.col === 'done') done = n;
    });
    const pct = total ? Math.round(done / total * 100) : 0;
    document.getElementById('progressText').textContent = pct + '% done';
    document.getElementById('progressBar').style.width = pct + '%';
  }

  function afterElement(col, y) {
    const cards = [...col.querySelectorAll('.card:not(.dragging)')];
    return cards.reduce((closest, el) => {
      const box = el.getBoundingClientRect();
      const offset = y - box.top - box.height / 2;
      return offset < 0 && offset > closest.offset ? { offset, el } : closest;
    }, { offset: -Infinity }).el;
  }

  board.addEventListener('dragstart', e => {
    if (!e.target.classList.contains('card')) return;
    dragged = e.target;
    dragged.classList.add('dragging');
    e.dataTransfer.effectAllowed = 'move';
    e.dataTransfer.setData('text/plain', '');
  });
  board.addEventListener('dragend', () => {
    if (dragged) dragged.classList.remove('dragging');
    cols.forEach(c => c.classList.remove('over'));
    dragged = null;
    updateBoard();
  });
  cols.forEach(col => {
    col.addEventListener('dragover', e => {
      e.preventDefault();
      if (!dragged) return;
      col.classList.add('over');
      const after = afterElement(col, e.clientY);
      after ? col.insertBefore(dragged, after) : col.appendChild(dragged);
    });
    col.addEventListener('dragleave', e => {
      if (!col.contains(e.relatedTarget)) col.classList.remove('over');
    });
    col.addEventListener('drop', e => { e.preventDefault(); col.classList.remove('over'); });
  });
  board.addEventListener('keydown', e => {
    const card = e.target.closest('.card');
    if (!card || !['ArrowLeft', 'ArrowRight'].includes(e.key)) return;
    e.preventDefault();
    const i = cols.indexOf(card.closest('.col'));
    const next = cols[i + (e.key === 'ArrowRight' ? 1 : -1)];
    if (next) { next.appendChild(card); card.focus(); updateBoard(); }
  });
  updateBoard();

  // Pricing toggle
  const toggles = document.querySelectorAll('[data-billing]');
  toggles.forEach(btn => btn.addEventListener('click', () => {
    toggles.forEach(b => b.setAttribute('aria-pressed', b === btn));
    document.querySelectorAll('.amt').forEach(a => a.textContent = '$' + a.dataset[btn.dataset.billing]);
  }));

  // Signup form
  document.getElementById('signup').addEventListener('submit', e => {
    e.preventDefault();
    const input = document.getElementById('email');
    const msg = document.getElementById('formMsg');
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(input.value.trim())) {
      msg.style.color = '#ffb39e';
      msg.textContent = 'Please enter a valid email address.';
      input.focus();
      return;
    }
    msg.style.color = '#9fd3b0';
    msg.textContent = "You're in! Check your inbox for a link to your new workspace.";
    input.value = '';
  });

  // Scroll reveal
  const io = new IntersectionObserver(entries => {
    entries.forEach(en => { if (en.isIntersecting) { en.target.classList.add('in'); io.unobserve(en.target); } });
  }, { threshold: 0.12 });
  document.querySelectorAll('.reveal').forEach(el => io.observe(el));
</script>

**What's in it:**

- **Design:** The palette is warm off-white and ink with a vermilion accent. Headings use serif type, and cards have a hard offset shadow. Everything uses system fonts, so the file needs no network requests.
- **Interactive hero demo:** The kanban board works. You can drag cards between columns, and the progress bar updates as you go. Keyboard users can focus a card and move it with ← / →.
- **Sections:**
  - Sticky nav
  - Hero
  - Customer logo strip
  - Uneven grid of features
  - Three-step "how it works"
  - Testimonial with stats
  - Pricing with a monthly/yearly toggle
  - FAQ that expands and collapses (uses the built-in `<details>` element)
  - Email signup with validation
  - Footer
- **Accessibility:**
  - Visible focus outlines
  - ARIA labels and live regions, so screen readers announce changes
  - Respects the user's reduced-motion setting
  - Layout adapts to tablet and mobile widths
