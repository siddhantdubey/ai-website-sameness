      function updateCounters() {
        const doneCount = tasks.filter(t => t.status === "done").length;
        const pct = Math.round((doneCount / tasks.length) * 100);
        progressBar.style.width = pct + "%";
        progressLabel.textContent = doneCount + "/" + tasks.length + " done";

        document.getElementById("badge-all").textContent = tasks.length;
        document.getElementById("badge-high").textContent = tasks.filter(t => t.priority === "High").length;
        document.getElementById("badge-active").textContent = tasks.filter(t => t.status === "active").length;
      }

      function getFilteredTasks() {
        if (currentFilter === "High") return tasks.filter(t => t.priority === "High");
        if (currentFilter === "active") return tasks.filter(t => t.status === "active");
        return tasks;
      }

      function toggleTaskDone(id) {
        const task = tasks.find(t => t.id === id);
        if (!task) return;
        task.status = task.status === "done" ? "active" : "done";
        renderWorkspace();
      }

      function prioClass(prio) {
        if (prio === "High") return "prio-high";
        if (prio === "Med") return "prio-med";
        return "prio-low";
      }

      function renderBoard(items) {
        const cols = [
          { key: "todo", label: "Backlog", dot: "#94a3b8" },
          { key: "active", label: "In Progress", dot: "#2f54eb" },
          { key: "done", label: "Shipped", dot: "#0d9488" }
        ];

        const html = cols.map(col => {
          const colTasks = items.filter(t => t.status === col.key);
          const cards = colTasks.map(t => `
            <div class="task-card ${t.status === "done" ? "is-done" : ""}" data-task-id="${t.id}">
              <div class="task-card-top">
                <span class="task-id">${t.id}</span>
                <span class="task-priority ${prioClass(t.priority)}">${t.priority}</span>
              </div>
              <div class="task-title">${t.title}</div>
              <div class="task-card-foot">
                <span>${t.team} · ${t.points}pt</span>
                <span class="avatar-mini" style="background:${t.color};">${t.owner}</span>
              </div>
            </div>
          `).join("");

          return `
            <div class="board-col">
              <div class="board-col-head">
                <span><span class="col-dot" style="background:${col.dot};"></span>${col.label}</span>
                <span>${colTasks.length}</span>
              </div>
              ${cards || '<div style="font-size:0.78rem; color:var(--text-muted); padding:14px 4px;">No matching tasks</div>'}
            </div>
          `;
        }).join("");

        container.innerHTML = `<div class="board-grid">${html}</div>`;
      }

      function renderTimeline(items) {
        const rows = items.map(t => {
          const barBg = t.status === "done" ? "#0d9488" : t.status === "active" ? "#2f54eb" : "#64748b";
          return `
            <div class="timeline-row">
              <div style="font-weight:600; white-space:nowrap; overflow:hidden; text-overflow:ellipsis;" title="${t.title}">
                <span style="font-family:var(--font-mono); font-size:0.72rem; color:var(--text-muted); margin-right:6px;">${t.id}</span>
                ${t.title}
              </div>
              <div class="timeline-track">
                <div class="gantt-bar" data-task-id="${t.id}" style="left:${t.start}%; width:${t.span}%; background:${barBg}; cursor:pointer;">
                  <span>${t.team}</span>
                  <span>${t.status === "done" ? "✓" : t.points + "pt"}</span>
                </div>
              </div>
            </div>
          `;
        }).join("");

          <div class="timeline-view">

              <div>Deliverable</div>
              <div>Week 1</div>
              <div>Week 2</div>
              <div>Week 3</div>
              <div>Week 4</div>
              <div>Release</div>
            </div>
            ${rows}
          </div>
        `;
      }

      function renderList(items) {
        const rows = items.map(t => {
          const isDone = t.status === "done";
          const statusLabel = isDone ? "Shipped" : t.status === "active" ? "In Progress" : "Backlog";
          return `
            <div class="list-row" data-task-id="${t.id}">
              <span class="check-circle ${isDone ? "checked" : ""}">
                ${isDone ? '<svg width="11" height="11" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="3"><polyline points="20 6 9 17 4 12"/></svg>' : ""}
              </span>
              <span class="task-id">${t.id}</span>
              <span style="font-weight:600; ${isDone ? "text-decoration:line-through; color:var(--text-muted);" : ""}">${t.title}</span>
              <span class="hide-sm" style="font-size:0.78rem; color:var(--text-secondary);">${statusLabel}</span>
              <span class="hide-sm"><span class="task-priority ${prioClass(t.priority)}">${t.priority}</span></span>
              <span class="avatar-mini" style="background:${t.color};">${t.owner}</span>
            </div>
          `;
        }).join("");

        container.innerHTML = `<div class="list-view">${rows}</div>`;
      }

      function renderWorkspace() {
        updateCounters();
        const items = getFilteredTasks();
        if (currentView === "board") renderBoard(items);
        else if (currentView === "timeline") renderTimeline(items);
        else renderList(items);
      }

      container.addEventListener("click", function (e) {

          renderWorkspace();
        });
      });

      document.querySelectorAll(".sidebar-item[data-filter]").forEach(btn => {
        btn.addEventListener("click", () => {
          document.querySelectorAll(".sidebar-item[data-filter]").forEach(b => b.classList.remove("active"));
          btn.classList.add("active");
          currentFilter = btn.getAttribute("data-filter");
          renderWorkspace();
        });
      });

      renderWorkspace();

        const q = (query || "").toLowerCase();
        const filtered = commands.filter(c => c.label.toLowerCase().includes(q));
        cmdList.innerHTML = filtered.map((c, idx) => `

          btn.addEventListener("click", () => {

            filtered[i].action();
