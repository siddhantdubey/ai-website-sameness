  <script>
    // Initial Issues State
    const issues = [
      { id: 'KX-201', title: 'Implement CRDT delta compression over WebSockets', tag: 'infra', priority: 'p-urgent', prioLabel: 'Urgent', status: 'todo', owner: '@marcus', pts: 5 },
      { id: 'KX-204', title: 'Zero-latency virtualized grid for 50k+ backlog rows', tag: 'frontend', priority: 'p-high', prioLabel: 'High', status: 'todo', owner: '@elena', pts: 3 },
      { id: 'KX-208', title: 'Rotate Ephemeral Ed25519 webhook signing keys', tag: 'security', priority: 'p-urgent', prioLabel: 'Urgent', status: 'doing', owner: '@priya', pts: 3 },
      { id: 'KX-211', title: 'Keyboard shortcut chord engine (g+i, c, Cmd+K)', tag: 'frontend', priority: 'p-med', prioLabel: 'Medium', status: 'doing', owner: '@elena', pts: 2 },
      { id: 'KX-215', title: 'Multi-region Postgres logical replication failover', tag: 'infra', priority: 'p-high', prioLabel: 'High', status: 'doing', owner: '@devon', pts: 8 },
      { id: 'KX-194', title: 'GitHub App PR auto-link & status check reporter', tag: 'infra', priority: 'p-med', prioLabel: 'Medium', status: 'done', owner: '@marcus', pts: 5 },
      { id: 'KX-198', title: 'Strict CSP headers & OAuth PKCE token hardening', tag: 'security', priority: 'p-high', prioLabel: 'High', status: 'done', owner: '@priya', pts: 3 }
    ];

    let activeFilter = 'all';
    let nextIdNum = 220;

    function showToast(msg) {
      const t = document.getElementById('toast');
      t.textContent = msg;
      t.classList.add('show');
      clearTimeout(window._toastTimer);
      window._toastTimer = setTimeout(() => t.classList.remove('show'), 2600);
    }

    function renderKanban() {
      const cols = { todo: [], doing: [], done: [] };
      const filtered = issues.filter(item => activeFilter === 'all' || item.tag === activeFilter);

      filtered.forEach(item => {
        cols[item.status].push(item);
      });

      ['todo', 'doing', 'done'].forEach(colKey => {
        const container = document.getElementById(`cards-${colKey}`);
        const countEl = document.getElementById(`count-${colKey}`);
        container.innerHTML = '';
        countEl.textContent = cols[colKey].length;

        cols[colKey].forEach(item => {
          const card = document.createElement('div');
          card.className = 'issue-card';
          const nextLabel = item.status === 'todo' ? 'Start →' : item.status === 'doing' ? 'Ship ✓' : 'Reopen ↺';
          card.innerHTML = `
            <div class="issue-top">
              <span>${item.id} • ${item.pts}pt</span>
              <span class="priority-badge ${item.priority}">${item.prioLabel}</span>
            </div>
            <div class="issue-title">${item.title}</div>
            <div class="issue-footer">
              <span class="tag-chip">#${item.tag}</span>
              <span class="advance-hint">${nextLabel}</span>
              <span class="mono">${item.owner}</span>
            </div>
          `;
          card.addEventListener('click', () => {
            if (item.status === 'todo') {
              item.status = 'doing';
              showToast(`${item.id} moved to In Progress`);
            } else if (item.status === 'doing') {
              item.status = 'done';
              showToast(`${item.id} shipped to production ✓`);
            } else {
              item.status = 'todo';
              showToast(`${item.id} moved back to Backlog`);
            }
            renderKanban();
          });
          container.appendChild(card);
        });
      });

      // Update Sprint progress bar
      const totalPts = issues.reduce((acc, i) => acc + i.pts, 0);
      const donePts = issues.filter(i => i.status === 'done').reduce((acc, i) => acc + i.pts, 0);
      const pct = totalPts ? Math.round((donePts / totalPts) * 100) : 0;
      document.getElementById('completionPct').textContent = `${pct}%`;
      document.getElementById('progressFill').style.width = `${pct}%`;
    }

    // View Tabs
    const tabBtns = document.querySelectorAll('.tab-btn');
    const views = {
      kanban: document.getElementById('view-kanban'),
      timeline: document.getElementById('view-timeline'),
      analytics: document.getElementById('view-analytics')
    };

    function switchView(viewName) {
      tabBtns.forEach(btn => btn.classList.toggle('active', btn.dataset.view === viewName));
      views.kanban.style.display = viewName === 'kanban' ? 'grid' : 'none';
      views.timeline.style.display = viewName === 'timeline' ? 'flex' : 'none';
      views.analytics.style.display = viewName === 'analytics' ? 'grid' : 'none';
    }

    tabBtns.forEach(btn => {
      btn.addEventListener('click', () => switchView(btn.dataset.view));
    });

    // Tag Filter Buttons
    const filterBtns = document.querySelectorAll('.filter-btn');
    filterBtns.forEach(btn => {
      btn.addEventListener('click', () => {
        filterBtns.forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        activeFilter = btn.dataset.filter;
        renderKanban();
      });
    });

    // Pricing Toggle
    const billAnnual = document.getElementById('billAnnual');
    const billMonthly = document.getElementById('billMonthly');
    const proPrice = document.getElementById('proPrice');
    const entPrice = document.getElementById('entPrice');

    billAnnual.addEventListener('click', () => {
      billAnnual.classList.add('active');
      billMonthly.classList.remove('active');
      proPrice.textContent = '$12';
      entPrice.textContent = '$28';
    });

    billMonthly.addEventListener('click', () => {
      billMonthly.classList.add('active');
      billAnnual.classList.remove('active');
      proPrice.textContent = '$15';
      entPrice.textContent = '$35';
    });

    // Command Palette Modal
    const cmdModal = document.getElementById('cmdModal');
    const cmdInput = document.getElementById('cmdInput');

    function openCmd() {
      cmdModal.classList.add('open');
      cmdInput.value = '';
      setTimeout(() => cmdInput.focus(), 40);
    }

    function closeCmd() {
      cmdModal.classList.remove('open');
    }

    document.getElementById('openCmdBtn').addEventListener('click', openCmd);
    document.getElementById('heroCmdTrigger').addEventListener('click', openCmd);

    cmdModal.addEventListener('click', (e) => {
      if (e.target === cmdModal) closeCmd();
    });

    window.addEventListener('keydown', (e) => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
        e.preventDefault();
        cmdModal.classList.contains('open') ? closeCmd() : openCmd();
      } else if (e.key === 'Escape' && cmdModal.classList.contains('open')) {
        closeCmd();
      }
    });

    document.querySelectorAll('.cmd-item').forEach(item => {
      item.addEventListener('click', () => {
        const action = item.dataset.action;
        if (action === 'create-task') {
          const id = `KX-${nextIdNum++}`;
          issues.unshift({
            id,
            title: 'Optimize WebAssembly worker thread pool saturation',
            tag: 'infra',
            priority: 'p-high',
            prioLabel: 'High',
            status: 'todo',
            owner: '@you',
            pts: 3
          });
          activeFilter = 'all';
          filterBtns.forEach(b => b.classList.toggle('active', b.dataset.filter === 'all'));
          switchView('kanban');
          renderKanban();
          showToast(`Created ${id} in Cycle-28 Backlog`);
        } else if (action === 'view-kanban') {
          switchView('kanban');
          showToast('Switched to Kanban Board');
        } else if (action === 'view-timeline') {
          switchView('timeline');
          showToast('Switched to Roadmap Timeline');
        } else if (action === 'view-analytics') {
          switchView('analytics');
          showToast('Switched to Velocity Telemetry');
        } else if (action === 'ship-all') {
          issues.forEach(i => { if (i.status === 'doing') i.status = 'done'; });
          switchView('kanban');
          renderKanban();
          showToast('All active sprint issues shipped to Done ✓');
        }
        closeCmd();
      });
    });

    // Initial render
    renderKanban();
  </script>
