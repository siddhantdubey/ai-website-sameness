  <script>
    const stages = ["To do","In progress","Done"];
    const tasks = [
      ["To do","Design","Explore campaign visuals","May 16","AL"],
      ["To do","Content","Draft launch email","May 17","JK"],
      ["In progress","Design","Build the brand kit","May 14","MR"],
      ["In progress","Content","Write homepage copy","May 15","JK"],
      ["Done","Strategy","Define the audience","May 10","SD"],
      ["Done","Strategy","Align on launch goals","May 11","AL"]
    ];
    const saved = [];
    try {
      const items = JSON.parse(localStorage.getItem("relay-tasks") || "[]");
      if (Array.isArray(items)) saved.push(...items.filter(item =>
        Array.isArray(item) && item.length === 5 && item.every(value => typeof value === "string") && stages.includes(item[0])
      ));
    } catch {}
    function element(tag, className, text) {
      const node = document.createElement(tag);
      if (className) node.className = className;
      if (text) node.textContent = text;
      return node;
    }
    function render() {
      const columns = document.querySelector("#columns");
      columns.replaceChildren();
      for (const stage of stages) {
        const items = [...saved,...tasks].filter(task => task[0] === stage);
        const column = element("section","stage");
        const heading = element("h3","stage-title");
        heading.append(element("span","dot"),stage,element("span","count",String(items.length)));
        column.append(heading);
        for (const [,category,title,date,owner] of items) {
          const card = element("article","card");
          const foot = element("div","card-foot");
          foot.append(element("span","",date),element("span","avatar",owner));
          card.append(element("span","tag "+category.toLowerCase(),category),element("h3","",title),foot);
          column.append(card);
        }
        columns.append(column);
      }
    }
    const dialog = document.querySelector("#task-dialog");
    const form = document.querySelector("#task-form");
    document.querySelectorAll(".new-task").forEach(button => button.addEventListener("click",() => dialog.showModal()));
    document.querySelector("#cancel").onclick = () => dialog.close();
    form.onsubmit = event => {
      event.preventDefault();
      const fields = new FormData(form);
      const title = fields.get("title").trim();
      if (!title) {form.elements.title.value = ""; form.reportValidity(); return;}
      saved.unshift([fields.get("stage"),fields.get("category"),title,"No due date","You"]);
      try {localStorage.setItem("relay-tasks",JSON.stringify(saved));} catch {}
      render();
      dialog.close();
      form.reset();
      document.querySelector("#workspace").scrollIntoView();
    };
    render();
  </script>
