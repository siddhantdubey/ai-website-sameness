    function getFilteredUnits() {
      return WORK_UNITS.filter(u => {

    function renderLedger() {
      const filtered = getFilteredUnits();
      rowCountDisplay.textContent = `SHOWING ${filtered.length} OF ${WORK_UNITS.length} WORK UNITS`;

      if (!filtered.some(u => u.id === selectedId) && filtered.length > 0) {
        selectedId = filtered[0].id;
      }

      ledgerBody.innerHTML = filtered.map(u => {
        const isSelected = u.id === selectedId ? "selected" : "";
        const blockerTag = u.blocker ? `<span class="blocker-pill">! ${u.blocker}</span>` : "";
        return `
          <tr class="ledger-row ${isSelected}" data-id="${u.id}">

            <td class="title-cell">${u.title}${blockerTag}</td>
