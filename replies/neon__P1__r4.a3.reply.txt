    (function () {

      const $$ = (sel, ctx = document) => Array.from(ctx.querySelectorAll(sel));

      const toastEl = $('#toast');
      let toastTimer;
      function showToast(msg) {
        toastEl.textContent = msg;
        toastEl.classList.add('show');
        clearTimeout(toastTimer);
        toastTimer = setTimeout(() => toastEl.classList.remove('show'), 2800);
      }

      // ---------- Mobile Drawer ----------
      const mobileBtn = $('#mobileMenuBtn');

        mobileBtn.setAttribute('aria-expanded', String(open));

        link.addEventListener('click', () => {
          mobileDrawer.classList.remove('open');
          mobileBtn.setAttribute('aria-expanded', 'false');
        });
      });

      // ---------- Interactive Workspace Preview ----------
      const STATUS_ORDER = ['In Progress', 'In Review', 'Shipped'];
      const STATUS_META = {
        'In Progress': { color: '#D94E28', nextLabel: '→ Review' },

      const TEAM_BAR_COLOR = {

        'Engineering': '#2B5B84',
        'Product': '#2D6A4F'
      };

      let tasks = [

        { id: 'CRN-208', title: 'Cycle velocity & burndown API', team: 'Engineering', status: 'In Progress', priority: 'High', owner: 'DL', ownerColor: '#161614', start: 40, width: 42 },

      let nextTaskNum = 219;
      let activeView = 'board';
      let activeTeam = 'all';

      function getFilteredTasks() {
        return activeTeam === 'all' ? tasks : tasks.filter(t => t.team === activeTeam);
      }

      function cycleTaskStatus(id) {
        const task = tasks.find(t => t.id === id);
        if (!task) return;
        const idx = STATUS_ORDER.indexOf(task.status);
        task.status = STATUS_ORDER[(idx + 1) % STATUS_ORDER.length];
        renderWorkspace();
        showToast(`${task.id} moved to ${task.status}`);
      }

      function updateSidebarStats() {
        const allCount = tasks.length;
        const shippedCount = tasks.filter(t => t.status === 'Shipped').length;
        const pct = allCount ? Math.round((shippedCount / allCount) * 100) : 0;
        $('#cyclePercent').textContent = `${pct}%`;
        $('#cycleProgressFill').style.width = `${pct}%`;
        $('#cycleDoneCount').textContent = `${shippedCount} of ${allCount} shipped`;

        $('[data-count="all"]').textContent = allCount;
        ['Product', 'Design', 'Engineering'].forEach(team => {
          const el = $(`[data-count="${team}"]`);
          if (el) el.textContent = tasks.filter(t => t.team === team).length;
        });
      }

      function escapeHtml(str) {
        return str.replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m]));
      }

      function renderBoard(items) {
        return `<div class="board-grid">${STATUS_ORDER.map(status => {
          const colTasks = items.filter(t => t.status === status);
          const meta = STATUS_META[status];
          return `
            <div class="board-col">
              <div class="board-col-head">
                <div class="board-col-title">
                  <span class="status-dot" style="background:${meta.color}"></span>
                  ${status}
                </div>
                <span class="col-count">${colTasks.length}</span>
              </div>
              ${colTasks.length ? colTasks.map(t => `
                <div class="task-card">
                  <div class="task-card-top">
                    <span class="task-id">${t.id}</span>
                    <span class="priority-pill priority-${t.priority.toLowerCase()}">${t.priority}</span>
                  </div>
                  <div class="task-title ${t.status === 'Shipped' ? 'done' : ''}">${escapeHtml(t.title)}</div>
                  <div class="task-footer">
                    <div class="task-tags">
                      <span class="avatar-mini" style="background:${t.ownerColor}" title="Assignee">${t.owner}</span>
                      <span class="team-tag ${TEAM_CLASS[t.team]}">${t.team}</span>
                    </div>
                    <button class="task-action-btn" type="button" data-advance="${t.id}">${meta.nextLabel}</button>
                  </div>
                </div>
              `).join('') : `<div style="font-size:0.8rem;color:var(--ink-muted);padding:16px 4px;text-align:center">No issues here</div>`}
            </div>
          `;
        }).join('')}</div>`;

        return `
          <div class="timeline-view">
            <div class="timeline-header">
              <div>Issue</div>
              <div>Week 1 · Nov 16</div>
              <div>Week 2 · Nov 23</div>
              <div>Week 3 · Nov 30</div>
              <div>Week 4 · Dec 07</div>
            </div>
            ${items.map(t => `
              <div class="timeline-row">
                <div class="timeline-label">
                  <span class="status-dot" style="background:${STATUS_META[t.status].color};flex-shrink:0"></span>
                  <span class="task-id">${t.id}</span>
                  <span class="timeline-label-text" title="${escapeHtml(t.title)}">${escapeHtml(t.title)}</span>
                </div>
                <div class="timeline-track">
                  <button type="button" class="timeline-bar" data-advance="${t.id}"
                    style="left:${t.start}%;width:${t.width}%;background:${t.status === 'Shipped' ? '#2D6A4F' : TEAM_BAR_COLOR[t.team]};opacity:${t.status === 'Shipped' ? '0.82' : '1'}"
                    title="Click to advance status">
                    <span>${t.team}</span>
                    <span style="font-family:var(--font-mono);font-size:10px;opacity:0.9">${t.status}</span>
                  </button>
                </div>
              </div>
            `).join('')}
          </div>
        `;
      }

      function renderList(items) {
        return `
          <div class="list-view">
            <div class="list-row header">
              <div></div>
              <div class="hide-sm">ID</div>
              <div>Title</div>
              <div class="hide-sm">Team</div>
              <div>Status</div>
              <div class="hide-sm">Priority</div>
            </div>
            ${items.map(t => `
              <div class="list-row">
                <div>
                  <button class="status-check ${t.status === 'Shipped' ? 'checked' : ''}" type="button" data-advance="${t.id}" aria-label="Cycle status for ${t.id}">
                    ${t.status === 'Shipped' ? '<svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="3"><path d="M20 6 9 17l-5-5"/></svg>' : ''}
                  </button>
                </div>
                <div class="task-id hide-sm">${t.id}</div>
                <div style="font-weight:600;${t.status === 'Shipped' ? 'text-decoration:line-through;color:var(--ink-muted)' : ''}">${escapeHtml(t.title)}</div>
                <div class="hide-sm"><span class="team-tag ${TEAM_CLASS[t.team]}">${t.team}</span></div>
                <div>
                  <button class="status-pill" type="button" data-advance="${t.id}">
                    <span class="status-dot" style="background:${STATUS_META[t.status].color}"></span>
                    <span>${t.status}</span>
                  </button>
                </div>
                <div class="hide-sm"><span class="priority-pill priority-${t.priority.toLowerCase()}">${t.priority}</span></div>
              </div>
            `).join('')}
          </div>
        `;
      }

      function renderWorkspace() {
        const canvas = $('#workspaceCanvas');
        const items = getFilteredTasks();
        if (activeView === 'board') canvas.innerHTML = renderBoard(items);
        else if (activeView === 'timeline') canvas.innerHTML = renderTimeline(items);
        else canvas.innerHTML = renderList(items);
        updateSidebarStats();
      }

      $('#workspaceCanvas').addEventListener('click', e => {
        const btn = e.target.closest('[data-advance]');
        if (btn) cycleTaskStatus(btn.getAttribute('data-advance'));

      $$('.view-tab').forEach(tab => {
        tab.addEventListener('click', () => {
          activeView = tab.dataset.view;
          $$('.view-tab').forEach(t => {
            const isActive = t === tab;
            t.classList.toggle('active', isActive);
            t.setAttribute('aria-selected', String(isActive));
          });
          renderWorkspace();
        });
      });

      $('#teamFilterList').addEventListener('click', e => {
        const btn = e.target.closest('[data-team]');
        if (!btn) return;
        activeTeam = btn.dataset.team;
        $$('#teamFilterList .sidebar-item').forEach(b => b.classList.toggle('active', b === btn));
        renderWorkspace();
      });

      $('#quickAddForm').addEventListener('submit', e => {
        e.preventDefault();
        const input = $('#quickAddInput');
        const title = input.value.trim();
        if (!title) return;
        const team = $('#quickAddTeam').value;
        const initials = { Engineering: 'ER', Design: 'MK', Product: 'NA' }[team];

          id: `CRN-${nextTaskNum++}`,
          title,
          team,
          status: 'In Progress',
          priority: 'Med',
          owner: initials,

          start: 18 + Math.floor(Math.random() * 35),
          width: 30

        tasks.unshift(newTask);
        input.value = '';
        if (activeTeam !== 'all' && activeTeam !== team) {
          activeTeam = 'all';
          $$('#teamFilterList .sidebar-item').forEach(b => b.classList.toggle('active', b.dataset.team === 'all'));

        showToast(`Created ${newTask.id} in In Progress`);

      renderWorkspace();

      // ---------- Workflow Tabs ----------
      const WORKFLOW_DATA = {
        product: {

      const cmdModal = $('#cmdModal');

      const cmdList = $('#cmdList');

      const COMMANDS = [
        { label: 'Try Interactive Workspace (Board View)', hint: 'Demo', run: () => { activeView = 'board'; $$('.view-tab')[0].click(); $('#interactive-demo').scrollIntoView(); } },
        { label: 'Switch Workspace to Timeline View', hint: 'Demo', run: () => { $$('.view-tab')[1].click(); $('#interactive-demo').scrollIntoView(); } },

        { label: 'Start 14-day Free Trial', hint: 'Signup', run: () => openSignup('Team') }
      ];

      function renderCommands(filter = '') {
        const q = filter.trim().toLowerCase();
        const filtered = COMMANDS.filter(c => c.label.toLowerCase().includes(q) || c.hint.toLowerCase().includes(q));
        cmdList.innerHTML = filtered.length ? filtered.map((c, i) => `
          <button class="cmd-option" type="button" data-cmd-idx="${COMMANDS.indexOf(c)}">
            <strong>${c.label}</strong>
            <span>${c.hint}</span>
          </button>
        `).join('') : `<div style="padding:18px;text-align:center;color:var(--ink-muted);font-size:0.875rem">No matching actions</div>`;
      }

      function openCmd() {
        cmdModal.classList.add('open');
        cmdSearchInput.value = '';
        renderCommands();
        setTimeout(() => cmdSearchInput.focus(), 30);
      }

      $('#openCmdBtn').addEventListener('click', openCmd);
      $('#footerCmdLink').addEventListener('click', e => { e.preventDefault(); openCmd(); });
      cmdSearchInput.addEventListener('input', () => renderCommands(cmdSearchInput.value));
      cmdList.addEventListener('click', e => {
        const btn = e.target.closest('[data-cmd-idx]');
        if (!btn) return;
        closeAllModals();
        COMMANDS[Number(btn.dataset.cmdIdx)].run();
      });

      document.addEventListener('keydown', e => {
        if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
          e.preventDefault();
          if (cmdModal.classList.contains('open')) closeAllModals();
          else openCmd();
        } else if (e.key === 'Escape') {
          closeAllModals();
        }
      });

      // ---------- Scroll Reveal ----------
      if ('IntersectionObserver' in window) {
        const io = new IntersectionObserver(entries => {
          entries.forEach(entry => {
            if (entry.isIntersecting) {
              entry.target.classList.add('visible');
              io.unobserve(entry.target);
            }
          });
        }, { threshold: 0.12 });
        $$('.reveal').forEach(el => io.observe(el));
      } else {
        $$('.reveal').forEach(el => el.classList.add('visible'));
      }
    })();
  </script>
