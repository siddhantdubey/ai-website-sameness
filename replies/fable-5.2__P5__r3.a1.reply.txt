        <div>
          <h2>What happens at six every morning</h2>
          <p>Most plans are built on the estimate from kickoff and never checked against it again. Plumb checks daily. If your team gets through about three quarters of a planned day per working day, every remaining task is stretched by that much.</p>
          <p class="aside">Pace is a rolling six-week figure per team. Holidays and booked leave are taken out before the sums are done.</p>
        </div>
        <ol class="log">
          <li><time>06:00</time><div><strong>Counts what was finished yesterday</strong> and compares it with what those tasks were estimated at.</div></li>
          <li><time>06:01</time><div><strong>Updates the team's pace.</strong> One slow day barely moves it. Three slow weeks do.</div></li>
          <li><time>06:02</time><div><strong>Reruns the end date of every open project</strong> using the work that's left and the pace you're really going at.</div></li>
          <li><time>06:05</time><div><strong>Sends one message if a date moved by more than a day.</strong> It goes to the project lead and says which tasks caused it. Everyone else is left alone.</div></li>
        </ol>
      </div>

(function () {
  var TOTAL = 28, DEADLINE = 16, CORE = 14, FULL = 23;
  var DAY = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
  var MON = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

  // working days starting Mon 23 Nov 2026
  var days = [], d = new Date(2026, 10, 23);
  while (days.length < TOTAL) {
    if (d.getDay() % 6 !== 0) days.push(new Date(d));
    d.setDate(d.getDate() + 1);
  }
  function fmt(x) { return DAY[x.getDay()] + ' ' + x.getDate() + ' ' + MON[x.getMonth()]; }
  function plural(n, w) { return n + ' ' + w + (n === 1 ? '' : 's'); }
  function list(a) {
    if (a.length < 2) return a[0] || '';
    return a.slice(0, -1).join(', ') + ' and ' + a[a.length - 1];
  }

  var demo = document.getElementById('demo');
  var boxes = demo.querySelectorAll('input[type=checkbox]');
  var elDate = document.getElementById('fc-date');
  var elStatus = document.getElementById('fc-status');
  var elFlag = document.getElementById('fc-flag');
  var elFlagDate = document.getElementById('fc-flag-date');
  var elNote = document.getElementById('fc-note');
  var defaultNote = elNote.textContent;

  function update() {
    var n = CORE, cut = [];
    boxes.forEach(function (b) {
      demo.querySelector('.seg[data-id="' + b.value + '"]').classList.toggle('cut', !b.checked);
      if (b.checked) n += Number(b.dataset.days);
      else cut.push(b.dataset.name);
    });

    var end = days[n - 1];
    var diff = Math.round((end - days[DEADLINE - 1]) / 864e5);
    var label = fmt(end);

    elDate.textContent = label;
    elFlagDate.textContent = label;
    elFlag.style.left = (n / TOTAL * 100) + '%';
    demo.classList.toggle('ok', diff <= 0);

    elStatus.textContent = diff > 0 ? plural(diff, 'day') + ' late'
      : diff === 0 ? 'On the day'
      : plural(-diff, 'day') + ' early';

    if (!cut.length) {
      elNote.textContent = defaultNote;
    } else {
      var s = 'Cutting ' + list(cut) + ' buys back ' + plural(FULL - n, 'working day') + '. ';
      if (diff > 0) s += 'Still ' + plural(diff, 'day') + ' over.';
      else if (diff === 0) s += 'That lands it on the deadline, and it is a conversation you can have with the client today, three weeks out.';
      else s += 'That leaves ' + plural(-diff, 'day') + ' of slack before the deadline.';
      elNote.textContent = s;
    }
  }

  boxes.forEach(function (b) { b.addEventListener('change', update); });
  update();
})();
