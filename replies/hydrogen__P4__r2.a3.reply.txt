        bioModal.addEventListener("click", function (e) {
          if (e.target === bioModal) {
            closeBioModal();
          }
        });
      }

      document.addEventListener("keydown", function (e) {
        if (e.key === "Escape" && bioModal && bioModal.classList.contains("is-open")) {
          closeBioModal();
        }
      });

      // Consultation Form Validation & Confirmation
      var form = document.getElementById("consultationForm");
      var formStatus = document.getElementById("formStatus");

      function setFieldError(id, msg) {
        var input = document.getElementById(id);
        var err = document.getElementById("err-" + id);
        var container = input ? input.closest(".form-field") : null;
        if (container) {
          container.classList.toggle("has-error", Boolean(msg));
        }
        if (err) {
          err.textContent = msg || "";
        }
      }

      if (form) {
        form.addEventListener("submit", function (e) {
          e.preventDefault();
          formStatus.className = "form-status";
          formStatus.textContent = "";

          var fullName = document.getElementById("fullName").value.trim();
          var email = document.getElementById("email").value.trim();
          var phone = document.getElementById("phone").value.trim();
          var practice = document.getElementById("practiceSelect").value.trim();
          var summary = document.getElementById("summary").value.trim();
          var consent = document.getElementById("consent").checked;

          var valid = true;

          if (fullName.length < 2) {
            setFieldError("fullName", "Please enter your full name.");
            valid = false;
          } else {
            setFieldError("fullName", "");
          }

          var emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
          if (!emailPattern.test(email)) {
            setFieldError("email", "Please enter a valid email address.");
            valid = false;
          } else {
            setFieldError("email", "");
          }

          var digits = phone.replace(/\D/g, "");
          if (digits.length < 10) {
            setFieldError("phone", "Please enter a 10-digit telephone number.");
            valid = false;
          } else {
            setFieldError("phone", "");
          }

          if (!practice) {
            setFieldError("practiceSelect", "Please select a practice area.");
            valid = false;
          } else {
            setFieldError("practiceSelect", "");
          }

          if (summary.length < 15) {
            setFieldError("summary", "Please share a brief overview (at least 15 characters).");
            valid = false;
          } else {
            setFieldError("summary", "");
          }

          if (!consent) {
            setFieldError("consent", "Please acknowledge the pre-engagement notice.");
            valid = false;
          } else {
            setFieldError("consent", "");
          }

          if (!valid) {
            var firstError = form.querySelector(".has-error input, .has-error select, .has-error textarea");
            if (firstError) firstError.focus();
            return;
          }

          var refId = "HL-2026-" + Math.floor(1000 + Math.random() * 9000);
          formStatus.className = "form-status is-success";
          formStatus.innerHTML =
            "<strong>Inquiry logged (Reference " +
            refId +
            ").</strong> Thank you, " +
            fullName.replace(/</g, "&lt;") +
            ". Your conflict check for <em>" +
            practice.replace(/</g, "&lt;") +
            "</em> has been routed to our Boston intake desk. A partner will respond within one business day.";

          form.reset();
        });
      }
    })();
  </script>
