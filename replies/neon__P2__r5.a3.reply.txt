          <h2>Three generations, one corner kitchen.</h2>
          <p>
            When Mirella and Lorenzo Ferretti opened the doors at 218 Clinton Street in the autumn of 1987, they brought two things from Emilia-Romagna: a six-foot wooden <em>mattarello</em> for rolling pasta dough and the conviction that a great neighborhood restaurant should feel like an extension of home.
          </p>
          <p>

The recipes haven't rushed to chase trends—our Bolognese still simmers for six hours, our bread is still baked in-house before service, and our dining room still keeps half its garden tables open for neighbors who simply decide to wander in for a bowl of pasta and a glass of Barbera.
          </p>

  <!-- Toast Notification -->
  <div class="toast" id="toast" role="status" aria-live="polite"></div>

  <script>
    (function () {
      // 1. Dynamic year & live service indicator
      const yearEl = document.getElementById('year');
      if (yearEl) yearEl.textContent = new Date().getFullYear();

      const now = new Date();
      const day = now.getDay(); // 0 Sun, 1 Mon ... 6 Sat
      const hour = now.getHours();
      const statusDot = document.getElementById('status-dot');
      const statusText = document.getElementById('status-text');

      if (day === 1) {
        statusDot.classList.add('closed');
        statusText.textContent = 'Closed Mondays (Family Day) • Reopening Tuesday at 5:00 PM';
      } else if (day === 0 && hour >= 13 && hour < 21) {
        statusText.textContent = 'Open now for Sunday Supper (until 9:00 PM) • Walk-ins welcome';
      } else if (day >= 2 && day <= 4 && hour >= 17 && hour < 22) {
        statusText.textContent = 'Open now for dinner (until 10:00 PM) • Pasta counter open for walk-ins';
      } else if ((day === 5 || day === 6) && hour >= 17 && hour < 23) {
        statusText.textContent = 'Open now for dinner (until 11:00 PM) • Pasta counter open for walk-ins';
      } else {
        statusDot.classList.add('closed');
        const nextOpen = day === 0 && hour < 13 ? 'Sunday Supper begins at 1:00 PM' : 'Dinner service begins at 5:00 PM';
        statusText.textContent = nextOpen + ' • Walk-ins welcome at the pasta counter';
      }

      // 2. Toast helper
      const toastEl = document.getElementById('toast');
      let toastTimer;
      function showToast(msg) {
        toastEl.textContent = msg;
        toastEl.classList.add('show');
        clearTimeout(toastTimer);
        toastTimer = setTimeout(() => toastEl.classList.remove('show'), 3200);
      }

      // 3. Sticky header scroll shadow & mobile drawer
      const header = document.getElementById('site-header');
      window.addEventListener('scroll', () => {
        header.classList.toggle('scrolled', window.scrollY > 18);
      }, { passive: true });

      const menuToggle = document.getElementById('menu-toggle');
      const mobileDrawer = document.getElementById('mobile-drawer');
      menuToggle.addEventListener('click', () => {
        const isOpen = mobileDrawer.classList.toggle('open');
        menuToggle.setAttribute('aria-expanded', String(isOpen));
      });

      document.querySelectorAll('.drawer-link').forEach((link) => {
        link.addEventListener('click', () => {
          mobileDrawer.classList.remove('open');
          menuToggle.setAttribute('aria-expanded', 'false');
        });
      });

      // 4. Interactive Menu Tabs & Dietary Filters
      const tabs = document.querySelectorAll('.menu-tab');
      const panels = document.querySelectorAll('.menu-panel');
      const filterPills = document.querySelectorAll('.filter-pill');
      let currentFilter = 'all';

      function applyFilter() {
        panels.forEach((panel) => {
          if (panel.hidden) return;
          const items = panel.querySelectorAll('.dish-item');
          let visibleCount = 0;
          items.forEach((item) => {
            const tags = (item.dataset.tags || '').split(/\s+/);
            const show = currentFilter === 'all' || tags.includes(currentFilter);
            item.classList.toggle('is-hidden', !show);
            if (show) visibleCount++;
          });
          const emptyMsg = panel.querySelector('.empty-filter-msg');
          if (emptyMsg) {
            emptyMsg.style.display = visibleCount === 0 ? 'block' : 'none';
          }
        });
      }

      tabs.forEach((tab) => {
        tab.addEventListener('click', () => {
          tabs.forEach((t) => t.setAttribute('aria-selected', 'false'));
          tab.setAttribute('aria-selected', 'true');
          const targetId = tab.getAttribute('aria-controls');
          panels.forEach((panel) => {
            panel.hidden = panel.id !== targetId;
          });
          applyFilter();
        });
      });

      filterPills.forEach((pill) => {
        pill.addEventListener('click', () => {
          filterPills.forEach((p) => p.classList.remove('active'));
          pill.classList.add('active');
          currentFilter = pill.dataset.filter;
          applyFilter();
        });
      });

      // 5. Reservations Logic
      const dateInput = document.getElementById('res-date');
      const todayISO = new Date(now.getTime() - now.getTimezoneOffset() * 60000).toISOString().split('T')[0];
      dateInput.min = todayISO;
      dateInput.value = todayISO;

      const timeButtons = document.querySelectorAll('.time-btn');
      const hiddenTime = document.getElementById('res-time');
      const seatingSelect = document.getElementById('res-seating');
      const form = document.getElementById('reservation-form');
      const confirmation = document.getElementById('booking-confirmation');

      function selectTimeSlot(timeStr) {
        hiddenTime.value = timeStr;
        timeButtons.forEach((btn) => {
          btn.classList.toggle('selected', btn.dataset.time === timeStr);
        });
      }

      timeButtons.forEach((btn) => {
        btn.addEventListener('click', () => selectTimeSlot(btn.dataset.time));
      });

      // Quick-book chips in Hero
      document.querySelectorAll('[data-quick-time]').forEach((chip) => {
        chip.addEventListener('click', () => {
          selectTimeSlot(chip.dataset.quickTime);
          form.style.display = 'block';
          confirmation.classList.remove('visible');
          document.getElementById('reservations').scrollIntoView({ behavior: 'smooth' });
          setTimeout(() => document.getElementById('res-name').focus(), 450);
          showToast('Selected ' + chip.dataset.quickTime + ' for tonight');
        });
      });

      // Book Sunday Supper button
      document.getElementById('book-sunday-btn').addEventListener('click', () => {
        const nextSunday = new Date(now);
        const daysUntilSunday = (7 - now.getDay()) % 7 || 7;
        nextSunday.setDate(now.getDate() + daysUntilSunday);
        dateInput.value = new Date(nextSunday.getTime() - nextSunday.getTimezoneOffset() * 60000).toISOString().split('T')[0];
        seatingSelect.value = 'Sunday Supper Feast';
        selectTimeSlot('5:30 PM');
        form.style.display = 'block';
        confirmation.classList.remove('visible');
        document.getElementById('reservations').scrollIntoView({ behavior: 'smooth' });
        showToast('Pre-filled for Sunday Supper');
      });

      form.addEventListener('submit', (e) => {
        e.preventDefault();
        const name = document.getElementById('res-name').value.trim();
        const email = document.getElementById('res-email').value.trim();
        const phone = document.getElementById('res-phone').value.trim();
        const guests = document.getElementById('res-guests').value;
        const dateVal = dateInput.value;
        const timeVal = hiddenTime.value;
        const seating = seatingSelect.value;

        if (!name || !email || !email.includes('@') || !phone || !dateVal) {
          showToast('Please complete your name, contact info, and date.');
          return;
        }

        const formattedDate = new Date(dateVal + 'T12:00:00').toLocaleDateString('en-US', {
          weekday: 'short',
          month: 'short',
          day: 'numeric'
        });
        const confCode = 'OM-' + Math.floor(1000 + Math.random() * 9000);

        document.getElementById('conf-name').textContent = name.split(' ')[0];
        document.getElementById('conf-email').textContent = email;
        document.getElementById('conf-details').innerHTML = `
          <div><span>Confirmation #</span><strong>${confCode}</strong></div>
          <div><span>Date &amp; Time</span><strong>${formattedDate} at ${timeVal}</strong></div>
          <div><span>Party</span><strong>${guests} ${guests === '1' ? 'Guest' : 'Guests'} • ${seating}</strong></div>
        `;

        form.style.display = 'none';
        confirmation.classList.add('visible');
        showToast('Reservation confirmed (' + confCode + ')');
      });

      document.getElementById('new-booking-btn').addEventListener('click', () => {
        form.reset();
        dateInput.value = todayISO;
        selectTimeSlot('6:15 PM');
        confirmation.classList.remove('visible');
        form.style.display = 'block';
      });

      // 6. Copy Address & Newsletter
      document.getElementById('copy-address-btn').addEventListener('click', () => {
        const addr = '218 Clinton Street, Brooklyn, NY 11201';
        if (navigator.clipboard && navigator.clipboard.writeText) {
          navigator.clipboard.writeText(addr).then(() => showToast('Address copied to clipboard'));
        } else {
          showToast(addr);
        }
      });

      document.getElementById('newsletter-form').addEventListener('submit', (e) => {
        e.preventDefault();
        const emailInput = document.getElementById('news-email');
        if (!emailInput.value.includes('@')) {
          showToast('Please enter a valid email address.');
          return;
        }
        emailInput.value = '';
        showToast('Grazie! You are on our seasonal notes list.');
      });

      // 7. Gentle Scroll Reveal
      const revealEls = document.querySelectorAll('.reveal');
      if ('IntersectionObserver' in window) {
        const observer = new IntersectionObserver((entries) => {
          entries.forEach((entry) => {
            if (entry.isIntersecting) {
              entry.target.classList.add('in-view');
              observer.unobserve(entry.target);
            }
          });
        }, { threshold: 0.12 });
        revealEls.forEach((el) => observer.observe(el));
      } else {
        revealEls.forEach((el) => el.classList.add('in-view'));
      }
    })();
  </script>
