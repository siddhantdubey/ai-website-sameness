      <path d="M${cx-w*.3} ${h} C${cx-w*.28} ${hy+h*.14} ${cx-hr*1.2} ${hy+hr*1.2} ${cx} ${hy+hr*1.6} C${cx+hr*1.2} ${hy+hr*1.2} ${cx+w*.28} ${hy+h*.14} ${cx+w*.3} ${h} Z"/>`;
    if(kind==="bokeh")g+=`<ellipse cx="${cx+w*.24}" cy="${hy+h*.03}" rx="${hr*.95}" ry="${hr*1.15}"/>
      <path d="M${cx+w*.0} ${h} C${cx+w*.02} ${hy+h*.16} ${cx+w*.12} ${hy+hr*1.4} ${cx+w*.24} ${hy+hr*1.6} C${cx+w*.36} ${hy+hr*1.4} ${cx+w*.46} ${hy+h*.16} ${cx+w*.5} ${h} Z"/>`;
    g+=`</g>`;
  }
  g+=`<rect width="${w}" height="${h}" fill="url(#vg)"/><rect width="${w}" height="${h}" filter="url(#gr)"/>`;
  return"data:image/svg+xml;charset=utf-8,"+encodeURIComponent(`<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${w} ${h}">${g}</svg>`);
