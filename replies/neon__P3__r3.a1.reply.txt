  <script>
    (function () {
      'use strict';

      /* ----------------------------------------------------------------------
         1. DETERMINISTIC PRNG & SHARED FILM GRAIN PATTERN
         ---------------------------------------------------------------------- */
      function mulberry32(seed) {
        return function () {
          let t = (seed += 0x6d2b79f5);
          t = Math.imul(t ^ (t >>> 15), t | 1);
          t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
          return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
        };
      }

      // Reusable 240x240 authentic silver-halide film grain tile
      function createGrainPattern(ctx, intensity = 0.065, monochrome = true) {
        const tile = document.createElement('canvas');
        tile.width = 240;
        tile.height = 240;
        const gctx = tile.getContext('2d');
        const img = gctx.createImageData(240, 240);
        const data = img.data;
        const rand = mulberry32(918273);
        for (let i = 0; i < data.length; i += 4) {
          const v = Math.floor((rand() - 0.5) * 255 + 128);
          if (monochrome) {
            data[i] = data[i + 1] = data[i + 2] = v;
          } else {
            data[i] = v;
            data[i + 1] = Math.min(255, Math.max(0, v + (rand() - 0.5) * 26));
            data[i + 2] = Math.min(255, Math.max(0, v + (rand() - 0.5) * 32));
          }
          data[i + 3] = Math.floor(255 * intensity);
        }
        gctx.putImageData(img, 0, 0);
        return ctx.createPattern(tile, 'repeat');
      }

      function applyFinish(ctx, w, h, { vignette = 0.32, grain = 0.075, mono = false, warmTint = null } = {}) {
        // Subtle optical vignette
        const vig = ctx.createRadialGradient(w * 0.5, h * 0.48, Math.min(w, h) * 0.28, w * 0.5, h * 0.5, Math.max(w, h) * 0.78);
        vig.addColorStop(0, 'rgba(0,0,0,0)');
        vig.addColorStop(1, `rgba(8,7,6,${vignette})`);
        ctx.fillStyle = vig;
        ctx.fillRect(0, 0, w, h);

        // Optional film color cast
        if (warmTint) {
          ctx.save();
          ctx.globalCompositeOperation = 'soft-light';
          ctx.fillStyle = warmTint;
          ctx.fillRect(0, 0, w, h);
          ctx.restore();
        }

        // Organic film grain overlay
        ctx.save();
        ctx.globalCompositeOperation = 'overlay';
        ctx.fillStyle = createGrainPattern(ctx, grain, mono);
        ctx.fillRect(0, 0, w, h);
        ctx.restore();
      }

      /* ----------------------------------------------------------------------
         4. STATE & DOM REFERENCES
         ---------------------------------------------------------------------- */
      const state = {
        activeCategory: 'All',
        viewMode: 'editorial',
        heroIdx: 0,
        lightboxList: PHOTOS,
        lightboxIdx: 0,
        selectedPrint: { size: '40 × 50 cm', price: '€480', ed: 'Edition of 15' }
      };

      const heroPhotos = PHOTOS.filter((p) => p.hero);
      const heroImage = document.getElementById('heroImage');
      const heroPlateTitle = document.getElementById('heroPlateTitle');
      const heroPlateExif = document.getElementById('heroPlateExif');
      const heroFrame = document.getElementById('heroFrame');
      const heroDots = Array.from(document.querySelectorAll('.plate-dot'));

      const filterGroup = document.getElementById('filterGroup');
      const portfolioGrid = document.getElementById('portfolioGrid');
      const viewEditorial = document.getElementById('viewEditorial');
      const viewContact = document.getElementById('viewContact');

      const lightbox = document.getElementById('lightbox');
      const lbImage = document.getElementById('lbImage');
      const lbCounter = document.getElementById('lbCounter');
      const lbCategory = document.getElementById('lbCategory');
      const lbTitle = document.getElementById('lbTitle');
      const lbStory = document.getElementById('lbStory');
      const lbCamera = document.getElementById('lbCamera');
      const lbExposure = document.getElementById('lbExposure');
      const lbStock = document.getElementById('lbStock');

      /* ----------------------------------------------------------------------
         5. HERO FEATURE SWITCHER
         ---------------------------------------------------------------------- */
      function setHeroPlate(idx) {
        state.heroIdx = idx;
        const photo = heroPhotos[idx];
        heroImage.style.opacity = '0.35';
        window.setTimeout(() => {
          heroImage.src = photo.src;
          heroImage.alt = `${photo.title} — ${photo.location}`;
          heroPlateTitle.textContent = `${photo.title} (${photo.year})`;
          heroPlateExif.textContent = `${photo.camera} · ${photo.exposure} · ${photo.stock}`;
          heroImage.style.opacity = '1';
        }, 120);

        heroDots.forEach((dot, i) => dot.classList.toggle('active', i === idx));
      }

      heroDots.forEach((dot) => {
        dot.addEventListener('click', () => setHeroPlate(Number(dot.dataset.heroIndex)));
      });

      function openHeroInLightbox() {
        const current = heroPhotos[state.heroIdx];
        const idx = PHOTOS.findIndex((p) => p.id === current.id);
        openLightbox(PHOTOS, idx);
      }

      heroFrame.addEventListener('click', openHeroInLightbox);
      heroFrame.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          openHeroInLightbox();
        }
      });

      setHeroPlate(0);

      /* ----------------------------------------------------------------------
         6. PORTFOLIO FILTERS, GRID & MONOGRAPH TRIPTYCH
         ---------------------------------------------------------------------- */
      const categories = ['All', 'Editorial', 'Architecture', 'Landscape', 'Monochrome'];

      function renderFilters() {
        filterGroup.innerHTML = '';
        categories.forEach((cat) => {
          const count = cat === 'All' ? PHOTOS.length : PHOTOS.filter((p) => p.category === cat).length;
          const btn = document.createElement('button');
          btn.type = 'button';
          btn.className = `filter-btn${state.activeCategory === cat ? ' active' : ''}`;
          btn.setAttribute('role', 'tab');
          btn.setAttribute('aria-selected', String(state.activeCategory === cat));
          btn.innerHTML = `${cat}<span class="filter-count">(${count})</span>`;
          btn.addEventListener('click', () => {
            state.activeCategory = cat;
            renderFilters();
            renderGallery();
          });
          filterGroup.appendChild(btn);
        });
      }

      function getFilteredPhotos() {
        return state.activeCategory === 'All'
          ? PHOTOS
          : PHOTOS.filter((p) => p.category === state.activeCategory);
      }

      function renderGallery() {
        const list = getFilteredPhotos();
        portfolioGrid.classList.toggle('contact-sheet', state.viewMode === 'contact');
        portfolioGrid.innerHTML = '';

        list.forEach((photo, idx) => {
          const card = document.createElement('article');
          card.className = 'photo-card';
          card.dataset.frame = photo.frame;
          card.tabIndex = 0;
          card.setAttribute('role', 'button');
          card.setAttribute('aria-label', `Inspect ${photo.title}, ${photo.location}`);
          card.style.animationDelay = `${Math.min(idx * 45, 360)}ms`;

          card.innerHTML = `
            <div class="photo-media">
              <img src="${photo.src}" alt="${photo.title} — ${photo.location}" loading="lazy" />
              <div class="photo-overlay" aria-hidden="true">
                <span style="font-family: var(--font-mono); font-size: 0.68rem;">${photo.stock}</span>
                <span class="photo-overlay-cta">Inspect</span>
              </div>
            </div>
            <div class="photo-caption">
              <div>
                <h3 class="photo-title">${photo.title}</h3>
                <p class="photo-sub">${photo.location} &middot; ${photo.year}</p>
              </div>
              <span class="photo-index">${String(idx + 1).padStart(2, '0')}</span>
            </div>
          `;

          const trigger = () => openLightbox(list, idx);
          card.addEventListener('click', trigger);
          card.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' || e.key === ' ') {
              e.preventDefault();
              trigger();
            }
          });
          portfolioGrid.appendChild(card);
        });
      }

      /* ----------------------------------------------------------------------
         7. LIGHTBOX VIEWER
         ---------------------------------------------------------------------- */
      let lastFocused = null;

      function updateLightbox() {
        const photo = state.lightboxList[state.lightboxIdx];
        lbImage.src = photo.src;
        lbImage.alt = `${photo.title} — ${photo.location}`;
        lbCounter.textContent = `${String(state.lightboxIdx + 1).padStart(2, '0')} / ${String(state.lightboxList.length).padStart(2, '0')}`;
        lbCategory.textContent = `${photo.category} · ${photo.year}`;
        lbTitle.textContent = `${photo.title} — ${photo.location}`;
        lbStory.textContent = photo.story;
        lbCamera.textContent = photo.camera;
        lbExposure.textContent = photo.exposure;
        lbStock.textContent = photo.stock;
      }

      function openLightbox(list, idx) {
        lastFocused = document.activeElement;
        state.lightboxList = list;
        state.lightboxIdx = idx;
        updateLightbox();
        lightbox.classList.add('open');
        document.body.style.overflow = 'hidden';
        document.getElementById('lbClose').focus();
      }

      function closeLightbox() {
        lightbox.classList.remove('open');
        document.body.style.overflow = '';
        if (lastFocused && typeof lastFocused.focus === 'function') {
          lastFocused.focus();
        }
      }

      function stepLightbox(dir) {
        const len = state.lightboxList.length;
        state.lightboxIdx = (state.lightboxIdx + dir + len) % len;
        updateLightbox();
      }

      document.getElementById('lbClose').addEventListener('click', closeLightbox);
      document.getElementById('lbPrev').addEventListener('click', () => stepLightbox(-1));
      document.getElementById('lbNext').addEventListener('click', () => stepLightbox(1));

      lightbox.addEventListener('click', (e) => {
        if (e.target === lightbox) closeLightbox();
      });

      document.addEventListener('keydown', (e) => {
        if (!lightbox.classList.contains('open')) return;
        if (e.key === 'Escape') closeLightbox();
        if (e.key === 'ArrowLeft') stepLightbox(-1);
        if (e.key === 'ArrowRight') stepLightbox(1);
      });

      document.getElementById('lbRequestPrint').addEventListener('click', () => {
        const photo = state.lightboxList[state.lightboxIdx];
        closeLightbox();
        prefillInquiry(
          'Archival Print',
          `Print inquiry for "${photo.title}" (${photo.location}, ${photo.year}) — ${state.selectedPrint.size} (${state.selectedPrint.ed}).`
        );
      });

      /* ----------------------------------------------------------------------
         8. DARKROOM / GALLERY THEME TOGGLE & MOBILE MENU
         ---------------------------------------------------------------------- */
      const rootEl = document.documentElement;
      const themeToggle = document.getElementById('themeToggle');
      const themeLabel = document.getElementById('themeLabel');

      function applyTheme(theme) {
        rootEl.setAttribute('data-theme', theme);
        themeLabel.textContent = theme === 'darkroom' ? 'Gallery' : 'Darkroom';
        try { localStorage.setItem('maren-theme', theme); } catch (_) {}
      }

      try {
        const saved = localStorage.getItem('maren-theme');
        if (saved === 'darkroom' || saved === 'gallery') applyTheme(saved);
      } catch (_) {}

      themeToggle.addEventListener('click', () => {
        const next = rootEl.getAttribute('data-theme') === 'darkroom' ? 'gallery' : 'darkroom';
        applyTheme(next);
        showToast(next === 'darkroom' ? 'Darkroom mode enabled' : 'Gallery daylight enabled');
      });

      const menuToggle = document.getElementById('menuToggle');
      const mobileDrawer = document.getElementById('mobileDrawer');
      menuToggle.addEventListener('click', () => {
        const open = mobileDrawer.classList.toggle('open');
        menuToggle.setAttribute('aria-expanded', String(open));
      });
      mobileDrawer.querySelectorAll('a').forEach((link) => {
        link.addEventListener('click', () => {
          mobileDrawer.classList.remove('open');
          menuToggle.setAttribute('aria-expanded', 'false');
        });
      });

         9. PRINT SIZE SELECTOR, PREFILL & INQUIRY FORM
         ---------------------------------------------------------------------- */
      const sizePills = document.querySelectorAll('.size-pill');
      const printEditionSpec = document.getElementById('printEditionSpec');
      const printPrice = document.getElementById('printPrice');

      sizePills.forEach((pill) => {
        pill.addEventListener('click', () => {
          sizePills.forEach((p) => p.classList.remove('active'));
          pill.classList.add('active');
          state.selectedPrint = {
            size: pill.dataset.size,
            price: pill.dataset.price,
            ed: pill.dataset.ed

          printEditionSpec.textContent = `${pill.dataset.ed} — ${pill.dataset.size} (+ 5 cm border)`;
          printPrice.textContent = `${pill.dataset.price} unframed`;
        });
      });

      const projectTypeSelect = document.getElementById('projectType');
      const projectBriefInput = document.getElementById('projectBrief');
      const inquiryForm = document.getElementById('inquiryForm');
      const formConfirmation = document.getElementById('formConfirmation');
      const confirmationSummary = document.getElementById('confirmationSummary');

      function prefillInquiry(type, briefText) {
        inquiryForm.style.display = 'grid';
        formConfirmation.classList.remove('visible');
        projectTypeSelect.value = type;
        if (briefText) projectBriefInput.value = briefText;
        document.getElementById('inquire').scrollIntoView({ behavior: 'smooth' });
        window.setTimeout(() => document.getElementById('clientName').focus(), 450);
      }

      document.querySelectorAll('[data-prefill-type]').forEach((btn) => {
        btn.addEventListener('click', () => prefillInquiry(btn.dataset.prefillType, ''));
      });

      document.getElementById('printInquireBtn').addEventListener('click', () => {
        prefillInquiry(
          'Archival Print',
          `Interested in acquiring an archival print (${state.selectedPrint.size}, ${state.selectedPrint.ed}, ${state.selectedPrint.price}).`
        );
      });

      inquiryForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const name = document.getElementById('clientName').value.trim();
        const email = document.getElementById('clientEmail').value.trim();
        const brief = projectBriefInput.value.trim();
        const type = projectTypeSelect.value;

        if (!name || !email || !email.includes('@') || !brief) {
          showToast('Please complete your name, valid email, and brief.');
          return;
        }

        inquiryForm.style.display = 'none';
        confirmationSummary.textContent = `We received your ${type.toLowerCase()} note from ${name} (${email}). Maren's studio will respond within 24–48 hours.`;
        formConfirmation.classList.add('visible');
        showToast('Studio inquiry recorded');
      });

      document.getElementById('resetFormBtn').addEventListener('click', () => {
        inquiryForm.reset();
        formConfirmation.classList.remove('visible');
        inquiryForm.style.display = 'grid';
      });

      // Copy Email to Clipboard
      document.getElementById('copyEmailBtn').addEventListener('click', async () => {
        const email = 'studio@marenlind.com';
        try {
          if (navigator.clipboard) {
            await navigator.clipboard.writeText(email);
          }
          showToast('Email copied to clipboard');
        } catch (_) {
          showToast(email);
        }
      });

      // Toast helper
      const toastEl = document.getElementById('toast');
      let toastTimer = null;
      function showToast(msg) {
        toastEl.textContent = msg;
        toastEl.classList.add('show');
        window.clearTimeout(toastTimer);
        toastTimer = window.setTimeout(() => toastEl.classList.remove('show'), 2600);
      }

      document.getElementById('yearNow').textContent = String(new Date().getFullYear());
    })();
