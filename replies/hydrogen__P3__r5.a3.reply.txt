      // Deterministic PRNG for procedural photographic rendering
      function mulberry32(seed) {
        return function () {
          let t = (seed += 0x6d2b79f5);
          t = Math.imul(t ^ (t >>> 15), t | 1);
          t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
          return ((t ^ (t >>> 14)) >>> 0) / 4294967296;

      function renderPhoto(spec) {

        const h = spec.h || 1200;

        const rand = mulberry32(spec.seed || 101);

        const bgGrad = ctx.createLinearGradient(0, 0, w * 0.25, h);

        // Scene archetypes
        if (spec.scene === "atrium") {

            ctx.fillRect(w * 0.22, y, w * 0.56, 3);

          mobileDrawer.classList.remove("open");
          mobileMenuBtn.setAttribute("aria-expanded", "false");
          mobileDrawer.setAttribute("aria-hidden", "true");
        });
      });

      // Lightbox Logic
      const lightbox = document.getElementById("lightbox");
      const lbImage = document.getElementById("lbImage");
      const lbCounter = document.getElementById("lbCounter");
      const lbCaption = document.getElementById("lbCaption");
      const lbQuickExif = document.getElementById("lbQuickExif");
      const lbStage = document.getElementById("lbStage");
      const lbZoomBtn = document.getElementById("lbZoomBtn");
      const lbInfoBtn = document.getElementById("lbInfoBtn");
      const lbDrawer = document.getElementById("lbDrawer");
      const lbTitle = document.getElementById("lbTitle");
      const lbCategory = document.getElementById("lbCategory");
      const lbNotes = document.getElementById("lbNotes");
      const lbExifTable = document.getElementById("lbExifTable");

      function updateLightbox() {
        const plate = filteredPlates[lbIndex];
        if (!plate) return;
        lbImage.src = plate.src;
        lbImage.alt = `${plate.title} — ${plate.location}`;
        lbCounter.textContent = `PLATE ${String(lbIndex + 1).padStart(2, "0")} / ${String(filteredPlates.length).padStart(2, "0")}`;
        lbCaption.textContent = `${plate.title} — ${plate.location} (${plate.year})`;
        lbQuickExif.textContent = `${plate.camera} · ${plate.settings}`;

        lbTitle.textContent = plate.title;
        lbCategory.textContent = plate.category;
        lbNotes.textContent = plate.notes;
        lbExifTable.innerHTML = `
          <div class="exif-row"><span>Camera</span><span>${plate.camera}</span></div>
          <div class="exif-row"><span>Lens</span><span>${plate.lens}</span></div>
          <div class="exif-row"><span>Exposure</span><span>${plate.settings}</span></div>
          <div class="exif-row"><span>Medium</span><span>${plate.film}</span></div>
          <div class="exif-row"><span>Location</span><span>${plate.location}</span></div>
          <div class="exif-row"><span>Year</span><span>${plate.year}</span></div>
        `;
      }

      function openLightbox(index) {
        lbIndex = index;
        lbStage.classList.remove("zoomed");
        lbZoomBtn.classList.remove("active");
        updateLightbox();
        lightbox.classList.add("open");
        document.body.style.overflow = "hidden";
      }

      function closeLightbox() {
        lightbox.classList.remove("open");
        lbDrawer.classList.remove("open");
        lbInfoBtn.classList.remove("active");
        lbStage.classList.remove("zoomed");
        lbZoomBtn.classList.remove("active");
        document.body.style.overflow = "";
      }

      function stepLightbox(delta) {
        lbStage.classList.remove("zoomed");
        lbZoomBtn.classList.remove("active");
        lbIndex = (lbIndex + delta + filteredPlates.length) % filteredPlates.length;
        updateLightbox();
      }

      function toggleZoom() {
        const zoomed = lbStage.classList.toggle("zoomed");
        lbZoomBtn.classList.toggle("active", zoomed);
      }

      function toggleInfo() {
        const open = lbDrawer.classList.toggle("open");
        lbInfoBtn.classList.toggle("active", open);
      }

      document.getElementById("heroFrameTrigger").addEventListener("click", () => {
        setFilter("all");
        openLightbox(0);
      });
      document.getElementById("heroFrameTrigger").addEventListener("keydown", (e) => {
        if (e.key === "Enter" || e.key === " ") {
          e.preventDefault();
          setFilter("all");
          openLightbox(0);
        }
      });

      document.getElementById("lbCloseBtn").addEventListener("click", closeLightbox);
      document.getElementById("lbPrev").addEventListener("click", () => stepLightbox(-1));
      document.getElementById("lbNext").addEventListener("click", () => stepLightbox(1));
      lbZoomBtn.addEventListener("click", toggleZoom);
      lbStage.addEventListener("click", toggleZoom);
      lbInfoBtn.addEventListener("click", toggleInfo);

      document.addEventListener("keydown", (e) => {
        if (!lightbox.classList.contains("open")) return;
        if (e.key === "Escape") closeLightbox();
        if (e.key === "ArrowRight") stepLightbox(1);
        if (e.key === "ArrowLeft") stepLightbox(-1);
        if (e.key.toLowerCase() === "z") toggleZoom();
        if (e.key.toLowerCase() === "i") toggleInfo();
      });

      // Commission Inquiry Form
      let selectedProjectType = "Editorial Assignment";
      document.querySelectorAll(".type-pill").forEach((pill) => {
        pill.addEventListener("click", () => {
          selectedProjectType = pill.dataset.type;
          document.querySelectorAll(".type-pill").forEach((p) => p.classList.toggle("active", p === pill));
        });
      });

      const inquiryForm = document.getElementById("inquiryForm");
      const formStatus = document.getElementById("formStatus");
      inquiryForm.addEventListener("submit", (e) => {
        e.preventDefault();
        const name = document.getElementById("clientName").value.trim();
        const email = document.getElementById("clientEmail").value.trim();
        const brief = document.getElementById("projectBrief").value.trim();
        const location = document.getElementById("shootLocation").value.trim() || "TBD";
        const timeline = document.getElementById("timeline").value;

        if (!name || !email || !email.includes("@") || !brief) {
          formStatus.textContent = "Please complete your name, a valid email address, and a brief project note.";
          formStatus.classList.add("visible");
          return;
        }

        formStatus.innerHTML = `
          <strong>Brief logged for Maren Solberg Studio.</strong><br />
          ${selectedProjectType} · ${location} (${timeline}) — confirmation prepared for <em>${email}</em>.
        `;
        formStatus.classList.add("visible");
        inquiryForm.reset();
      });
    })();
  </script>
