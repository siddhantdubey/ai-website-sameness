    const menuDisplay = document.getElementById("menuDisplay");
    const menuTabs = document.querySelectorAll(".menu-tab");

    function renderMenu(categoryKey) {
      const cat = MENU_DATA[categoryKey];
      if (!cat) return;

      const itemsHtml = cat.items.map(item => {
        const tagsHtml = item.tags.map(tag => {
          const isFav = tag === "House Favorite";
          return `<span class="diet-tag ${isFav ? "fav-tag" : ""}">${tag}</span>`;
        }).join("");

        return `
          <div class="menu-item">
            <div>
              <div class="menu-item-name">
                <span>${item.name}</span>
                ${tagsHtml}
              </div>
              <p class="menu-item-desc">${item.desc}</p>
            </div>
            <div class="menu-item-price">${item.price}</div>
          </div>
        `;
      }).join("");

      menuDisplay.innerHTML = `
        <div class="menu-category-head">
          <h3>${cat.title}</h3>
          <span>${cat.subtitle}</span>
        </div>
        <div class="menu-list">
          ${itemsHtml}
        </div>
      `;
    }

        tab.classList.add("active");
        tab.setAttribute("aria-selected", "true");
        renderMenu(tab.dataset.category);
      });

    function openModal(seatingPreset) {
      mobileMenu.classList.remove("open");
      reserveFormWrap.style.display = "block";
      reserveSuccessWrap.style.display = "none";

        resSeatingSelect.value = seatingPreset;

      reserveModal.classList.remove("open");
