         1. PROCEDURAL PHOTOGRAPH PLATE GENERATOR (OFFLINE & SELF-CONTAINED)
         Generates rich, atmospheric photographic compositions with film grain,
         optical vignettes, and authentic tonal palettes.
         ---------------------------------------------------------------------- */
      function createFilmGrainPattern(ctx) {
        const tile = document.createElement("canvas");
        tile.width = 160;
        tile.height = 160;
        const gctx = tile.getContext("2d");
        const img = gctx.createImageData(160, 160);
        const data = img.data;
        let seed = 1337;
        for (let i = 0; i < data.length; i += 4) {
          seed = (seed * 16807) % 2147483647;
          const v = (seed & 255);
          data[i] = v;
          data[i + 1] = v;
          data[i + 2] = v;
          data[i + 3] = 255;
        }
        gctx.putImageData(img, 0, 0);
        return ctx.createPattern(tile, "repeat");
      }

      function applyFinish(ctx, w, h, grainPattern, opts = {}) {
        // Subtle optical vignette
        const vig = ctx.createRadialGradient(w * 0.5, h * 0.48, Math.min(w, h) * 0.28, w * 0.5, h * 0.5, Math.max(w, h) * 0.78);
        vig.addColorStop(0, "rgba(0,0,0,0)");
        vig.addColorStop(1, `rgba(6, 5, 4, ${opts.vignette ?? 0.42})`);
        ctx.fillStyle = vig;
        ctx.fillRect(0, 0, w, h);

        // Film grain overlay
        ctx.save();
        ctx.globalCompositeOperation = "overlay";
        ctx.globalAlpha = opts.grain ?? 0.085;
        ctx.fillStyle = grainPattern;
        ctx.fillRect(0, 0, w, h);
        ctx.restore();
      }

          ctx.fillRect(w * 0.18, h * 0.54, w * 0.64, h * 0.045);

          // Warm interior tungsten glow
          const glow = ctx.createLinearGradient(w * 0.22, h * 0.37, w * 0.76, h * 0.54);
          glow.addColorStop(0, "#ffb86b");
          glow.addColorStop(0.5, "#ffdca8");
          glow.addColorStop(1, "#d98448");
          ctx.fillStyle = glow;
          ctx.fillRect(w * 0.24, h * 0.375, w * 0.52, h * 0.165);

          // Mullions
          ctx.fillStyle = "#161513";
          [0.36, 0.5, 0.64].forEach(mx => ctx.fillRect(w * mx, h * 0.375, 5, h * 0.165));

          // Pool reflection
          ctx.save();
          ctx.globalAlpha = 0.38;
          ctx.fillStyle = glow;
          ctx.fillRect(w * 0.24, h * 0.59, w * 0.52, h * 0.18);
          ctx.restore();

          applyFinish(ctx, w, h, grain, { vignette: 0.36, grain: 0.075 });
        } else if (sceneId === "portrait") {
          // Backlit Atelier Window Portrait
          const bg = ctx.createLinearGradient(0, 0, w, h);
          bg.addColorStop(0, "#d8cebe");
          bg.addColorStop(0.45, "#a69783");
          bg.addColorStop(1, "#332c24");
          ctx.fillStyle = bg;
          ctx.fillRect(0, 0, w, h);

          // Sheer curtain vertical folds
          for (let i = 0; i < 8; i++) {
            ctx.fillStyle = i % 2 === 0 ? "rgba(255,248,236,0.18)" : "rgba(40,32,24,0.08)";
            ctx.fillRect(w * (i * 0.125), 0, w * 0.06, h);
          }

          // Soft rim halo
          const halo = ctx.createRadialGradient(w * 0.44, h * 0.38, w * 0.08, w * 0.44, h * 0.38, w * 0.46);
          halo.addColorStop(0, "rgba(255, 240, 214, 0.72)");
          halo.addColorStop(1, "rgba(255, 240, 214, 0)");
          ctx.fillStyle = halo;
          ctx.fillRect(0, 0, w, h);

          // Profile silhouette
          ctx.fillStyle = "#1c1714";
          ctx.beginPath();
          ctx.arc(w * 0.48, h * 0.37, w * 0.125, 0, Math.PI * 2);
          ctx.fill();
          ctx.beginPath();
          ctx.moveTo(w * 0.25, h);
          ctx.bezierCurveTo(w * 0.27, h * 0.62, w * 0.42, h * 0.52, w * 0.49, h * 0.52);
          ctx.bezierCurveTo(w * 0.58, h * 0.52, w * 0.71, h * 0.64, w * 0.74, h);
          ctx.closePath();
          ctx.fill();

          applyFinish(ctx, w, h, grain, { vignette: 0.34, grain: 0.085 });
        } else if (sceneId === "saltpan") {
          // Aerial Salt Pan Geometry
          ctx.fillStyle = "#d9cfc1";
          ctx.fillRect(0, 0, w, h);

          const cells = [
            [0, 0, 0.48, 0.46, "#c98b7c"],
            [0.48, 0, 0.52, 0.34, "#7ea09d"],
            [0.48, 0.34, 0.52, 0.32, "#d8b48a"],
            [0, 0.46, 0.34, 0.54, "#5d8485"],
            [0.34, 0.46, 0.38, 0.54, "#e6ddd0"],
            [0.72, 0.66, 0.28, 0.34, "#b87868"]
          ];
          cells.forEach(([x, y, cw, ch, col]) => {
            ctx.fillStyle = col;
            ctx.fillRect(x * w + 4, y * h + 4, cw * w - 8, ch * h - 8);
          });

          // Earthen dykes
          ctx.strokeStyle = "#f0eae0";
          ctx.lineWidth = 7;
          ctx.beginPath();
          ctx.moveTo(w * 0.48, 0); ctx.lineTo(w * 0.48, h * 0.46);
          ctx.moveTo(0, h * 0.46); ctx.lineTo(w * 0.72, h * 0.46);
          ctx.moveTo(w * 0.48, h * 0.34); ctx.lineTo(w, h * 0.34);
          ctx.moveTo(w * 0.34, h * 0.46); ctx.lineTo(w * 0.34, h);
          ctx.moveTo(w * 0.72, h * 0.46); ctx.lineTo(w * 0.72, h);
          ctx.stroke();

          applyFinish(ctx, w, h, grain, { vignette: 0.25, grain: 0.08 });
        } else {
          // Greenhouse Fog (Botanical Conservatory)
          const bg = ctx.createLinearGradient(0, 0, 0, h);
          bg.addColorStop(0, "#aeb8ae");
          bg.addColorStop(0.5, "#67786a");
          bg.addColorStop(1, "#222c24");
          ctx.fillStyle = bg;
          ctx.fillRect(0, 0, w, h);

          // Conservatory arched mullions
          ctx.strokeStyle = "rgba(32, 38, 33, 0.65)";
          ctx.lineWidth = 10;
          for (let i = 1; i < 4; i++) {
            ctx.beginPath();
            ctx.moveTo(w * (i * 0.25), 0);
            ctx.lineTo(w * (i * 0.25), h);
            ctx.stroke();
          }

          // Botanical fronds silhouettes
          ctx.fillStyle = "rgba(24, 34, 26, 0.72)";
          for (let i = 0; i < 7; i++) {
            ctx.save();
            ctx.translate(w * 0.48, h * 0.86);
            ctx.rotate((-0.8 + i * 0.26));
            ctx.beginPath();
            ctx.ellipse(0, -h * 0.32, w * 0.045, h * 0.34, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.restore();
          }

          // Steamy morning light haze
          const haze = ctx.createRadialGradient(w * 0.35, h * 0.28, 20, w * 0.45, h * 0.38, w * 0.7);
          haze.addColorStop(0, "rgba(246, 244, 232, 0.42)");
          haze.addColorStop(1, "rgba(246, 244, 232, 0)");
          ctx.fillStyle = haze;
          ctx.fillRect(0, 0, w, h);

          applyFinish(ctx, w, h, grain, { vignette: 0.36, grain: 0.085 });
        }

        return canvas.toDataURL("image/jpeg", 0.92);
      }

      lbPrevBtn.addEventListener("click", prevPlate);
      lbNextBtn.addEventListener("click", nextPlate);

      lightbox.addEventListener("click", (e) => {
        if (e.target === lightbox || e.target.classList.contains("lightbox-stage")) {
          closeLightbox();
        }
      });

      document.addEventListener("keydown", (e) => {
        if (!lightbox.classList.contains("open")) return;
        if (e.key === "Escape") closeLightbox();
        if (e.key === "ArrowLeft") prevPlate();
        if (e.key === "ArrowRight") nextPlate();
      });

      function inquireAboutCurrentPlate() {
        const plate = PLATES[activePlateIndex];
        closeLightbox();
        selectInquiryType("Fine Art Print");
        const brief = document.getElementById("projectBrief");
        brief.value = `Hello Clara — I would like to inquire about acquiring an archival print of "${plate.title}" (${plate.edition}, ${plate.location}).`;
        document.getElementById("contact").scrollIntoView({ behavior: "smooth" });
        setTimeout(() => document.getElementById("clientName").focus(), 450);
      }

      lbInquireBtn.addEventListener("click", inquireAboutCurrentPlate);
      lbPrintCta.addEventListener("click", inquireAboutCurrentPlate);

      /* ----------------------------------------------------------------------
         10. CONTACT FORM, INQUIRY CHIPS & EMAIL COPY
         ---------------------------------------------------------------------- */
      const inquiryChips = document.querySelectorAll(".inquiry-chip");
      const inquiryTypeInput = document.getElementById("inquiryTypeInput");

      function selectInquiryType(type) {
        inquiryTypeInput.value = type;
        inquiryChips.forEach((chip) => {
          chip.classList.toggle("active", chip.getAttribute("data-type") === type);
        });
      }

      inquiryChips.forEach((chip) => {
        chip.addEventListener("click", () => selectInquiryType(chip.getAttribute("data-type")));
      });

      const copyEmailBtn = document.getElementById("copyEmailBtn");
      copyEmailBtn.addEventListener("click", () => {
        const email = "studio@claramoreau.photo";
        if (navigator.clipboard && navigator.clipboard.writeText) {
          navigator.clipboard.writeText(email).catch(() => {});
        }
        copyEmailBtn.textContent = "Copied";
        setTimeout(() => {
          copyEmailBtn.textContent = "Copy";
        }, 1800);
      });

      const commissionForm = document.getElementById("commissionForm");
      const formStatus = document.getElementById("formStatus");
      const formStatusTitle = document.getElementById("formStatusTitle");
      const formStatusMsg = document.getElementById("formStatusMsg");

      commissionForm.addEventListener("submit", (e) => {
        e.preventDefault();
        const name = document.getElementById("clientName").value.trim();
        const email = document.getElementById("clientEmail").value.trim();
        const brief = document.getElementById("projectBrief").value.trim();
        const type = inquiryTypeInput.value;

        if (!name || !email || !email.includes("@") || !brief) {
          formStatusTitle.textContent = "Please complete required fields.";
          formStatusMsg.textContent = "Enter your name, a valid email address, and a brief note about your commission or print inquiry.";
          formStatus.classList.add("visible");
          return;
        }

        const refCode = "CM-" + Math.floor(100000 + Math.random() * 900000);
        formStatusTitle.textContent = `Inquiry Logged · Ref #${refCode}`;
        formStatusMsg.textContent = `Thank you, ${name}. Your ${type.toLowerCase()} note has been prepared for Clara Moreau's studio (${email}).`;
        formStatus.classList.add("visible");
        commissionForm.reset();
        selectInquiryType("Editorial Commission");
      });

      /* ----------------------------------------------------------------------
         11. SCROLL REVEAL & ACTIVE NAV HIGHLIGHT
         ---------------------------------------------------------------------- */
      const revealEls = document.querySelectorAll(".reveal");
      if ("IntersectionObserver" in window) {
        const io = new IntersectionObserver(
          (entries) => {
            entries.forEach((entry) => {
              if (entry.isIntersecting) {
                entry.target.classList.add("visible");
                io.unobserve(entry.target);
              }
            });
          },
          { threshold: 0.12 }
        );
        revealEls.forEach((el) => io.observe(el));
      } else {
        revealEls.forEach((el) => el.classList.add("visible"));
      }
    })();
