/* Original procedural landscape studies keep the portfolio entirely offline. */
const fields = [...document.querySelectorAll("[data-scene]")];
const table = new Float32Array(65536);
let seed = 1732;
for (let i = 0; i < table.length; i++) {
  seed = (Math.imul(seed, 1664525) + 1013904223) >>> 0;
  table[i] = seed / 4294967296;
}
function noise(x, y) {
  const ix = Math.floor(x), iy = Math.floor(y);
  let u = x - ix, v = y - iy;
  u *= u * (3 - 2*u);
  v *= v * (3 - 2*v);
  const k = (ix + iy * 251) & 65535;
  const a = table[k], b = table[(k+1)&65535];
  const c = table[(k+251)&65535], d = table[(k+252)&65535];
  return (a+(b-a)*u)*(1-v)+(c+(d-c)*u)*v;
}
function terrain(x,y) {
  return .52*noise(x,y)+.27*noise(x*2.1,y*2.1)+.14*noise(x*4.4,y*4.4)+.07*noise(x*9,y*9);
}
const skies = [[165,179,179],[170,184,185],[211,173,144],[163,176,187]];
const earth = [[78,83,56],[61,70,61],[175,113,76],[91,102,111]];
function render(canvas, scene) {
  const ctx = canvas.getContext("2d");
  const w = canvas.width, h = canvas.height;
  const frame = ctx.createImageData(w,h), pixels = frame.data;
  const sky = skies[scene], soil = earth[scene];
  for (let col=0; col<w; col++) {
    const x=col/w, ridges=[];
    for (let layer=0; layer<5; layer++) {
      const n=terrain(x*(scene===2?2:4)+layer*13,scene*19+layer);
      const valley=scene===0 ? .17*Math.cos((x-.48)*6+layer*.62) : 0;
      ridges.push(.27+layer*.119+(n-.5)*(scene===2?.30:.65)+valley);
    }
    for (let row=0; row<h; row++) {
      const y=row/h, i=(row*w+col)*4;
      const grain=(table[(row*571+col*17)&65535]-.5)*7;
      let layer=-1;
      for(let j=0;j<5;j++) if(y>=ridges[j]) layer=j;
      const cloud=terrain(x*5+14,y*9+scene*11);
      let r,g,b;
      if(layer<0) {
        const light=y*72+cloud*37+Math.max(0,1-Math.abs(x-.72)*2)*9;
        r=sky[0]+light; g=sky[1]+light*.84; b=sky[2]+light*.71;
      } else if(scene===1 && y>.47+.32*x+.09*noise(x*5,2)) {
        const line=y-(.47+.32*x+.09*noise(x*5,2));
        const ripple=noise(x*38,y*290);
        const foam=Math.max(0,1-line*22)*noise(x*92,y*55)*96;
        r=113+ripple*18+foam; g=137+ripple*17+foam; b=139+ripple*15+foam;
      } else {
        const depth=y-ridges[layer];
        const fold=terrain(x*13+depth*4+layer,depth*5+layer*9);
        const grass=noise(x*160,y*90);
        const shade=(fold-.5)*(scene===2?100:138)+(grass-.5)*17-depth*23;
        const haze=(4-layer)*.15;
        r=soil[0]*(1-haze)+(sky[0]+10)*haze+shade;
        g=soil[1]*(1-haze)+(sky[1]+10)*haze+shade*.91;
        b=soil[2]*(1-haze)+(sky[2]+10)*haze+shade*.73;
        if(scene===3) {
          const snow=Math.min(1,Math.max(0,(fold-depth*.85+.13*grass-.47)*10));
          r+=(225-r)*snow; g+=(230-g)*snow; b+=(231-b)*snow;
        }
      }
      pixels[i]=r+grain; pixels[i+1]=g+grain; pixels[i+2]=b+grain; pixels[i+3]=255;
    }
  }
  ctx.putImageData(frame,0,0);
}
fields.forEach((button,index)=>render(button.firstElementChild,index));
const dialog=document.querySelector("dialog"), enlarged=document.querySelector("#enlarged");
let active=0;
function show(index) {
  active=(index+fields.length)%fields.length;
  const source=fields[active].firstElementChild;
  enlarged.width=source.width; enlarged.height=source.height;
  enlarged.getContext("2d").drawImage(source,0,0);
  enlarged.setAttribute("aria-label",source.getAttribute("aria-label"));
  document.querySelector("#viewer-title").textContent=fields[active].nextElementSibling.querySelector(".caption").textContent;
  document.querySelector("#count").textContent=`Fig. 0${active+1} / 04`;
  if(!dialog.open) dialog.showModal();
}
fields.forEach((button,index)=>button.addEventListener("click",()=>show(index)));
document.querySelector("#prev").onclick=()=>show(active-1);
document.querySelector("#next").onclick=()=>show(active+1);
document.querySelector("#close").onclick=()=>dialog.close();
dialog.addEventListener("keydown",event=>{
  if(event.key==="ArrowLeft") show(active-1);
  if(event.key==="ArrowRight") show(active+1);
});
dialog.addEventListener("click",event=>{
  if(event.target!==dialog) return;
  const box=dialog.getBoundingClientRect();
  if(event.clientX<box.left||event.clientX>box.right||event.clientY<box.top||event.clientY>box.bottom) dialog.close();
});
