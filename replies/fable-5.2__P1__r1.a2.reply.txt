(function () {
  'use strict';

  /* ---------- Fig. 1: timeline slip demo ---------- */
  var demo = document.getElementById('demo');
  var range = document.getElementById('slip');
  var out = document.getElementById('slip-out');
  var chip = document.getElementById('demo-chip');
  var msg = document.getElementById('demo-msg');
  var BUFFER = 3, LAUNCH = 17;

  function workday(i) { // index of working day -> date, starting Mon Nov 23, 2026
    var d = new Date(2026, 10, 23 + Math.floor(i / 5) * 7 + (i % 5));
    return d.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });
  }
  function days(n) { return n + (n === 1 ? ' day' : ' days'); }

  function updateDemo() {
    var s = parseInt(range.value, 10) || 0;
    var buf = Math.max(0, BUFFER - s);
    var late = Math.max(0, s - BUFFER);
    demo.style.setProperty('--slip', s);
    demo.style.setProperty('--buf', buf);
    demo.style.setProperty('--late', late);
    out.textContent = days(s);

    if (s === 0) {
      demo.dataset.state = 'ok';
      chip.textContent = 'On plan';
      msg.textContent = 'Launch is ' + workday(LAUNCH) + ' with ' + days(BUFFER) + ' of buffer. Drag the slider to break something.';
    } else if (late === 0) {
      demo.dataset.state = 'ok';
      chip.textContent = 'Launch holds';
      msg.textContent = 'Review runs ' + days(s) + ' over. 3 tasks rescheduled and their owners notified. ' +
        (buf === 0 ? 'Buffer is fully spent.' : days(buf) + ' of buffer left.') + ' Launch stays ' + workday(LAUNCH) + '.';
    } else {
      demo.dataset.state = 'late';
      chip.textContent = 'Launch moves';
      msg.textContent = 'Review runs ' + days(s) + ' over and the buffer is gone. Launch moves to ' + workday(LAUNCH + late) +
        '. Keel drafted the heads-up for 4 stakeholders, ready for you to send.';
    }
  }
  range.addEventListener('input', updateDemo);
  updateDemo();

  /* ---------- Workload rebalance ---------- */
  var rows = document.querySelectorAll('#load li');
  var rb = document.getElementById('rebalance');
  function paintLoad(balanced) {
    rows.forEach(function (li) {
      var p = parseInt(balanced ? li.dataset.b : li.dataset.a, 10);
      li.querySelector('.fill').style.setProperty('--p', p);
      li.querySelector('.pct').textContent = p + '%';
      li.classList.toggle('over', p > 100);
    });
  }
  rb.addEventListener('click', function () {
