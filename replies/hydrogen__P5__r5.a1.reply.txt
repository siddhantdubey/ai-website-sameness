    document.getElementById("builderSlider").addEventListener("input", updatePricing);
    document.getElementById("viewerSlider").addEventListener("input", updatePricing);

    // ---------- Command Palette (⌘K) ----------
    const commands = [
      { label: "Switch demo to Board view", hint: "⌘B", action: () => { setWorkspaceView("board"); scrollToEl("#demo-workspace"); } },
      { label: "Switch demo to List view", hint: "L", action: () => { setWorkspaceView("list"); scrollToEl("#demo-workspace"); } },
      { label: "Focus new issue input", hint: "C", action: () => { scrollToEl("#demo-workspace"); setTimeout(() => document.getElementById("quickAddInput").focus(), 250); } },
      { label: "Toggle compact row density", hint: "View", action: () => { document.body.classList.toggle("compact-mode"); showToast("Toggled compact density"); } },
      { label: "Jump to Manifesto", hint: "01", action: () => scrollToEl("#ledger") },
      { label: "Jump to Cycle Mechanics", hint: "02", action: () => scrollToEl("#mechanics") },
      { label: "Jump to Pricing & Estimator", hint: "04", action: () => scrollToEl("#pricing") },
      { label: "Create a workspace", hint: "Free", action: () => openSignupModal("Start workspace") }
    ];

    const cmdModal = document.getElementById("cmdModal");
    const cmdInput = document.getElementById("cmdInput");
    const cmdResults = document.getElementById("cmdResults");
    let selectedCmdIdx = 0;

    function openCmdPalette() {
      cmdModal.classList.add("open");
      cmdInput.value = "";
      selectedCmdIdx = 0;
      renderCmdResults("");
      setTimeout(() => cmdInput.focus(), 20);
    }

    function closeCmdPalette() {
      cmdModal.classList.remove("open");
    }

    function renderCmdResults(query) {
      const q = query.trim().toLowerCase();
      const matches = commands.filter(c => c.label.toLowerCase().includes(q));
      if (selectedCmdIdx >= matches.length) selectedCmdIdx = 0;
      if (matches.length === 0) {
        cmdResults.innerHTML = `<div style="padding: 16px; text-align: center; color: var(--ink-3); font-size: 13px;">No matching commands</div>`;
        return;
      }
      cmdResults.innerHTML = matches.map((c, idx) => `
        <button type="button" class="cmd-item ${idx === selectedCmdIdx ? "selected" : ""}" data-cmd-idx="${idx}">
          <span>${c.label}</span>
          <span class="cmd-item-hint">${c.hint}</span>
        </button>
      `).join("");

      cmdResults.querySelectorAll("[data-cmd-idx]").forEach(btn => {
        btn.addEventListener("click", () => {
          const item = matches[Number(btn.dataset.cmdIdx)];
          closeCmdPalette();
          item.action();
        });
      });
    }

    cmdInput.addEventListener("input", () => {
      selectedCmdIdx = 0;
      renderCmdResults(cmdInput.value);
    });

    cmdInput.addEventListener("keydown", e => {
      const matches = commands.filter(c => c.label.toLowerCase().includes(cmdInput.value.trim().toLowerCase()));
      if (e.key === "ArrowDown") {
        e.preventDefault();
        selectedCmdIdx = (selectedCmdIdx + 1) % Math.max(matches.length, 1);
        renderCmdResults(cmdInput.value);
      } else if (e.key === "ArrowUp") {
        e.preventDefault();
        selectedCmdIdx = (selectedCmdIdx - 1 + Math.max(matches.length, 1)) % Math.max(matches.length, 1);
        renderCmdResults(cmdInput.value);
      } else if (e.key === "Enter" && matches[selectedCmdIdx]) {
        e.preventDefault();
        const item = matches[selectedCmdIdx];
        closeCmdPalette();
        item.action();
      }
    });

    document.getElementById("openCmdBtn").addEventListener("click", openCmdPalette);
    document.getElementById("demoCmdShortcut").addEventListener("click", openCmdPalette);
    cmdModal.addEventListener("click", e => {
      if (e.target === cmdModal) closeCmdPalette();
    });

    // Global keyboard shortcuts
    document.addEventListener("keydown", e => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
        e.preventDefault();
        cmdModal.classList.contains("open") ? closeCmdPalette() : openCmdPalette();
      } else if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "b") {
        e.preventDefault();
        setWorkspaceView(activeView === "list" ? "board" : "list");
      } else if (e.key === "Escape") {
        closeCmdPalette();
        closeSignupModal();
      }
    });

    function openSignupModal(heading) {

      const ws = document.getElementById("wsNameInput").value.trim();
      closeSignupModal();
      e.target.reset();
      showToast(`Invite link dispatched for "${ws}"`);
    });

    // ---------- Utilities ----------
    let toastTimer;
    function showToast(msg) {
      const el = document.getElementById("toast");
      el.textContent = msg;
      el.classList.add("show");
      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => el.classList.remove("show"), 2600);
    }

    function scrollToEl(selector) {
      const el = document.querySelector(selector);
      if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
    }

    function escapeHtml(str) {
      return str
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;");
    }
