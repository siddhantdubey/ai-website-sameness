      /* Graceful SVG Fallback Generator (ensures zero broken images offline) */
      function createFallbackImage(title) {
        const safeTitle = (title || 'Archive Plate').replace(/[<>&'"]/g, '');
        const svg = `
          <svg xmlns="http://www.w3.org/2000/svg" width="800" height="1000" viewBox="0 0 800 1000">
            <rect width="800" height="1000" fill="#E6E2D8"/>
            <rect x="40" y="40" width="720" height="920" fill="none" stroke="#141413" stroke-opacity="0.14" stroke-width="1"/>
            <text x="400" y="490" text-anchor="middle" font-family="Georgia, serif" font-size="32" fill="#141413" fill-opacity="0.72">${safeTitle}</text>
            <text x="400" y="535" text-anchor="middle" font-family="sans-serif" font-size="13" letter-spacing="4" fill="#65635C">ELENA MOREAU ARCHIVE</text>
          </svg>
        `;
        return 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svg.trim());
      }

      function attachImageFallback(img, fallbackTitle) {
        img.addEventListener('error', () => {
          if (img.dataset.fallbackApplied) return;
          img.dataset.fallbackApplied = 'true';
          img.src = createFallbackImage(fallbackTitle);
          img.classList.add('loaded');
        });
        img.addEventListener('load', () => {
          img.classList.add('loaded');
        });
        if (img.complete && img.naturalWidth > 0) {
          img.classList.add('loaded');
        }
      }

      /* Apply fallback to static hero & portrait images */
      document.querySelectorAll('img[data-fallback-title]').forEach((img) => {
        attachImageFallback(img, img.dataset.fallbackTitle);
      });

      /* ----------------------------------------------------------------------
         2. Portfolio Filtering & View Rendering
         ---------------------------------------------------------------------- */
      const CATEGORIES = ['All', 'Editorial', 'Portraits', 'Architecture', 'Monochrome'];
      let activeCategory = 'All';
      let filteredItems = [...PORTFOLIO];

      const filterGroup = document.getElementById('filterGroup');
      const galleryGrid = document.getElementById('galleryGrid');
      const galleryIndex = document.getElementById('galleryIndex');
      const gridViewBtn = document.getElementById('gridViewBtn');
      const indexViewBtn = document.getElementById('indexViewBtn');

      function renderFilters() {
        filterGroup.innerHTML = '';
        CATEGORIES.forEach((cat) => {
          const count = cat === 'All'
            ? PORTFOLIO.length
            : PORTFOLIO.filter((item) => item.category === cat).length;

          const btn = document.createElement('button');
          btn.className = 'filter-btn' + (cat === activeCategory ? ' active' : '');
          btn.setAttribute('role', 'tab');
          btn.setAttribute('aria-selected', String(cat === activeCategory));
          btn.innerHTML = `${cat}<span class="filter-count">${count}</span>`;

          btn.addEventListener('click', () => {
            if (activeCategory === cat) return;
            activeCategory = cat;
            filteredItems = cat === 'All'
              ? [...PORTFOLIO]
              : PORTFOLIO.filter((item) => item.category === cat);
            renderFilters();
            renderGallery();
          });

          filterGroup.appendChild(btn);
        });
      }

      function renderGallery() {
        galleryGrid.innerHTML = '';
        galleryIndex.innerHTML = '';

        filteredItems.forEach((item, idx) => {

          const layoutClass = activeCategory === 'All' ? item.layout : (idx % 2 === 0 ? 'layout-half' : 'layout-half offset-top');
          article.className = `gallery-item ${layoutClass} ${item.aspect}`;
          article.setAttribute('tabindex', '0');

          img.loading = 'lazy';

          frame.appendChild(img);

              <h3 class="photo-title">${item.title}</h3>

            <span class="photo-year">${item.year}</span>

          article.appendChild(frame);
          article.appendChild(meta);

          const openHandler = () => openLightbox(idx);
          article.addEventListener('click', openHandler);
          article.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' || e.key === ' ') {
              e.preventDefault();
              openHandler();
            }
          });

          galleryGrid.appendChild(article);

          // 2) Index Row
          const row = document.createElement('div');
          row.className = 'index-row';
          row.setAttribute('tabindex', '0');
          row.setAttribute('role', 'button');
          row.setAttribute('aria-label', `Open ${item.title} in lightbox`);

          const thumbImg = document.createElement('img');
          thumbImg.src = item.thumb;
          thumbImg.alt = item.title;
          thumbImg.loading = 'lazy';
          thumbImg.className = 'index-thumb' + (item.monochrome ? ' is-monochrome' : '');
          attachImageFallback(thumbImg, item.title);

          row.appendChild(thumbImg);
          row.insertAdjacentHTML('beforeend', `
            <div>
              <div class="index-title">${item.title}</div>
              <div class="photo-sub">${item.client} &middot; ${item.location}</div>

            <div class="index-details">${item.camera}</div>

          row.addEventListener('click', openHandler);
          row.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' || e.key === ' ') {
              e.preventDefault();
              openHandler();
            }
          });

          galleryIndex.appendChild(row);

      /* View Switcher (Grid vs Index) */
      gridViewBtn.addEventListener('click', () => {

        galleryGrid.classList.add('hidden');

      });

      /* ----------------------------------------------------------------------
         3. Museum Lightbox (Keyboard + Touch Swipe Support)
         ---------------------------------------------------------------------- */
      const lightbox = document.getElementById('lightbox');
      const lightboxImg = document.getElementById('lightboxImg');
      const lightboxTitle = document.getElementById('lightboxTitle');
      const lightboxStory = document.getElementById('lightboxStory');
      const lightboxSpecs = document.getElementById('lightboxSpecs');
      const lightboxCounter = document.getElementById('lightboxCounter');
      const lightboxClose = document.getElementById('lightboxClose');
      const lightboxPrev = document.getElementById('lightboxPrev');
      const lightboxNext = document.getElementById('lightboxNext');
      const lightboxStage = document.getElementById('lightboxStage');

      let currentLightboxIndex = 0;
      let lastFocusedEl = null;

      function updateLightboxView() {
        const item = filteredItems[currentLightboxIndex];
        if (!item) return;

        lightboxImg.style.opacity = '0.25';
        lightboxImg.classList.toggle('is-monochrome', item.monochrome);

        const nextImg = new Image();
        nextImg.onload = () => {
          lightboxImg.src = item.src;
          lightboxImg.alt = `${item.title} — ${item.story}`;
          lightboxImg.style.opacity = '1';
        };
        nextImg.onerror = () => {
          lightboxImg.src = createFallbackImage(item.title);
          lightboxImg.alt = item.title;
          lightboxImg.style.opacity = '1';
        };
        nextImg.src = item.src;

        lightboxTitle.textContent = `${item.title} — ${item.client}`;
        lightboxStory.textContent = item.story;
        lightboxSpecs.textContent = `${item.location} · ${item.camera} · ${item.year}`;

        const num = String(currentLightboxIndex + 1).padStart(2, '0');
        const total = String(filteredItems.length).padStart(2, '0');
        lightboxCounter.textContent = `${num} / ${total}`;
      }

      function openLightbox(index) {
        lastFocusedEl = document.activeElement;
        currentLightboxIndex = index;
        updateLightboxView();
        lightbox.classList.add('open');
        lightbox.setAttribute('aria-hidden', 'false');
        document.body.classList.add('no-scroll');
        lightboxClose.focus();
      }

      function closeLightbox() {
        lightbox.classList.remove('open');
        lightbox.setAttribute('aria-hidden', 'true');
        document.body.classList.remove('no-scroll');
        if (lastFocusedEl && typeof lastFocusedEl.focus === 'function') {
          lastFocusedEl.focus();
        }
      }

      function stepLightbox(delta) {
        currentLightboxIndex = (currentLightboxIndex + delta + filteredItems.length) % filteredItems.length;
        updateLightboxView();
      }

      lightboxClose.addEventListener('click', closeLightbox);
      lightboxPrev.addEventListener('click', () => stepLightbox(-1));
      lightboxNext.addEventListener('click', () => stepLightbox(1));

      lightboxStage.addEventListener('click', (e) => {
        if (e.target === lightboxStage) closeLightbox();
      });

      document.addEventListener('keydown', (e) => {
        if (!lightbox.classList.contains('open')) return;
        if (e.key === 'Escape') closeLightbox();
        if (e.key === 'ArrowLeft') stepLightbox(-1);
        if (e.key === 'ArrowRight') stepLightbox(1);
      });

      /* Touch swipe support for mobile lightbox */
      let touchStartX = 0;
      lightboxStage.addEventListener('touchstart', (e) => {
        touchStartX = e.changedTouches[0].screenX;
      }, { passive: true });

      lightboxStage.addEventListener('touchend', (e) => {
        const diff = e.changedTouches[0].screenX - touchStartX;
        if (Math.abs(diff) > 45) {
          stepLightbox(diff > 0 ? -1 : 1);
        }
      }, { passive: true });

      /* Hero featured photo also opens the lightbox */
      const heroFeatureTrigger = document.getElementById('heroFeatureTrigger');
      function openHeroLightbox() {
        activeCategory = 'All';
        filteredItems = [...PORTFOLIO];
        renderFilters();
        renderGallery();
        openLightbox(0);
      }
      heroFeatureTrigger.addEventListener('click', openHeroLightbox);
      heroFeatureTrigger.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          openHeroLightbox();
        }
      });

      /* ----------------------------------------------------------------------
         4. Header Scroll, Progress Bar & Mobile Navigation
         ---------------------------------------------------------------------- */
      const siteHeader = document.getElementById('siteHeader');
      const scrollProgress = document.getElementById('scrollProgress');
      const menuToggle = document.getElementById('menuToggle');
      const mobileDrawer = document.getElementById('mobileDrawer');

      function onScroll() {
        const scrollTop = window.scrollY || document.documentElement.scrollTop;
        siteHeader.classList.toggle('scrolled', scrollTop > 24);

        const docHeight = document.documentElement.scrollHeight - window.innerHeight;
        const pct = docHeight > 0 ? Math.min(100, (scrollTop / docHeight) * 100) : 0;
        scrollProgress.style.width = pct + '%';
      }

      window.addEventListener('scroll', onScroll, { passive: true });
      onScroll();

      function setMobileMenu(isOpen) {
        menuToggle.setAttribute('aria-expanded', String(isOpen));
        mobileDrawer.classList.toggle('open', isOpen);
        mobileDrawer.setAttribute('aria-hidden', String(!isOpen));
        document.body.classList.toggle('no-scroll', isOpen);
      }

      menuToggle.addEventListener('click', () => {
        const isOpen = menuToggle.getAttribute('aria-expanded') === 'true';
        setMobileMenu(!isOpen);
      });

      document.querySelectorAll('.mobile-link').forEach((link) => {
        link.addEventListener('click', () => setMobileMenu(false));
      });

      /* ----------------------------------------------------------------------
         5. Scroll Reveal Observer
         ---------------------------------------------------------------------- */
      const revealEls = document.querySelectorAll('.reveal');
      if ('IntersectionObserver' in window) {
        const observer = new IntersectionObserver((entries, obs) => {
          entries.forEach((entry) => {
            if (entry.isIntersecting) {
              entry.target.classList.add('visible');
              obs.unobserve(entry.target);
            }
          });
        }, { threshold: 0.1, rootMargin: '0px 0px -40px 0px' });

        revealEls.forEach((el) => observer.observe(el));
      } else {
        revealEls.forEach((el) => el.classList.add('visible'));
      }

      /* ----------------------------------------------------------------------
         6. Copy Email, Live Paris Clock & Inquiry Form
         ---------------------------------------------------------------------- */
      const copyEmailBtn = document.getElementById('copyEmailBtn');
      const copyEmailText = document.getElementById('copyEmailText');

      copyEmailBtn.addEventListener('click', async () => {
        const email = 'studio@elenamoreau.com';
        try {
          await navigator.clipboard.writeText(email);
          copyEmailText.textContent = 'Copied to clipboard';
        } catch {
          copyEmailText.textContent = email;
        }
        setTimeout(() => {
          copyEmailText.textContent = 'Copy email address';
        }, 2200);
      });

      /* Live Paris Time */
      const parisClock = document.getElementById('parisClock');
      function updateParisTime() {
        try {
          const formatter = new Intl.DateTimeFormat('en-GB', {
            timeZone: 'Europe/Paris',
            hour: '2-digit',
            minute: '2-digit',
            hour12: false,
            timeZoneName: 'short'
          });
          parisClock.textContent = formatter.format(new Date());
        } catch {
          parisClock.textContent = 'Paris, CET';
        }
      }
      updateParisTime();
      setInterval(updateParisTime, 30000);

      document.getElementById('currentYear').textContent = new Date().getFullYear();

      /* Inquiry Form Validation & Feedback */
      const inquiryForm = document.getElementById('inquiryForm');
      const formStatus = document.getElementById('formStatus');

      inquiryForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const name = inquiryForm.name.value.trim();
        const email = inquiryForm.email.value.trim();
        const message = inquiryForm.message.value.trim();
        const projectType = inquiryForm.projectType.value;

        if (!name || !email || !message || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
          formStatus.textContent = 'Please complete your name, a valid email address, and a brief description of your project.';
          formStatus.classList.add('visible');
          return;
        }

        formStatus.textContent = `Thank you, ${name}. Your ${projectType.toLowerCase()} inquiry has been received—Elena's studio will reply to ${email} within 24 hours.`;
        formStatus.classList.add('visible');
        inquiryForm.reset();
      });

      /* Initialize Portfolio */
      renderFilters();
      renderGallery();
    })();
  </script>
