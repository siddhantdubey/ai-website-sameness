  <script>
    const stages = ['Up next', 'In progress', 'Done'];
    const example = {
      title: 'Autumn issue', example: true,
      tasks: [
        { title: 'Select the cover photograph', owner: 'MK', stage: 0 },
        { title: 'Edit the studio interview', owner: 'JL', stage: 1 },
        { title: 'Approve the flatplan', owner: 'AS', stage: 2 }
      ]
    };
    let board = example;
    try {
      const saved = JSON.parse(localStorage.getItem('relay-board'));
      if (saved && typeof saved.title === 'string' && Array.isArray(saved.tasks)) board = saved;
    } catch {}
    const get = id => document.getElementById(id);
    const dialog = get('editor');
    let mode;
    function element(tag, className, text) {
      const node = document.createElement(tag);
      node.className = className;
      node.textContent = text;
      return node;
    }
    function save() {
      try { localStorage.setItem('relay-board', JSON.stringify(board)); } catch {}
      render();
    }
    function render() {
      get('board-title').textContent = board.title;
      get('board-label').textContent = board.example ? 'Example project' : 'Your project';
      get('lanes').replaceChildren();
      stages.forEach((stage, index) => {
        const tasks = board.tasks.filter(task => task.stage === index);
        const lane = element('section', 'lane', '');
        const heading = element('h3', 'lane-head', '');
        heading.append(element('span', 'dot', ''), stage, element('span', 'count', tasks.length));
        lane.append(heading);
        tasks.forEach(task => {
          const card = element('article', 'card', '');
          const foot = element('div', 'card-foot', '');
          foot.append(element('span', 'person', task.owner || 'Unassigned'));
          if (index < 2) {
            const advance = element('button', 'small', index === 0 ? 'Start →' : 'Finish ✓');
            advance.style.cssText = 'border:0;padding:4px;background:transparent;font:inherit';
            advance.setAttribute('aria-label', `${advance.textContent} ${task.title}`);
            advance.onclick = () => { task.stage++; save(); };
            foot.append(advance);
          } else foot.append('Complete ✓');
          card.append(element('h4', '', task.title), foot);
          lane.append(card);
        });
        get('lanes').append(lane);
      });
    }
    document.querySelectorAll('[data-open]').forEach(button => {
      button.onclick = () => {
        mode = button.dataset.open;
        const project = mode === 'project';
        get('form').reset();
        get('dialog-title').textContent = project ? 'Start a project' : 'Add a task';
        get('name-label').textContent = project ? 'Project name' : 'Task name';
        get('owner-field').hidden = project;
        get('submit').textContent = project ? 'Create project' : 'Add task';
        if (project && !board.example) get('dialog-title').textContent = 'Replace current project';
        dialog.showModal();
      };
    });
    get('cancel').onclick = () => dialog.close();
    get('form').onsubmit = event => {
      event.preventDefault();
      const title = get('name').value.trim();
      if (!title) { get('name').focus(); return; }
      if (mode === 'project') board = { title, tasks: [], example: false };
      else board.tasks.push({ title, stage: 0, owner: get('owner').value.trim().toUpperCase() });
      save();
      dialog.close();
      get('project').scrollIntoView();
    };
    render();
  </script>
