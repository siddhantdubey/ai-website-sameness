<script>
(function () {
  // ---- Restaurant settings: edit these ----
  var TIMEZONE = "America/New_York";
  var EMAIL = "ciao@dapina.example";
  // Opening hours in 24h, keyed by day (0 = Sunday). null = closed.
  var HOURS = { 0: [12, 21], 1: null, 2: [17, 22], 3: [17, 22], 4: [17, 22], 5: [17, 23], 6: [17, 23] };
  var DAY_NAMES = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];

  function fmtHour(h) {
    var whole = Math.floor(h), mins = Math.round((h - whole) * 60);
    var suffix = whole >= 12 ? "pm" : "am";
    var twelve = whole % 12 === 0 ? 12 : whole % 12;
    return twelve + (mins ? ":" + String(mins).padStart(2, "0") : "") + " " + suffix;
  }

  // Current day and time at the restaurant, wherever the visitor is.
  function restaurantNow() {
    try {
      var parts = new Intl.DateTimeFormat("en-US", {
        timeZone: TIMEZONE, weekday: "long", hour: "numeric", minute: "numeric", hourCycle: "h23"
      }).formatToParts(new Date());
      var get = function (t) { return parts.filter(function (p) { return p.type === t; })[0].value; };
      return { day: DAY_NAMES.indexOf(get("weekday")), hour: Number(get("hour")) + Number(get("minute")) / 60 };
    } catch (e) {
      var d = new Date();
      return { day: d.getDay(), hour: d.getHours() + d.getMinutes() / 60 };
    }
  }

  var now = restaurantNow();

  // ---- Open / closed status ----
  var statusEl = document.getElementById("status");
  var statusText = document.getElementById("status-text");
  var today = HOURS[now.day];

  if (today && now.hour >= today[0] && now.hour < today[1]) {
    statusEl.classList.add("open");
    statusText.textContent = "Open now · kitchen until " + fmtHour(today[1]);
  } else if (today && now.hour < today[0]) {
    statusText.textContent = "Opens today at " + fmtHour(today[0]);
  } else {
    for (var i = 1; i <= 7; i++) {
      var next = (now.day + i) % 7;
      if (HOURS[next]) {
        statusText.textContent = "Closed now · opens " + (i === 1 ? "tomorrow" : DAY_NAMES[next]) + " at " + fmtHour(HOURS[next][0]);
        break;
      }
    }
  }

  // ---- Highlight today in the specials and hours ----
  document.querySelectorAll('[data-day="' + now.day + '"]').forEach(function (el) {
    el.classList.add("today");
  });

  // ---- Booking form ----
  var form = document.getElementById("book");
  var dateEl = document.getElementById("f-date");
  var timeEl = document.getElementById("f-time");
  var submitEl = document.getElementById("f-submit");
  var msgEl = document.getElementById("form-msg");

  function isoLocal(d) {
    return d.getFullYear() + "-" + String(d.getMonth() + 1).padStart(2, "0") + "-" + String(d.getDate()).padStart(2, "0");
  }

  function fillTimes() {
    var picked = dateEl.value ? new Date(dateEl.value + "T12:00") : null;
    var hours = picked ? HOURS[picked.getDay()] : null;
    timeEl.innerHTML = "";
    msgEl.textContent = "";

    if (!picked || !hours) {
      var opt = document.createElement("option");
      opt.textContent = picked ? "Closed" : "Pick a date";
      opt.value = "";
      timeEl.appendChild(opt);
      timeEl.disabled = true;
      submitEl.disabled = true;
      if (picked) msgEl.textContent = "We're closed on " + DAY_NAMES[picked.getDay()] + "s. Try another day?";
      return;
    }

    timeEl.disabled = false;
    submitEl.disabled = false;
    // Last seating is 90 minutes before close.
    for (var t = hours[0]; t <= hours[1] - 1.5; t += 0.5) {
      var o = document.createElement("option");
      o.textContent = fmtHour(t);
      o.value = fmtHour(t);
      if (t === 19) o.selected = true;
      timeEl.appendChild(o);
    }
  }

  // Default to the next day we're open.
  var start = new Date();
  start.setDate(start.getDate() + 1);
  while (!HOURS[start.getDay()]) start.setDate(start.getDate() + 1);
  dateEl.min = isoLocal(new Date());
  dateEl.value = isoLocal(start);
  fillTimes();
  dateEl.addEventListener("change", fillTimes);

  form.addEventListener("submit", function (e) {
    e.preventDefault();
    var name = form.elements.name.value.trim();
    if (!name) {
      msgEl.textContent = "We'll need a name for the table.";
      form.elements.name.focus();
      return;
    }
    if (!dateEl.value || !timeEl.value) {
      msgEl.textContent = "Pick a date and time first.";
      dateEl.focus();
      return;
    }

    var when = new Date(dateEl.value + "T12:00").toLocaleDateString("en-US", { weekday: "long", month: "long", day: "numeric" });
    var party = form.elements.party.value;
    var subject = "Table request: " + when + ", " + timeEl.value + ", " + party + (party === "1" ? " guest" : " guests");
    var lines = [
      "Name: " + name,
      "Date: " + when,
      "Time: " + timeEl.value,
      "Guests: " + party
    ];
    if (form.elements.phone.value.trim()) lines.push("Phone: " + form.elements.phone.value.trim());
    if (form.elements.note.value.trim()) lines.push("", "Note: " + form.elements.note.value.trim());

    window.location.href = "mailto:" + EMAIL + "?subject=" + encodeURIComponent(subject) + "&body=" + encodeURIComponent(lines.join("\n"));
    msgEl.textContent = "Your email app should open with the request ready to send. Nothing happening? Ring us on (555) 014-0187.";
  });

  document.getElementById("year").textContent = new Date().getFullYear();
})();

The landing page is above as one HTML file with no external fonts, images, or scripts; save it as `index.html` and open it in a browser. I haven't opened it in a browser myself, so give it a look at desktop and phone widths.

The restaurant is invented: "Da Pina", a family trattoria with a striped-awning shopfront look. The page has the story, a four-course menu with prices, a dish-of-the-day list, hours, and a booking form.

- **Live status:** the page shows whether the restaurant is open right now and highlights today in the specials and hours.
- **Booking form:** there is no backend, so submitting opens the visitor's email app with the request filled in. It blocks closed days and only offers times within opening hours.
- **Placeholders to replace:** the name, address, phone number, email, menu, prices, and family story are all made up.
- **Where the settings live:** timezone, email, and opening hours sit in a short settings block at the top of the script. The hours are also written out in the hours table and the search-engine data in the `<head>`, so change them in all three places.
