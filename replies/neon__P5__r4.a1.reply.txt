      <div class="container">
        <div class="section-header">
          <div><span class="section-index">01 / Restraint</span></div>
          <div>
            <h2 class="section-title">What we deliberately left out of Trestle.</h2>
            <p class="section-lead">
              Most project tools turn into a second job because they are built to satisfy reporting
              layers rather than the people shipping the work. We designed Trestle by subtraction.
            </p>
          </div>
        </div>

        <div class="ledger-table">
          <div class="ledger-row ledger-head">
            <span>No.</span>
            <span>Omitted ceremony</span>
            <span>What Trestle does instead</span>
          </div>

          <div class="ledger-row">
            <span class="ledger-num">01</span>
            <div>
              <div class="ledger-omitted">Story points &amp; velocity charts</div>
              <p class="ledger-omitted-note">Abstract numbers inevitably get compared across people and gamed by quarter-end.</p>
            </div>
            <div>
              <div class="ledger-instead">Appetite budgets in real days</div>
              <p class="ledger-instead-note">Decide how many days a problem is worth up front, then shape the scope to fit inside that ceiling.</p>
            </div>
          </div>

          <div class="ledger-row">
            <span class="ledger-num">02</span>
            <div>
              <div class="ledger-omitted">Eleven-column Kanban graveyards</div>
              <p class="ledger-omitted-note">Cards sit in "Blocked - External" for four months until nobody remembers why they exist.</p>
            </div>
            <div>
              <div class="ledger-instead">Four honest states &amp; automatic cycle wrap</div>
              <p class="ledger-instead-note">Queued, Active, Review, Done. Unfinished work isn’t quietly rolled over—it is re-decided consciously.</p>
            </div>
          </div>

          <div class="ledger-row">
            <span class="ledger-num">03</span>
            <div>
              <div class="ledger-omitted">Infinite threaded ticket debates</div>
              <p class="ledger-omitted-note">Critical decisions get buried under comment #47 where nobody reads before implementing.</p>
            </div>
            <div>
              <div class="ledger-instead">Living plain-text briefs with explicit boundaries</div>
              <p class="ledger-instead-note">Every deliverable has one readable spec with an explicit "Out of scope" block that stays pinned at the top.</p>
            </div>
          </div>

          <div class="ledger-row">
            <span class="ledger-num">04</span>
            <div>
              <div class="ledger-omitted">Real-time notification anxiety</div>
              <p class="ledger-omitted-note">Constant pings turn deep engineering mornings into fragmented chat support.</p>
            </div>
            <div>
              <div class="ledger-instead">Batched 9am &amp; 2pm written digests</div>
              <p class="ledger-instead-note">Updates arrive twice a day as one calm sheet, unless someone explicitly flags a production blocker.</p>
            </div>
          </div>

              <li>Automatic snapshots let you replay how a cycle moved without interrogating anyone.</li>

              '<circle cx="' + pt.x.toFixed(1) + '" cy="' + pt.y.toFixed(1) + '" r="7" fill="' + d.color + '" stroke="#fbf9f4" stroke-width="2" />' +
            '</g>'
          );
        }).join('');

        controls.innerHTML = hillDeliverables.map(function (d, i) {
          return (
            '<div class="hill-item">' +
              '<div class="hill-item-left">' +
                '<span class="hill-swatch" style="background:' + d.color + ';"></span>' +
                '<strong>' + d.name + '</strong>' +
                '<span class="hill-phase">&middot; ' + phaseLabel(d.pos) + '</span>' +
              '</div>' +
              '<button type="button" class="hill-nudge" data-hill-idx="' + i + '">' + (d.pos >= 0.92 ? 'Reset' : 'Advance →') + '</button>' +
            '</div>'
          );
        }).join('');
      }

      document.getElementById('hill-controls').addEventListener('click', function (e) {
        const btn = e.target.closest('[data-hill-idx]');
        if (!btn) return;
        const idx = Number(btn.getAttribute('data-hill-idx'));
        const item = hillDeliverables[idx];
        item.pos = item.pos >= 0.92 ? 0.16 : Math.min(0.95, item.pos + 0.18);
        renderHill();
      });

      renderHill();

        btn.addEventListener('click', function () {
          const mode = btn.getAttribute('data-brief-mode');
          document.querySelectorAll('[data-brief-mode]').forEach(function (b) {
            b.classList.toggle('is-active', b === btn);
          });
          document.getElementById('brief-rendered').style.display = mode === 'rendered' ? 'block' : 'none';
          document.getElementById('brief-source').style.display = mode === 'source' ? 'block' : 'none';
        });

      const loadSimBtn = document.getElementById('load-sim-btn');
      const devonLabel = document.getElementById('devon-load-label');
      const devonBar = document.getElementById('devon-load-bar');
      const loadWarn = document.getElementById('load-warning');

      loadSimBtn.addEventListener('click', function () {
        urgentAdded = !urgentAdded;
        loadSimBtn.classList.toggle('is-on', urgentAdded);
        loadSimBtn.textContent = urgentAdded ? 'Remove request' : '+ Add request';
        devonLabel.textContent = urgentAdded ? '5.0d / 4.0d (over)' : '3.0d / 4.0d';
        devonBar.style.width = urgentAdded ? '100%' : '75%';
        devonBar.classList.toggle('is-over', urgentAdded);
        loadWarn.classList.toggle('is-visible', urgentAdded);
      });

      let billing = 'annual';
      const slider = document.getElementById('team-slider');
      const unitPriceEl = document.getElementById('unit-price');
      const unitLabelEl = document.getElementById('unit-label');
      const teamSizeValEl = document.getElementById('team-size-val');
      const teamTotalEl = document.getElementById('team-total');

      function updatePricing() {
        const seats = Number(slider.value);
        const rate = billing === 'annual' ? 11 : 14;
        unitPriceEl.textContent = '$' + rate;
        unitLabelEl.textContent = billing === 'annual'
          ? 'per active seat / month, billed annually'
          : 'per active seat / month, billed monthly';
        teamSizeValEl.textContent = seats + ' people';
        if (seats <= 2) {
          teamTotalEl.textContent = '$0 / mo (Free tier)';
        } else {
          teamTotalEl.textContent = '$' + (seats * rate) + ' / mo';
        }
      }

        btn.addEventListener('click', function () {

          document.querySelectorAll('[data-billing]').forEach(function (b) {
            b.classList.toggle('is-active', b === btn);
          });
          updatePricing();

        if (!input.value.trim()) return;
        msg.textContent = 'Subscribed ' + input.value.trim() + '. Next issue goes out Dec 1.';

      /* ---------------- Modals ---------------- */
      function closeAllModals() {
        document.querySelectorAll('.modal-backdrop.is-open').forEach(function (m) {
          m.classList.remove('is-open');
        });
      }

      function openModal(name) {
        closeAllModals();
        const el = document.getElementById('modal-' + name);
        if (!el) return;
        el.classList.add('is-open');
        const firstInput = el.querySelector('input, button.modal-close');
        if (firstInput) firstInput.focus();
      }

      document.querySelectorAll('[data-open-modal]').forEach(function (trigger) {
        trigger.addEventListener('click', function (e) {
          e.preventDefault();
          openModal(trigger.getAttribute('data-open-modal'));
        });
      });

      document.querySelectorAll('[data-close-modal]').forEach(function (btn) {
        btn.addEventListener('click', closeAllModals);
      });

      document.querySelectorAll('.modal-backdrop').forEach(function (backdrop) {
        backdrop.addEventListener('click', function (e) {
          if (e.target === backdrop) closeAllModals();
        });
      });

      document.addEventListener('keydown', function (e) {
        if (e.key === 'Escape') closeAllModals();
      });

          });
        });
      });

        const name = wsNameInput.value.trim() || 'Your studio';
        const email = document.getElementById('ws-email').value.trim();
        document.getElementById('ws-form').style.display = 'none';

        document.getElementById('ws-confirm-copy').textContent =
          'We prepared ' + wsSlugEl.textContent + ' (' + name + ') with ' + selectedCycle +
          ' cycles and sent a sign-in link to ' + email + '.';
        confirm.style.display = 'block';
      });
    })();
