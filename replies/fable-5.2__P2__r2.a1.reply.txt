(function () {
  var now = new Date();
  var day = now.getDay();
  var mins = now.getHours() * 60 + now.getMinutes();

  // Opening hours in minutes from midnight, indexed by Date.getDay()
  var hours = { 0: [900, 1260], 1: null, 2: [1020, 1320], 3: [1020, 1320], 4: [1020, 1320], 5: [1020, 1380], 6: [1020, 1380] };
  var names = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];

  function clock(m) {
    var h = Math.floor(m / 60), mm = m % 60, h12 = h % 12 || 12;
    return h12 + (mm ? ':' + String(mm).padStart(2, '0') : '') + (h < 12 ? ' am' : ' pm');
  }

  // Open / closed status
  var status = document.getElementById('status');
  var label = status.querySelector('span');
  var today = hours[day];
  if (today && mins >= today[0] && mins < today[1]) {
    status.classList.add('open');
    label.textContent = 'Open now · kitchen closes at ' + clock(today[1]);
  } else if (today && mins < today[0]) {
    status.classList.add('closed');
    label.textContent = 'Opens today at ' + clock(today[0]);
  } else {
    var next = day, i;
    for (i = 1; i <= 7; i++) { next = (day + i) % 7; if (hours[next]) break; }
    status.classList.add('closed');
    label.textContent = 'Closed now · opens ' + (i === 1 ? 'tomorrow' : names[next]) + ' at ' + clock(hours[next][0]);
  }

  // Highlight today in the week strip and hours table
  document.querySelectorAll('#days li, #hours tr').forEach(function (el) {
    if (Number(el.dataset.day) === day) el.classList.add('today');
  });

  // Tonight's card
  var specials = {
    0: ['Sunday supper', 'Four courses served family style: antipasti, pasta, a roast and dessert. From 3 pm.', '$45', 'Per person · kids $18'],
    2: ['Pasta e fagioli', 'A big bowl of pasta and bean soup with grilled bread and a glass of house red.', '$19', 'Tuesdays only'],
    3: ['Wine night', 'Order two mains and every bottle on the list is half price. Luca will help you choose.', '½ off', 'Wednesdays only'],
    4: ['Gnocchi alla sorrentina', 'Potato gnocchi baked in tomato with mozzarella and basil, straight from the oven.', '$20', 'Thursdays only'],
    5: ['Cioppino', 'Clams, mussels and shrimp in a garlicky tomato broth with grilled bread for mopping.', '$29', 'Fridays only'],
    6: ['Osso buco', 'Veal shank braised until it falls off the bone, on saffron risotto.', '$34', 'Saturdays only']
  };
  var s = specials[day];
  if (s) {
    document.getElementById('tonight-label').textContent = day === 0 ? "Today's special" : "Tonight's special";
    document.getElementById('tonight-name').textContent = s[0];
    document.getElementById('tonight-desc').textContent = s[1];
    document.getElementById('tonight-price').textContent = s[2];
    document.getElementById('tonight-note').textContent = s[3];
  }

  document.getElementById('year').textContent = now.getFullYear();

  // Booking form
  var form = document.getElementById('book');
  var msg = document.getElementById('msg');
  var dateInput = document.getElementById('date');
  var iso = function (d) {
    return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');
  };
  dateInput.min = iso(now);

  function fail(text, field) {
    msg.className = 'msg err';
    msg.textContent = text;
    if (field) field.focus();
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    var name = form.elements.name, phone = form.elements.phone, time = form.elements.time;
    if (!name.value.trim()) return fail('Please tell us your name.', name);
    if (phone.value.replace(/\D/g, '').length < 7) return fail('Please add a phone number we can reach you on.', phone);
    if (!dateInput.value) return fail('Please pick a date.', dateInput);
    if (dateInput.value < dateInput.min) return fail('That date has already passed.', dateInput);

    var parts = dateInput.value.split('-');
    var chosen = new Date(parts[0], parts[1] - 1, parts[2]);
    var h = hours[chosen.getDay()];
    if (!h) return fail('We are closed on Mondays. Please pick another day.', dateInput);
    if (!time.value) return fail('Please pick a time.', time);

    var t = time.value.match(/(\d+):(\d+)/);
    var slot = (Number(t[1]) % 12 + 12) * 60 + Number(t[2]);
    if (slot < h[0] && chosen.getDay() !== 0) return fail('We open at ' + clock(h[0]) + ' that day.', time);
    if (slot > h[1] - 60) return fail('Our last seating on ' + names[chosen.getDay()] + ' is ' + clock(h[1] - 60) + '.', time);

    var nice = chosen.toLocaleDateString(undefined, { weekday: 'long', month: 'long', day: 'numeric' });
    msg.className = 'msg ok';
    msg.textContent = 'Grazie, ' + name.value.trim().split(' ')[0] + '. We have your request for ' +
      form.elements.guests.value + ' on ' + nice + ' at ' + time.value + ' and will call to confirm.';
    form.reset();
  });

The page is a single HTML file with no external fonts, images or scripts, so it works offline once saved as `index.html`. I have not opened it in a browser.

- **Restaurant details:** the name (Da Ottavio), address, phone, email, family story, menu, prices and hours are all placeholders I invented, to swap for the real ones.
- **Booking form:** it validates input (including rejecting Mondays and times past the last seating) and shows a confirmation message, but it does not send the request anywhere. It needs connecting to a booking service or email handler before going live.
- **Live touches:** the open/closed line, the highlighted day and the special on the hero card follow the visitor's clock. The hours are set in both the hours table and the `hours` object in the script, so change them in both places.
