    const dialog = document.getElementById('signup-dialog');
    document.querySelectorAll('[data-signup]').forEach(button => {
      button.addEventListener('click', () => dialog.showModal());
    });
    document.querySelector('.dialog-close').addEventListener('click', () => dialog.close());
    dialog.addEventListener('click', event => {
      const box = dialog.getBoundingClientRect();
      if (event.clientX < box.left || event.clientX > box.right ||
          event.clientY < box.top || event.clientY > box.bottom) dialog.close();
    });
    document.getElementById('signup-form').addEventListener('submit', event => {
      event.preventDefault();
      document.querySelector('.form-message').textContent =
        'You’re all set for this demo. In the live app, this is where your workspace would begin.';
    });

    const tabs = [...document.querySelectorAll('[data-view]')];
    const taskView = document.getElementById('task-view');
    function selectView(tab) {
      tabs.forEach(item => {
        const active = item === tab;
        item.setAttribute('aria-selected', String(active));
        item.tabIndex = active ? 0 : -1;
      });
      taskView.classList.toggle('list-mode', tab.dataset.view === 'list');
      taskView.setAttribute('aria-labelledby', tab.id);
    }
    tabs.forEach((tab, index) => {
      tab.addEventListener('click', () => selectView(tab));
      tab.addEventListener('keydown', event => {
        let next;
        if (event.key === 'ArrowRight' || event.key === 'ArrowLeft') next = tabs[(index + 1) % tabs.length];
        if (event.key === 'Home') next = tabs[0];
        if (event.key === 'End') next = tabs[tabs.length - 1];
        if (next) {
          event.preventDefault();
          selectView(next);
          next.focus();
        }
      });
    });

    document.querySelectorAll('.task-toggle').forEach(button => {
      const task = button.closest('.task');
      task.dataset.originalColumn = task.closest('.column').dataset.column === 'done'
        ? 'todo' : task.closest('.column').dataset.column;

      button.addEventListener('click', () => {
        const wasDone = button.getAttribute('aria-pressed') === 'true';
        const title = task.querySelector('h4').textContent;
        const destination = wasDone ? task.dataset.originalColumn : 'done';
        document.querySelector('[data-column="' + destination + '"]').appendChild(task);
        button.setAttribute('aria-pressed', String(!wasDone));
        button.setAttribute('aria-label', (wasDone ? 'Complete ' : 'Reopen ') + title);
        button.textContent = wasDone ? '' : '✓';
        document.querySelectorAll('.column').forEach(column => {
          column.querySelector('.count').textContent = column.querySelectorAll('.task').length;
        });
        button.focus({ preventScroll: true });
        document.getElementById('board-announcement').textContent =
          title + (wasDone ? ' moved back to its original column.' : ' completed. Nice forward motion.');
      });
    });
  </script>
