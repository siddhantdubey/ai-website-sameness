  var hours = { 0: [12, 20.5], 1: null, 2: [17, 21.5], 3: [17, 21.5], 4: [17, 21.5], 5: [17, 22.5], 6: [12, 22.5] };

  function fmt(h) {
    var hr = Math.floor(h), min = Math.round((h - hr) * 60);
    var suffix = hr >= 12 ? 'pm' : 'am';
    var h12 = hr % 12 || 12;
    return h12 + (min ? ':' + (min < 10 ? '0' : '') + min : '') + ' ' + suffix;
  }

  var now = new Date();
  var day = now.getDay();
  var t = now.getHours() + now.getMinutes() / 60;
  var today = hours[day];
  var el = document.getElementById('status');
  var text, open = false;

  if (today && t >= today[0] && t < today[1]) {
    open = true;
    text = 'Open now, kitchen until ' + fmt(today[1]);
  } else if (today && t < today[0]) {
    text = 'Closed now, opens today at ' + fmt(today[0]);
  } else {
    for (var i = 1; i <= 7; i++) {
      var d = (day + i) % 7;
      if (hours[d]) {

        break;
      }
    }
  }
  el.textContent = text;
  el.classList.add(open ? 'open' : 'closed');

The landing page is written to `index.html` in the working directory, as one file with no external fonts, images or scripts. I haven't opened it in a browser, so the layout is untested.

The restaurant is an invented one, Trattoria Pina on Alder Street, with a look taken from old blue pasta-box packaging: cobalt, semolina yellow and tomato red.

- **Hero:** oversized wordmark, a short pitch, call and menu buttons, and an inline SVG plate of spaghetti.
- **Menu:** four courses (antipasti, primi, secondi, dolci) with prices and vegetarian markers.
- **Story:** a short three-generation family history.
- **Weekly nights:** Tuesday no-corkage, Thursday gnocchi, Sunday sugo lunch.
- **Visit:** address, hours table, a drawn street map and a call-to-book button.

A small script shows "Open now" or the next opening time from the visitor's clock and highlights today's row in the hours table. The page is responsive and honours reduced-motion settings.

Everything specific is placeholder: the name, address, phone number, email, menu, prices and hours all need swapping for the real restaurant's details. The hours live in two places, the table and the `hours` object in the script, so change both. Booking is by phone link only; there is no reservation form.
