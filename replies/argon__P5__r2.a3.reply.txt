          renderTable();
        });
      });
    }

    // Filter tabs
    document.querySelectorAll('.view-tab').forEach(tab => {
      tab.addEventListener('click', () => {
        document.querySelectorAll('.view-tab').forEach(t => t.classList.remove('active'));
        tab.classList.add('active');
        currentFilter = tab.getAttribute('data-filter');
        renderTable();
      });
    });

    // CLI Copy button
    const copyCli = document.getElementById('copyCli');
    const copyStatus = document.getElementById('copyStatus');
    copyCli.addEventListener('click', () => {
      navigator.clipboard?.writeText('brew install datum-hq/tap/datum');
      copyStatus.textContent = 'COPIED ✓';
      setTimeout(() => { copyStatus.textContent = 'COPY'; }, 1800);
    });

    // Command Palette (Cmd+K)
    const cmdBackdrop = document.getElementById('cmdBackdrop');
    const cmdInput = document.getElementById('cmdInput');
    const openCmdBtn = document.getElementById('openCmdBtn');

    function openCmd() {
      cmdBackdrop.classList.add('open');
      cmdInput.value = '';
      cmdInput.focus();
    }
    function closeCmd() {
      cmdBackdrop.classList.remove('open');
    }

    openCmdBtn.addEventListener('click', openCmd);
    cmdBackdrop.addEventListener('click', (e) => {
      if (e.target === cmdBackdrop) closeCmd();
    });

    window.addEventListener('keydown', (e) => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
        e.preventDefault();
        cmdBackdrop.classList.contains('open') ? closeCmd() : openCmd();
      } else if (e.key === 'Escape') {
        closeCmd();
      }
    });

    document.querySelectorAll('.cmd-item').forEach(item => {
      item.addEventListener('click', () => {
        const action = item.getAttribute('data-action');
        closeCmd();
        if (action === 'filter-blocked') {
          document.querySelector('[data-filter="blocked"]').click();
          document.getElementById('workbench').scrollIntoView({ behavior: 'smooth' });
        } else if (action === 'copy-brew') {
          copyCli.click();
        } else if (action === 'jump-pricing') {
          document.getElementById('pricing').scrollIntoView({ behavior: 'smooth' });
        }
      });
    });

    renderTable();
  </script>
