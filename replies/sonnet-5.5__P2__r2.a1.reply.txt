<script>
  // Menu tabs
  const tabs = [...document.querySelectorAll('.tab')];
  function select(tab){
    tabs.forEach(t=>{
      const on = t===tab;
      t.setAttribute('aria-selected', on);
      t.tabIndex = on ? 0 : -1;
      const p = document.getElementById(t.getAttribute('aria-controls'));
      p.classList.toggle('active', on);
      p.hidden = !on;
    });
  }
  tabs.forEach((t,i)=>{
    t.addEventListener('click',()=>select(t));
    t.addEventListener('keydown',e=>{
      if(e.key!=='ArrowRight' && e.key!=='ArrowLeft') return;
      const n = tabs[(i + (e.key==='ArrowRight'?1:-1) + tabs.length) % tabs.length];
      n.focus(); select(n);
    });
  });

  // Reservation form (demo: no backend; wire to your booking service)
  const date = document.getElementById('date');
  date.min = new Date().toISOString().split('T')[0];
  document.getElementById('reserve').addEventListener('submit',e=>{
    e.preventDefault();
    const f = new FormData(e.target);
    const out = document.getElementById('confirm');
    out.textContent = `Grazie, ${f.get('name')}! We've noted a table for ${f.get('size')} on ${f.get('date')} at ${f.get('time')}. We'll call to confirm.`;
    out.style.display = 'block';
    e.target.reset();
  });
</script>
