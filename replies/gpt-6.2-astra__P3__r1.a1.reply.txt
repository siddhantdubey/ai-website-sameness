  // Embedded procedural landscape plates keep the portfolio usable offline.
  const scenes = [
    {title:'Where the fog settles', place:'Cascade Range, Oregon', year:2025, seed:59, kind:'forest', sky:[176,187,180], land:[29,55,49], alt:'Layers of evergreen ridges recede into pale morning fog.'},
    {title:'The painted hills', place:'John Day, Oregon', year:2024, seed:172, kind:'desert', sky:[226,202,175], land:[129,65,43], alt:'Ochre and rust-colored hills catch the last light.'},
    {title:'Blue distance', place:'Olympic Peninsula, Washington', year:2023, seed:12, kind:'alpine', sky:[182,196,207], land:[44,66,81], alt:'Blue mountain folds rise beneath a cool, clouded sky.'}
  ];
  const byId = id => document.getElementById(id);
  function hash(x,y,s){
    let n=Math.imul(x,374761393)+Math.imul(y,668265263)+Math.imul(s,69069);
    n=Math.imul(n^(n>>>13),1274126177);
    return ((n^(n>>>16))>>>0)/4294967295;
  }
  function noise(x,y,s){
    const i=Math.floor(x), j=Math.floor(y);
    let u=x-i,v=y-j;
    u=u*u*(3-2*u); v=v*v*(3-2*v);
    const a=hash(i,j,s),b=hash(i+1,j,s),c=hash(i,j+1,s),d=hash(i+1,j+1,s);
    return (a+(b-a)*u)*(1-v)+(c+(d-c)*u)*v;
  }
  function ridge(x,k,s){
    return .16+k*.088+(noise(x*3,k,s)-.5)*.35+
      (noise(x*10,k,s+1)-.5)*.105+(noise(x*40,k,s+2)-.5)*.025;
  }
  function draw(canvas,scene){
    const ctx=canvas.getContext('2d'),w=canvas.width,h=canvas.height;
    const img=ctx.createImageData(w,h),pixels=img.data;
    const {sky,land,seed,kind}=scene;
    for(let x=0;x<w;x++){
      const u=x/w, heights=[];
      for(let k=0;k<9;k++){
        let height=ridge(u,k,seed);
        if(kind==='forest'){
          const treeWidth=1+k*.55,tree=Math.floor(x/treeWidth);
          const tip=1-Math.abs(2*(x/treeWidth-tree)-1);
          height-=tip*hash(tree,k,seed)*(.001+k*.0018);
        }
        heights.push(height);
      }
      for(let y=0;y<h;y++){
        const v=y/h;
        let k=8;
        while(k>=0 && v<heights[k])k--;
        const cloud=noise(u*6,v*12,seed+3)*.7+noise(u*23,v*36,seed+4)*.3;
        const grain=(hash(x,y,seed)-.5)*6;
        const depth=k<0?0:Math.pow((k+1)/9,1.35);
        const relief=k<0?0:v-heights[k];
        const texture=k<0?0:noise(u*95+v*24,v*36,seed+k)-.5;
        let light=(cloud-.45)*(k<0?34:15)+grain;
        if(k<0)light+=13*v;
        else if(kind==='desert')light+=Math.sin(relief*80+cloud*4)*7+texture*15;
        else light+=texture*(6+20*depth)+Math.exp(-relief*10)*8;
        const mist=k<0?0:Math.max(0,cloud-.32)*Math.exp(-Math.abs(v-.5)*5)*.48;
        const mix=depth*(1-mist),i=(y*w+x)*4;
        for(let c=0;c<3;c++)pixels[i+c]=sky[c]*(1-mix)+land[c]*mix+light;
        pixels[i+3]=255;
      }
    }
    ctx.putImageData(img,0,0);canvas.setAttribute('aria-label',scene.alt);
  }
  let current=0;
  const cache=new Map();
  function show(index){
    current=(index+scenes.length)%scenes.length;
    const s=scenes[current],canvas=byId('feature');
    if(!cache.has(current)){
      const plate=document.createElement('canvas');
      plate.width=canvas.width;plate.height=canvas.height;
      draw(plate,s);cache.set(current,plate);
    }
    canvas.getContext('2d').drawImage(cache.get(current),0,0);
    canvas.setAttribute('aria-label',s.alt);
    byId('title').textContent=s.title;
    byId('location').textContent=s.place+' / '+s.year;
    byId('count').textContent='0'+(current+1)+' / 03';
  }
  byId('prev').onclick=()=>show(current-1);
  byId('next').onclick=()=>show(current+1);
  show(0);
  for(const s of scenes){
    const card=document.createElement('button');
    card.className='plate';card.setAttribute('aria-label','Enlarge '+s.title);
    card.innerHTML='<canvas width="660" height="446" aria-hidden="true"></canvas><span class="caption"><span style="font:inherit;font-family:Georgia,serif;font-size:25px;letter-spacing:-.05em"></span><span class="meta"></span></span>';
    card.querySelector('.caption span').textContent=s.title;
    card.querySelector('.meta').textContent=s.year+' ↗';
    draw(card.firstElementChild,s);
    card.onclick=()=>{
      byId('viewTitle').textContent=s.title;
      draw(byId('large'),s);byId('viewer').showModal();
    };
    byId('gallery').append(card);
  }
</script>
