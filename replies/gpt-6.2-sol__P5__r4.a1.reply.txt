  <script>
    const board = document.getElementById("board");
    const list = document.getElementById("list");
    const rows = list.querySelector("tbody");
    const controls = document.querySelector(".views");
    const viewButtons = controls.querySelectorAll("button");

    for (const lane of board.querySelectorAll(".lane")) {
      const status = lane.querySelector("h3").childNodes;
      const label = Array.from(status)
        .filter(node => node.nodeType === Node.TEXT_NODE)
        .map(node => node.textContent.trim())
        .join("").trim();

      for (const task of lane.querySelectorAll(".task")) {
        const row = rows.insertRow();
        const fields = [
          task.querySelector(".task-name").textContent,
          label,
          task.dataset.owner,
          task.querySelector(".task-meta span").textContent
        ];

        for (const field of fields) {
          row.insertCell().textContent = field;
        }
      }
    }

    for (const button of viewButtons) {
      button.addEventListener("click", () => {
        const showBoard = button.dataset.view === "board";
        board.hidden = !showBoard;
        list.hidden = showBoard;

        for (const option of viewButtons) {
          option.setAttribute("aria-pressed", String(option === button));
        }
      });
    }

    controls.hidden = false;
  </script>
