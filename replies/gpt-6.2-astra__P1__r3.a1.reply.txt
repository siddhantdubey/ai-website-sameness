  <script>
    const byId = id => document.getElementById(id);
    const dialog = byId('workspace');
    const statuses = ['To do', 'In progress', 'Done'];
    let project = null;
    try {
      const saved = JSON.parse(localStorage.getItem('fieldwork-project-v1'));
      if (saved && typeof saved.name === 'string' && Array.isArray(saved.tasks) &&
          saved.tasks.every(task => task && typeof task.name === 'string' && statuses.includes(task.status))) project = saved;
    } catch {}
    function save() {
      try { localStorage.setItem('fieldwork-project-v1', JSON.stringify(project)); }
      catch { byId('storage-note').textContent = 'Browser storage is unavailable. Changes last for this visit.'; }
    }
    function updateSummary() {
      const done = project.tasks.filter(task => task.status === 'Done').length;
      byId('summary').textContent = project.tasks.length ? `${done} of ${project.tasks.length} tasks complete` : 'Add the first task to your project.';
    }
    function render() {
      byId('create').hidden = Boolean(project);
      byId('project').hidden = !project;
      if (!project) return;
      byId('workspace-title').textContent = project.name;
      updateSummary();
      byId('entries').replaceChildren();
      for (const task of project.tasks) {
        const row = document.createElement('div');
        row.className = 'entry';
        const title = document.createElement('span');
        title.textContent = task.name;
        const select = document.createElement('select');
        select.setAttribute('aria-label', `Status for ${task.name}`);
        for (const status of statuses) select.add(new Option(status, status));
        select.value = task.status;
        select.onchange = () => { task.status = select.value; save(); updateSummary(); };
        row.append(title, select);
        byId('entries').append(row);
      }
    }
    document.querySelectorAll('.launch').forEach(button => button.onclick = () => {
      render();
      dialog.showModal();
      byId(project ? 'task-name' : 'project-name').focus();
    });
    document.querySelector('.close').onclick = () => dialog.close();
    byId('create').onsubmit = event => {
      event.preventDefault();
      const name = byId('project-name').value.trim();
      if (!name) return byId('project-name').focus();
      project = { name, tasks: [] };
      save();
      render();
      byId('task-name').focus();
    };
    byId('add').onsubmit = event => {
      event.preventDefault();
      const input = byId('task-name');
      const name = input.value.trim();
      if (!name) return input.focus();
      project.tasks.push({ name, status: statuses[0] });
      save();
      render();
      input.value = '';
      input.focus();
    };
  </script>
