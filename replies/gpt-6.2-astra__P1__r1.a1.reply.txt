  <script>
    const stages = ["To do", "In progress", "Done"];
    const samples = [
      ["Write the launch brief", "Strategy", "Mia Chen", 0],
      ["Build the landing page", "Design", "Theo Park", 1],
      ["Map the launch timeline", "Planning", "Ada Lee", 2]
    ];
    const colors = ["var(--lime)", "var(--peach)", "var(--mint)"];
    const tasks = samples.map(([name, tag, owner, stage]) => ({name, tag, owner, stage}));
    const storageKey = "fieldwork-board-v1";
    try {
      const saved = JSON.parse(localStorage.getItem(storageKey));
      if (Array.isArray(saved) && saved.every(t => t && typeof t.name === "string" &&
          typeof t.owner === "string" && typeof t.tag === "string" && Number.isInteger(t.stage) &&
          t.stage >= 0 && t.stage < stages.length)) tasks.splice(0, tasks.length, ...saved);
    } catch {}
    const editor = document.querySelector("#editor");
    const form = document.querySelector("#task-form");
    const nameField = document.querySelector("#task-name");
    const ownerField = document.querySelector("#assignee");
    const stageField = document.querySelector("#stage");
    let editing = null;

    function openEditor(index = null) {
      editing = index;
      const task = tasks[index];
      form.reset();
      nameField.setCustomValidity("");
      nameField.value = task?.name || "";
      ownerField.value = task?.owner || "";
      stageField.selectedIndex = task?.stage ?? 0;
      document.querySelector("#dialog-title").textContent = task ? "Edit task" : "Add a task";
      form.querySelector(".button").textContent = task ? "Save changes" : "Add task";
      editor.showModal();
      nameField.focus();
    }
    function render() {
      const columns = document.querySelector("#columns");
      columns.replaceChildren();
      stages.forEach((stage, s) => {
        const column = document.createElement("section");
        column.className = "column";
        column.innerHTML = `<h3 class="legend"><span class="dot"></span>${stage} <span class="count">${tasks.filter(t => t.stage === s).length}</span></h3>`;
        tasks.forEach((task, index) => {
          if (task.stage !== s) return;
          const card = document.createElement("article");
          card.className = "card";
          card.innerHTML = '<span class="tag"></span><h4><button type="button"></button></h4><div class="task-meta"><span></span><span class="avatar"></span></div>';
          const tag = card.querySelector(".tag");
          tag.textContent = task.tag;
          tag.style.background = colors[s];
          const title = card.querySelector("button");
          title.textContent = task.name;
          title.style.cssText = "background:none;border:0;padding:0;font:inherit;text-align:left";
          title.setAttribute("aria-label", "Edit task " + task.name);
          title.dataset.index = index;
          title.onclick = () => openEditor(index);
          card.querySelector(".task-meta span").textContent = task.owner || "Unassigned";
          card.querySelector(".avatar").textContent = task.owner.trim().split(/\s+/).slice(0, 2).map(w => w[0]).join("").toUpperCase() || "–";
          column.append(card);
        });
        columns.append(column);
      });
    }
    document.querySelectorAll("[data-add]").forEach(button => button.onclick = () => openEditor());
    document.querySelector("#cancel").onclick = () => editor.close();
    nameField.oninput = () => nameField.setCustomValidity("");
    form.onsubmit = event => {
      event.preventDefault();
      if (!nameField.value.trim()) {
        nameField.setCustomValidity("Enter a task name.");
        nameField.reportValidity();
        return;
      }
      const index = editing ?? tasks.length;
      tasks[index] = {
        name: nameField.value.trim(), owner: ownerField.value.trim(),
        stage: stageField.selectedIndex, tag: tasks[index]?.tag || "Task"
      };
      try { localStorage.setItem(storageKey, JSON.stringify(tasks)); } catch {}
      editor.close();
      render();
      document.querySelector(`[data-index="${index}"]`).focus();
      document.querySelector("#status").textContent = `${tasks[index].name} saved in ${stages[tasks[index].stage]}.`;
    };
    render();
  </script>
