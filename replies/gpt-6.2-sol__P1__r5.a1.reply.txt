  <script>
    const projects = {
      launch: {
        name: "Website launch",
        tasks: [
          { title: "Write launch announcement", tag: "Content", owner: "Maya Chen", status: "todo" },
          { title: "Check mobile layouts", tag: "Engineering", owner: "Leo Ross", status: "todo" },
          { title: "Design pricing page", tag: "Design", owner: "Ivy Cole", status: "doing" },
          { title: "Prepare product screenshots", tag: "Design", owner: "Maya Chen", status: "doing" },
          { title: "Approve homepage copy", tag: "Content", owner: "Leo Ross", status: "done" }
        ]
      },
      brand: {
        name: "Brand refresh",
        tasks: [
          { title: "Update presentation template", tag: "Design", owner: "Ivy Cole", status: "todo" },
          { title: "Draft voice guidelines", tag: "Content", owner: "Maya Chen", status: "doing" },
          { title: "Audit existing brand assets", tag: "Research", owner: "Leo Ross", status: "done" }
        ]
      }
    };

    const statuses = [
      { key: "todo", label: "To do" },
      { key: "doing", label: "In progress" },
      { key: "done", label: "Done" }
    ];

    let selectedProject = "launch";
    let selectedView = "board";

    const taskArea = document.querySelector("#tasks");
    const projectTitle = document.querySelector("#project-title");
    const progress = document.querySelector("#progress");
    const projectButtons = document.querySelectorAll("[data-project]");
    const viewButtons = document.querySelectorAll("[data-view]");

    function render() {
      const project = projects[selectedProject];
      projectTitle.textContent = project.name;
      taskArea.className = selectedView;

      const completed = project.tasks.filter(task => task.status === "done").length;
      const progressText = `${completed} of ${project.tasks.length} done`;
      if (progress.textContent !== progressText) progress.textContent = progressText;

      taskArea.innerHTML = statuses.map(({ key, label }) => {
        const tasks = project.tasks
          .map((task, index) => ({ ...task, index }))
          .filter(task => task.status === key);

        const cards = tasks.map(task => {
          const initials = task.owner.split(" ").map(name => name[0]).join("");
          return `
            <div class="task">
              <label>
                <input type="checkbox" data-task="${task.index}" ${key === "done" ? "checked" : ""}>
                <span>${task.title}</span>
              </label>
              <div class="task-meta">
                <span class="tag">${task.tag}</span>
                <span class="avatar" role="img" aria-label="Assigned to ${task.owner}" title="${task.owner}">${initials}</span>
              </div>
            </div>`;
        }).join("");

        return `
          <section class="column" aria-labelledby="status-${key}">
            <h3 id="status-${key}"><span class="status-dot ${key}" aria-hidden="true"></span>${label} <span class="count">${tasks.length}</span></h3>
            ${cards}
          </section>`;
      }).join("");
    }

    projectButtons.forEach(button => button.addEventListener("click", () => {
      selectedProject = button.dataset.project;
      projectButtons.forEach(item => item.setAttribute("aria-pressed", String(item === button)));
      render();
    }));

    viewButtons.forEach(button => button.addEventListener("click", () => {
      selectedView = button.dataset.view;
      viewButtons.forEach(item => item.setAttribute("aria-pressed", String(item === button)));
      render();
    }));

    taskArea.addEventListener("change", event => {
      const checkbox = event.target.closest("input[data-task]");
      if (!checkbox) return;

      const task = projects[selectedProject].tasks[Number(checkbox.dataset.task)];
      if (checkbox.checked) {
        task.previousStatus = task.status;
        task.status = "done";
      } else {
        task.status = task.previousStatus || "todo";
      }

      render();
      taskArea.querySelector(`[data-task="${checkbox.dataset.task}"]`).focus();
    });

    render();
  </script>
</body>
