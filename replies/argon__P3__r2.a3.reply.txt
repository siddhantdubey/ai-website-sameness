        const galleryGrid = document.getElementById('galleryGrid');
        let currentFilter = 'all';
        let currentVisibleIndices = PHOTO_ARCHIVE.map((_, idx) => idx);
        let activeLightboxPos = 0;

        // Render Gallery Cards
        function renderGallery(filter = 'all') {
            currentFilter = filter;
            galleryGrid.innerHTML = '';
            currentVisibleIndices = [];

            PHOTO_ARCHIVE.forEach((item, index) => {
                if (filter !== 'all' && item.category !== filter) return;
                currentVisibleIndices.push(index);

                const card = document.createElement('figure');
                card.className = 'photo-card';
                card.setAttribute('data-span', item.span);
                card.setAttribute('data-offset', item.offset);
                card.setAttribute('data-index', index);

                card.innerHTML = `
                    <div class="photo-aspect ${item.ratio}">
                        <img src="${item.src}" alt="${item.title}" loading="lazy">
                        <span class="photo-overlay-badge mono">${item.category} · ${item.stock}</span>
                    </div>
                    <figcaption class="photo-meta">
                        <div class="photo-title-wrap">
                            <span class="photo-num mono">#${item.id}</span>
                            <h3 class="photo-title">${item.title}</h3>
                        </div>
                        <div class="photo-exif mono">${item.location}</div>
                    </figcaption>
                `;

                card.addEventListener('click', () => openLightboxByArchiveIndex(index));
                galleryGrid.appendChild(card);
            });
        }

        // Lightbox Logic
        const lightbox = document.getElementById('lightbox');
        const lbImage = document.getElementById('lbImage');
        const lbNumber = document.getElementById('lbNumber');
        const lbCategory = document.getElementById('lbCategory');
        const lbTitle = document.getElementById('lbTitle');
        const lbLocation = document.getElementById('lbLocation');
        const lbCamera = document.getElementById('lbCamera');
        const lbLens = document.getElementById('lbLens');
        const lbStock = document.getElementById('lbStock');

        function openLightboxByArchiveIndex(archiveIdx) {
            const posInVisible = currentVisibleIndices.indexOf(archiveIdx);
            if (posInVisible === -1) {
                currentVisibleIndices = PHOTO_ARCHIVE.map((_, i) => i);
                activeLightboxPos = archiveIdx;
            } else {
                activeLightboxPos = posInVisible;
            }
            updateLightboxSlide();
            lightbox.classList.add('open');
            document.body.style.overflow = 'hidden';
        }

        function updateLightboxSlide() {
            const archiveIdx = currentVisibleIndices[activeLightboxPos];
            const item = PHOTO_ARCHIVE[archiveIdx];
            if (!item) return;

            lbImage.style.opacity = '0.35';
            setTimeout(() => {
                lbImage.src = item.src;
                lbImage.alt = item.title;
                lbNumber.textContent = `PLATE ${item.id} / ${String(PHOTO_ARCHIVE.length).padStart(2, '0')}`;
                lbCategory.textContent = item.category.toUpperCase();
                lbTitle.textContent = item.title;
                lbLocation.textContent = item.location;
                lbCamera.textContent = item.camera;
                lbLens.textContent = item.lens;
                lbStock.textContent = item.stock;
                lbImage.style.opacity = '1';
            }, 120);
        }

        function closeLightbox() {
            lightbox.classList.remove('open');
            document.body.style.overflow = '';
        }

        function nextSlide() {
            activeLightboxPos = (activeLightboxPos + 1) % currentVisibleIndices.length;
            updateLightboxSlide();
        }

        function prevSlide() {
            activeLightboxPos = (activeLightboxPos - 1 + currentVisibleIndices.length) % currentVisibleIndices.length;
            updateLightboxSlide();
        }

        document.getElementById('lbClose').addEventListener('click', closeLightbox);
        document.getElementById('lbNext').addEventListener('click', nextSlide);
        document.getElementById('lbPrev').addEventListener('click', prevSlide);

        // Acquire Print Button pre-fills commission form
        document.getElementById('lbAcquireBtn').addEventListener('click', () => {
            const archiveIdx = currentVisibleIndices[activeLightboxPos];
            const item = PHOTO_ARCHIVE[archiveIdx];
            closeLightbox();
            document.getElementById('typeSelect').value = 'print';
            document.getElementById('plateInput').value = `Plate #${item.id} — ${item.title}`;
            document.getElementById('commission').scrollIntoView({ behavior: 'smooth' });
            document.getElementById('briefInput').focus();
        });

        // Keyboard Navigation for Lightbox
        window.addEventListener('keydown', (e) => {
            if (!lightbox.classList.contains('open')) return;
            if (e.key === 'Escape') closeLightbox();
            if (e.key === 'ArrowRight') nextSlide();
            if (e.key === 'ArrowLeft') prevSlide();
        });

        // Darkroom / Paper Mode Theme Switcher
        const themeToggle = document.getElementById('themeToggle');
        const themeLabel = document.getElementById('themeLabel');

        themeToggle.addEventListener('click', () => {
            const root = document.documentElement;
            const isLight = root.getAttribute('data-theme') === 'light';
            if (isLight) {
                root.removeAttribute('data-theme');
                themeLabel.textContent = 'PAPER MODE';
            } else {
                root.setAttribute('data-theme', 'light');
                themeLabel.textContent = 'DARKROOM MODE';
            }
        });

            e.preventDefault();
            const feedback = document.getElementById('formFeedback');
            feedback.style.display = 'block';
            e.target.reset();
        });
