        bookingSummary.innerHTML = `
          <strong>${escapeHtml(name)}</strong> &bull; ${escapeHtml(guests)}<br />
          ${formattedDate} at <strong>${escapeHtml(selectedTime)}</strong><br />
          Seating: ${escapeHtml(area)}
          ${notes ? `<br /><em>Note: ${escapeHtml(notes)}</em>` : ''}
        `;

        resForm.style.display = 'none';
        confirmationBox.classList.add('visible');
      });

      resetBookingBtn.addEventListener('click', () => {
        confirmationBox.classList.remove('visible');
        resForm.style.display = 'block';
      });

      // Newsletter signup
      const newsletterForm = document.getElementById('newsletterForm');
      const newsletterMsg = document.getElementById('newsletterMsg');
      newsletterForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const emailInput = document.getElementById('newsletterEmail');
        if (emailInput.value.trim()) {
          newsletterMsg.textContent = 'Grazie! You’re on the list for seasonal kitchen notes.';
          emailInput.value = '';
        }
      });

      function escapeHtml(str) {
        return str.replace(/[&<>"']/g, (char) => ({
          '&': '&amp;',
          '<': '&lt;',
          '>': '&gt;',
          '"': '&quot;',
          "'": '&#39;'
        }[char]));
      }
    })();
  </script>
