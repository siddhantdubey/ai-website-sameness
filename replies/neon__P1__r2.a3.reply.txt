      function renderCmdList(query = '') {
        const q = query.toLowerCase().trim();
        const list = actions.filter(a => a.label.toLowerCase().includes(q));
        globalCmdList.innerHTML = list.length
          ? list.map((a, idx) => `
              <button type="button" class="mini-cmd-item ${idx === 0 ? 'highlight' : ''}" data-cmd-idx="${actions.indexOf(a)}" style="width:100%; text-align:left;">
                <span>${a.label}</span>
                <span class="kbd">${a.hint}</span>
              </button>
            `).join('')
          : '<div class="mini-cmd-item">No matching actions</div>';
      }

      document.getElementById('openCmdBtn').addEventListener('click', () => {
        renderCmdList();
        openModal(cmdModal);
        globalCmdInput.value = '';
        setTimeout(() => globalCmdInput.focus(), 40);
      });

      globalCmdInput.addEventListener('input', () => renderCmdList(globalCmdInput.value));
      globalCmdList.addEventListener('click', (e) => {
        const item = e.target.closest('[data-cmd-idx]');
        if (!item) return;
        const action = actions[Number(item.dataset.cmdIdx)];
        closeModals();
        if (action) action.run();
      });

      document.addEventListener('keydown', (e) => {
        if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
          e.preventDefault();
          if (cmdModal.classList.contains('open')) closeModals();
          else {
            renderCmdList();
            openModal(cmdModal);
            globalCmdInput.value = '';
            setTimeout(() => globalCmdInput.focus(), 40);
          }
        } else if (e.key === 'Escape') {
          closeModals();
        }
      });

        entries.forEach(entry => {
          if (entry.isIntersecting) {

            observer.unobserve(entry.target);
          }
        });
      }, { threshold: 0.12 });

      document.querySelectorAll('.reveal').forEach(el => observer.observe(el));
    })();
