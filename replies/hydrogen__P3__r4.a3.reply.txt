        viewGridBtn.classList.remove("active");
        viewSheetBtn.setAttribute("aria-pressed", "true");
        viewGridBtn.setAttribute("aria-pressed", "false");
      });

      /* --------------------------------------------------------------------
         5. FULL-SCREEN LIGHTBOX WITH EXIF DRAWER & KEYBOARD NAVIGATION
         -------------------------------------------------------------------- */
      const lightbox = document.getElementById("lightbox");
      const lbStage = document.getElementById("lbStage");
      const lbImage = document.getElementById("lbImage");
      const lbCounter = document.getElementById("lbCounter");
      const lbTitle = document.getElementById("lbTitle");
      const lbSub = document.getElementById("lbSub");
      const lbExifTitle = document.getElementById("lbExifTitle");
      const lbExifDesc = document.getElementById("lbExifDesc");
      const lbSpecList = document.getElementById("lbSpecList");
      const lbStrip = document.getElementById("lbStrip");
      const lbExifToggle = document.getElementById("lbExifToggle");

      let currentLbIdx = 0;

      // Build thumbnail strip
      PHOTOS.forEach((p, idx) => {
        const thumbBtn = document.createElement("button");
        thumbBtn.type = "button";
        thumbBtn.className = "lb-thumb";
        thumbBtn.setAttribute("aria-label", `View ${p.title}`);
        thumbBtn.innerHTML = `<img src="${imageCache[p.id]}" alt="" />`;
        thumbBtn.addEventListener("click", () => updateLightbox(idx));
        lbStrip.appendChild(thumbBtn);
      });

      function updateLightbox(idx) {
        currentLbIdx = (idx + PHOTOS.length) % PHOTOS.length;
        const p = PHOTOS[currentLbIdx];
        lbImage.src = imageCache[p.id];
        lbImage.alt = `${p.title} — ${p.location}`;
        lbCounter.textContent = `PLATE ${String(currentLbIdx + 1).padStart(2, "0")} / ${PHOTOS.length}`;
        lbTitle.textContent = p.title;
        lbSub.textContent = `${p.location} · ${p.year} · ${p.camera}`;
        lbExifTitle.textContent = p.title;
        lbExifDesc.textContent = p.note;
        lbSpecList.innerHTML = `
          <li><span>CAMERA</span><span>${p.camera}</span></li>
          <li><span>OPTICS</span><span>${p.lens}</span></li>
          <li><span>EXPOSURE</span><span>${p.settings}</span></li>
          <li><span>EMULSION</span><span>${p.stock}</span></li>
          <li><span>LOCATION</span><span>${p.location}</span></li>
          <li><span>ARCHIVAL PRINT</span><span>${p.edition}</span></li>
        `;

        lbStrip.querySelectorAll(".lb-thumb").forEach((t, i) => {
          t.classList.toggle("active", i === currentLbIdx);
        });
      }

      function openLightbox(idx) {
        updateLightbox(idx);
        lightbox.classList.add("open");
        lightbox.setAttribute("aria-hidden", "false");
        document.body.style.overflow = "hidden";
        document.getElementById("lbCloseBtn").focus();
      }

      function closeLightbox() {
        lightbox.classList.remove("open");
        lightbox.setAttribute("aria-hidden", "true");
        document.body.style.overflow = "";
      }

      document.getElementById("lbCloseBtn").addEventListener("click", closeLightbox);
      document.getElementById("lbPrev").addEventListener("click", () => updateLightbox(currentLbIdx - 1));
      document.getElementById("lbNext").addEventListener("click", () => updateLightbox(currentLbIdx + 1));

      lbExifToggle.addEventListener("click", () => {
        const on = lbStage.classList.toggle("with-exif");
        lbExifToggle.classList.toggle("active", on);
        lbExifToggle.setAttribute("aria-pressed", String(on));
      });

      lightbox.addEventListener("click", (e) => {
        if (e.target === lightbox || e.target === lbStage) {
          closeLightbox();
        }
      });

      document.addEventListener("keydown", (e) => {
        if (!lightbox.classList.contains("open")) return;
        if (e.key === "Escape") closeLightbox();
        if (e.key === "ArrowLeft") updateLightbox(currentLbIdx - 1);
        if (e.key === "ArrowRight") updateLightbox(currentLbIdx + 1);
        if (e.key.toLowerCase() === "i") lbExifToggle.click();
      });

      /* --------------------------------------------------------------------
         6. THEME TOGGLE, STUDIO CLOCK, MOBILE DRAWER & COMMISSION FORM
         -------------------------------------------------------------------- */
      const themeToggle = document.getElementById("themeToggle");
      const themeLabel = document.getElementById("themeLabel");
      themeToggle.addEventListener("click", () => {
        const html = document.documentElement;
        const next = html.getAttribute("data-theme") === "dark" ? "light" : "dark";
        html.setAttribute("data-theme", next);
        themeLabel.textContent = next === "dark" ? "Paper" : "Darkroom";
      });

      function updateClock() {
        const el = document.getElementById("studioClock");
        try {
          const now = new Date();
          const parisTime = new Intl.DateTimeFormat("en-GB", {
            timeZone: "Europe/Paris",
            hour: "2-digit",
            minute: "2-digit",
            hour12: false
          }).format(now);
          el.textContent = `${parisTime} CET`;
        } catch {
          el.textContent = "CET";
        }
      }
      updateClock();
      setInterval(updateClock, 30000);

      // Mobile drawer
      const mobileMenuBtn = document.getElementById("mobileMenuBtn");
      const mobileDrawer = document.getElementById("mobileDrawer");
      mobileMenuBtn.addEventListener("click", () => {
        const open = mobileDrawer.classList.toggle("open");
        mobileMenuBtn.setAttribute("aria-expanded", String(open));
        mobileDrawer.setAttribute("aria-hidden", String(!open));
      });
      document.querySelectorAll(".mobile-nav-item").forEach((link) => {
        link.addEventListener("click", () => {
          mobileDrawer.classList.remove("open");
          mobileMenuBtn.setAttribute("aria-expanded", "false");
          mobileDrawer.setAttribute("aria-hidden", "true");
        });
      });

      // Copy email button
      const copyEmailBtn = document.getElementById("copyEmailBtn");
      copyEmailBtn.addEventListener("click", async () => {
        try {
          await navigator.clipboard.writeText("studio@claramoreau.photo");
          copyEmailBtn.textContent = "Copied!";
          setTimeout(() => { copyEmailBtn.textContent = "Copy Email"; }, 2200);
        } catch {
          copyEmailBtn.textContent = "studio@claramoreau.photo";
        }
      });

      // Commission form submission
      const inquiryForm = document.getElementById("inquiryForm");
      const formStatus = document.getElementById("formStatus");
      inquiryForm.addEventListener("submit", (e) => {
        e.preventDefault();
        const name = document.getElementById("clientName").value.trim();
        const email = document.getElementById("clientEmail").value.trim();
        const type = document.getElementById("shootType").value;
        const brief = document.getElementById("shootBrief").value.trim();

        if (!name || !email || !email.includes("@") || !brief) {
          formStatus.textContent = "Please complete your name, a valid email address, and brief project notes.";
          formStatus.classList.add("visible");
          return;
        }

        formStatus.innerHTML = `Thank you, <strong>${name.replace(/[<>&]/g, "")}</strong>. Your <em>${type}</em> inquiry has been logged for Clara Moreau's studio desk—expect a response within 24 hours.`;
        formStatus.classList.add("visible");
        inquiryForm.reset();
      });
    })();
  </script>
