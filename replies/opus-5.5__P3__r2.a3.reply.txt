function rng(seed) {
  return function () {
    seed |= 0; seed = seed + 0x6D2B79F5 | 0;
    let t = Math.imul(seed ^ seed >>> 15, 1 | seed);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}

function ridge(r, w, h, base, amp, rough) {
  let y = base, d = `M0 ${h} L0 ${y.toFixed(1)}`;
  const steps = 40;
  for (let i = 1; i <= steps; i++) {
    y += (r() - .5) * amp * rough;
    y = Math.max(base - amp, Math.min(base + amp * .5, y));
    d += ` L${(w * i / steps).toFixed(1)} ${y.toFixed(1)}`;
  }
  return d + ` L${w} ${h} Z`;
}

function makeScene(type, palette, seed, w, h) {
  const r = rng(seed), p = PALETTES[palette];
  let body = "";
  const defs = `<linearGradient id="sky" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="${p[0]}"/><stop offset="1" stop-color="${p[1]}"/></linearGradient>
    <radialGradient id="glow"><stop offset="0" stop-color="${p[2]}" stop-opacity=".9"/><stop offset="1" stop-color="${p[2]}" stop-opacity="0"/></radialGradient>`;

  if (type === "mountains") {
    const sx = w * (.25 + r() * .5), sy = h * (.35 + r() * .2), sr = w * (.05 + r() * .04);
    body += `<rect width="${w}" height="${h}" fill="url(#sky)"/>`;
    body += `<circle cx="${sx}" cy="${sy}" r="${sr * 5}" fill="url(#glow)"/><circle cx="${sx}" cy="${sy}" r="${sr}" fill="${p[2]}"/>`;
    [[.5, .14, 1.4, 3, .55], [.62, .12, 1.2, 4, .8], [.75, .1, 1, 5, 1]].forEach(([b, a, ro, ci, op]) => {
      body += `<path d="${ridge(r, w, h, h * b, h * a, ro)}" fill="${p[ci]}" opacity="${op}"/>`;
    });
  } else if (type === "sea") {
    const hz = h * (.5 + r() * .12), sx = w * (.3 + r() * .4);
    body += `<rect width="${w}" height="${hz}" fill="url(#sky)"/>`;
    body += `<circle cx="${sx}" cy="${hz}" r="${w * .45}" fill="url(#glow)"/><circle cx="${sx}" cy="${hz - w * .03}" r="${w * .06}" fill="${p[2]}"/>`;
    body += `<linearGradient id="sea" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="${p[3]}"/><stop offset="1" stop-color="${p[5]}"/></linearGradient>`;
    body += `<rect y="${hz}" width="${w}" height="${h - hz}" fill="url(#sea)"/>`;
    for (let i = 0; i < 26; i++) {
      const y = hz + Math.pow(r(), 1.6) * (h - hz), len = w * (.05 + r() * .2) * (1 + (y - hz) / h);
      body += `<rect x="${sx - len / 2 + (r() - .5) * w * .15}" y="${y}" width="${len}" height="${1 + (y - hz) / 90}" fill="${p[2]}" opacity="${.15 + r() * .45}"/>`;
    }
  } else if (type === "city") {
    body += `<rect width="${w}" height="${h}" fill="url(#sky)"/>`;
    body += `<circle cx="${w * (.2 + r() * .6)}" cy="${h * .28}" r="${w * .04}" fill="${p[2]}" opacity=".9"/>`;
    [[.45, p[4], .7], [.6, p[5], 1]].forEach(([top, col, op]) => {
      let x = -10;
      while (x < w) {
        const bw = w * (.06 + r() * .1), bh = h * (1 - top) * (.5 + r() * .8);
        body += `<rect x="${x}" y="${h - bh}" width="${bw}" height="${bh}" fill="${col}" opacity="${op}"/>`;
        if (op === 1) for (let wy = h - bh + 12; wy < h - 10; wy += 16) for (let wx = x + 6; wx < x + bw - 8; wx += 12)
          if (r() > .72) body += `<rect x="${wx}" y="${wy}" width="5" height="7" fill="${p[2]}" opacity="${.5 + r() * .5}"/>`;
        x += bw + 2;
      }
    });
  } else if (type === "portrait" || type === "crowd") {
    body += `<radialGradient id="bg" cx="${.35 + r() * .3}" cy=".35" r=".85"><stop offset="0" stop-color="${p[1]}"/><stop offset="1" stop-color="${p[0]}"/></radialGradient>`;
    body += `<rect width="${w}" height="${h}" fill="url(#bg)"/>`;
    const figures = type === "crowd" ? 6 + Math.floor(r() * 4) : 1;
    for (let i = 0; i < figures; i++) {
      const s = type === "crowd" ? .35 + r() * .35 : 1;
      const cx = type === "crowd" ? w * (i + .5) / figures + (r() - .5) * 30 : w * (.42 + r() * .16);
      const hr = w * .13 * s, cy = h - (type === "crowd" ? h * .28 * s + r() * 20 : h * .5);
      const side = r() > .5 ? 1 : -1;
      body += `<path d="M${cx - hr * 2.6} ${h} C${cx - hr * 2.4} ${cy + hr * 2.2} ${cx - hr * 1.2} ${cy + hr * 1.6} ${cx} ${cy + hr * 1.6} C${cx + hr * 1.2} ${cy + hr * 1.6} ${cx + hr * 2.4} ${cy + hr * 2.2} ${cx + hr * 2.6} ${h} Z" fill="${p[3]}"/>`;
      body += `<rect x="${cx - hr * .38}" y="${cy + hr * .6}" width="${hr * .76}" height="${hr * 1.2}" fill="${p[3]}"/>`;
      body += `<ellipse cx="${cx}" cy="${cy}" rx="${hr * .82}" ry="${hr}" fill="${p[3]}"/>`;
      body += `<ellipse cx="${cx}" cy="${cy}" rx="${hr * .82}" ry="${hr}" fill="none" stroke="${p[4]}" stroke-width="${hr * .09}" stroke-dasharray="${hr * 2.2} ${hr * 4}" stroke-dashoffset="${side > 0 ? hr * .4 : hr * 3.2}" opacity=".85"/>`;
      if (type === "crowd" && r() > .6) body += `<circle cx="${cx + side * hr * 1.4}" cy="${cy - hr * 1.6}" r="${hr * .25}" fill="${p[2]}" opacity=".7"/>`;
    }
    body += `<rect width="${w}" height="${h}" fill="${p[5]}" opacity=".18"/>`;
  } else if (type === "still") {
    body += `<rect width="${w}" height="${h}" fill="${p[1]}"/>`;
    body += `<rect y="${h * .62}" width="${w}" height="${h * .38}" fill="${p[0]}" opacity=".55"/>`;
    const n = 2 + Math.floor(r() * 3);
    for (let i = 0; i < n; i++) {
      const x = w * (.18 + i * (.64 / n)) + r() * 30, bw = w * (.1 + r() * .1), bh = h * (.15 + r() * .3);
      const col = p[3 + Math.floor(r() * 3)];
      if (r() > .5) body += `<rect x="${x}" y="${h * .66 - bh}" width="${bw}" height="${bh}" rx="${bw * .15}" fill="${col}"/><ellipse cx="${x + bw / 2}" cy="${h * .66 - bh}" rx="${bw / 2}" ry="${bw * .12}" fill="${p[2]}" opacity=".5"/>`;
      else body += `<circle cx="${x + bw / 2}" cy="${h * .66 - bw / 2}" r="${bw / 2}" fill="${col}"/>`;
      body += `<ellipse cx="${x + bw * .7}" cy="${h * .665}" rx="${bw * .8}" ry="${bw * .08}" fill="#000" opacity=".18"/>`;
    }
    body += `<rect width="${w}" height="${h}" fill="url(#glow)" opacity=".25"/>`;
  }

  body += `<rect width="${w}" height="${h}" filter="url(#g)" opacity=".35"/>`;
  return "data:image/svg+xml;charset=utf-8," + encodeURIComponent(
    `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${w} ${h}" width="${w}" height="${h}" preserveAspectRatio="xMidYMid slice"><defs>${defs}${grain}</defs>${body}</svg>`);
}

I've built the portfolio site as a single file: `index.html`. It doesn't load anything from outside: no fonts, images or libraries. The JavaScript passes a syntax check, but I haven't opened the page in a browser.

**What's in it**, for a made-up Lisbon photographer called "Ines Calder":
- **Navigation:** a bar that stays at the top and collapses to a menu button on phones.
- **Hero:** a large serif headline, two buttons ("View portfolio" and "Book a session") and a featured image.
- **Stats bar:** years working, number of commissions, countries, turnaround time.
- **Portfolio:** 14 images in an uneven grid, filtered by Portrait, Landscape, Editorial or Events.
- **Image viewer:** clicking an image opens it full-screen. You can move through the images with the arrow buttons, the arrow keys or a swipe on phones, and Esc closes it. It only cycles through the images in the current filter.
- **About:** a dark section with a bio and a list of clients.
- **Services:** three packages with starting prices.
- **Testimonials**, then a **contact form** that checks fields before sending (required fields, email format, date not in the past).
- **Accessibility:** a skip link, visible focus outlines, keyboard support in the image viewer, and gentler animation for people who turn motion down.

**Things to know:**
- **The images are generated placeholders.** Each "photo" is a drawn scene made in the browser: mountains, sea, city at night, portrait silhouettes, crowds, still life. To use real photos, add a `src` to each item in the `PHOTOS` list in the script and it will be used instead. Also replace the `hero-img` and `about-img` sources.
- **The contact form doesn't send anything yet.** A single file has no server, so it only shows a thank-you message. To make it work, connect it to a form service such as Formspree or Netlify Forms (the comment in the submit code marks the spot).
- **All the content is invented.** The name, bio, client list, prices, testimonials and contact details are examples to replace with real ones.
