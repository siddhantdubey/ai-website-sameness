      const footerForm = document.getElementById('footerSignupForm');
      const footerEmail = document.getElementById('footerEmail');
      const footerFeedback = document.getElementById('footerFeedback');

      footerForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const email = footerEmail.value.trim();
        if (!email) return;
        footerFeedback.classList.add('success');
        footerFeedback.textContent = `Invite link prepared for ${email}. Check your inbox to name your first ledger.`;
        footerEmail.value = '';
      });

      /* ==================== MOBILE NAV ==================== */
      const mobileMenuBtn = document.getElementById('mobileMenuBtn');
      const mainNav = document.getElementById('mainNav');
      mobileMenuBtn.addEventListener('click', () => {
        const open = mainNav.classList.toggle('mobile-open');
        mobileMenuBtn.setAttribute('aria-expanded', String(open));
      });
      mainNav.querySelectorAll('a').forEach(a => {
        a.addEventListener('click', () => {
          mainNav.classList.remove('mobile-open');
          mobileMenuBtn.setAttribute('aria-expanded', 'false');
        });
      });

      /* ==================== MODALS ==================== */
      const modalBackdrop = document.getElementById('modalBackdrop');
      const modalTitle = document.getElementById('modalTitle');
      const modalBody = document.getElementById('modalBody');
      const modalClose = document.getElementById('modalClose');

      function openModal(type) {
        if (type === 'handbook') {
          modalTitle.textContent = 'The Plumb Field Guide';
          modalBody.innerHTML = `
            <p class="mono" style="font-size:0.75rem;color:var(--vermilion);margin-bottom:14px;">EXCERPT · CHAPTER 1: KEEP THE LEDGER FLAT</p>
            <p>Most project software grows by accretion. A team adds one custom field for quarterly planning, another for customer tier, and three workflow columns for QA handoffs. Six months later, opening a ticket feels like filing a tax return.</p>
            <p>We built Plumb around three constraints we refuse to relax:</p>
            <p><strong>1. One week at a time.</strong> Two-week sprints hide slippage until day eleven. Weekly cycles keep commitments small enough that everyone remembers why they picked them on Monday.</p>
            <p><strong>2. Flat lists beat nested trees.</strong> Sub-sub-tasks hide real work. If a deliverable takes more than three days, split it into two plain items on the same list.</p>
            <p><strong>3. Written Friday notes beat daily standups.</strong> Synchronous check-ins interrupt makers. A four-minute written digest builds an archive you can actually search six months later.</p>
            <div style="margin-top:20px;display:flex;justify-content:flex-end;">
              <button type="button" class="btn btn-primary" data-switch-modal="signup">Start a workspace</button>
            </div>
          `;
        } else if (type === 'signin') {
          modalTitle.textContent = 'Sign in to Plumb';
          modalBody.innerHTML = `
            <p>Enter your work email and we’ll send a one-time sign-in link, or continue with your workspace SSO.</p>
            <form id="modalAuthForm">
              <div class="field-group">
                <label for="signinEmail">Work email</label>
                <input id="signinEmail" type="email" placeholder="maren@northbound.co" required />
              </div>
              <button type="submit" class="btn btn-primary btn-lg" style="width:100%;">Send sign-in link</button>
            </form>
          `;
        } else {
          modalTitle.textContent = 'Start your Plumb workspace';
          modalBody.innerHTML = `
            <p>Free forever for up to 5 people. Takes 30 seconds — and you can import existing issues from Jira, Linear, or GitHub right after.</p>
            <form id="modalAuthForm">
              <div class="field-group">
                <label for="wsName">Team or studio name</label>
                <input id="wsName" type="text" placeholder="Alder Audio" required />
              </div>
              <div class="field-group">
                <label for="wsEmail">Work email</label>
                <input id="wsEmail" type="email" placeholder="you@company.com" required />
              </div>
              <button type="submit" class="btn btn-primary btn-lg" style="width:100%;">Create workspace</button>
            </form>
          `;
        }

        modalBackdrop.classList.add('open');
        const firstInput = modalBody.querySelector('input');
        if (firstInput) setTimeout(() => firstInput.focus(), 40);

        const switchBtn = modalBody.querySelector('[data-switch-modal]');
        if (switchBtn) {
          switchBtn.addEventListener('click', () => openModal(switchBtn.dataset.switchModal));
        }

        const form = modalBody.querySelector('#modalAuthForm');
        if (form) {
          form.addEventListener('submit', (e) => {
            e.preventDefault();
            const emailVal = form.querySelector('input[type="email"]').value.trim();
            modalBody.innerHTML = `
              <p style="color:var(--moss);font-weight:500;">Link dispatched to ${escapeHtml(emailVal)}.</p>
              <p>Because this is a standalone HTML landing page preview, no real email was sent — but your form flow is wired and ready to connect to your auth endpoint.</p>
              <div style="margin-top:18px;text-align:right;">
                <button type="button" class="btn btn-secondary" id="doneModalBtn">Back to page</button>
              </div>
            `;
            document.getElementById('doneModalBtn').addEventListener('click', closeModal);
          });
        }
      }

      function closeModal() {
        modalBackdrop.classList.remove('open');
      }

      document.querySelectorAll('[data-open-modal]').forEach(el => {
        el.addEventListener('click', (e) => {
          e.preventDefault();
          openModal(el.dataset.openModal);
        });
      });

      document.getElementById('navHandbookLink').addEventListener('click', (e) => {
        e.preventDefault();
        openModal('handbook');
      });

      modalClose.addEventListener('click', closeModal);
      modalBackdrop.addEventListener('click', (e) => {
        if (e.target === modalBackdrop) closeModal();
      });
    })();
