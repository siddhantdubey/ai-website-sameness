      document.getElementById("modal-close").addEventListener("click", closeModal);
      document.getElementById("modal-cancel").addEventListener("click", closeModal);
      modalBackdrop.addEventListener("click", function (e) {
        if (e.target === modalBackdrop) closeModal();
      });

      document.getElementById("modal-form").addEventListener("submit", function (e) {
        e.preventDefault();
        const email = modalEmailInput.value.trim();
        if (modalMode === "signin") {
          modalFeedback.textContent = "If an account exists for " + email + ", a single-use sign-in link has been dispatched.";
        } else {
          const ws = modalWsInput.value.trim();
          modalFeedback.textContent = "Workspace \"" + ws + "\" reserved. Check " + email + " to finish setup.";
        }
        modalSubmit.disabled = true;
      });

      render();
    })();
  </script>
