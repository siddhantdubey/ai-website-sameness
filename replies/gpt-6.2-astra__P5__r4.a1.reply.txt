<script>
const byId = id => document.getElementById(id);
const sample = {
  title: "Print issue 04", brief: "Get the summer issue from final proof to press.", sample: true,
  tasks: [
    ["Approve cover artwork", "JL", true],
    ["Sign off final copy", "MK", true],
    ["Review printer’s proof", "AS", false],
    ["Send files to press", "JL", false]
  ]
};
let project = sample;
try {
  const saved = JSON.parse(localStorage.getItem("relay-sheet-v1"));
  if (saved && typeof saved.title === "string" && typeof saved.brief === "string" &&
      Array.isArray(saved.tasks) && saved.tasks.length && saved.tasks.every(task =>
      Array.isArray(task) && typeof task[0] === "string" && typeof task[1] === "string" &&
      typeof task[2] === "boolean")) project = saved;
} catch {}
function save() {
  try { localStorage.setItem("relay-sheet-v1", JSON.stringify(project)); } catch {}
}
function updateCount() {
  const complete = project.tasks.filter(task => task[2]).length;
  byId("count").textContent = `${complete} of ${project.tasks.length} complete`;
  byId("meter").max = project.tasks.length;
  byId("meter").value = complete;
}
function render() {
  byId("project-title").textContent = project.title;
  byId("brief").textContent = project.brief;
  byId("edition").textContent = project.sample ? "Studio sample" : "Local project";
  byId("tasks").replaceChildren();
  for (const task of project.tasks) {
    const row = document.createElement("label");
    row.className = "task";
    const check = document.createElement("input");
    check.type = "checkbox";
    check.checked = task[2];
    check.addEventListener("change", () => {
      task[2] = check.checked;
      updateCount();
      save();
    });
    const name = document.createElement("span");
    name.className = "name";
    name.textContent = task[0];
    const owner = document.createElement("span");
    owner.className = "owner";
    owner.textContent = task[1];
    row.append(check, name, owner);
    byId("tasks").append(row);
  }
  updateCount();
}
byId("start").onclick = () => {
  if (!project.sample) {
    byId("board").focus();
    return;
  }
  byId("creator").showModal();
};
byId("cancel").onclick = () => byId("creator").close();
byId("form").onsubmit = event => {
  event.preventDefault();
  const field = byId("title-input");
  if (!field.value.trim()) {
    field.setCustomValidity("Enter a project name.");
    field.reportValidity();
    return;
  }
  project = {
    title: field.value.trim(), brief: "Take this project from first brief to final handoff.",
    tasks: [["Write the project brief", "You", false], ["Prepare the first draft", "You", false], ["Review the final handoff", "You", false]]
  };
  save();
  render();
  byId("start").textContent = "Open your project ↗";
  byId("creator").close();
  byId("board").focus();
};
byId("title-input").oninput = event => event.target.setCustomValidity("");
if (!project.sample) byId("start").textContent = "Open your project ↗";
render();
</script>
