      const STATUS_ORDER = ['planned', 'progress', 'done'];
      const STATUS_LABELS = {
        planned: 'Planned',
        progress: 'In Progress',
        done: 'Shipped'
      };
      const TEAM_TAG_CLASS = {
        Engineering: 'tag-eng',
        Design: 'tag-design',
        Ops: 'tag-ops'
      };

      function getFilteredTasks() {
        if (currentFilter === 'all') return tasks;
        return tasks.filter(t => t.team === currentFilter);
      }

      function updateProgress() {
        const total = tasks.length;
        const done = tasks.filter(t => t.status === 'done').length;
        const pct = total ? Math.round((done / total) * 100) : 0;
        progressLabel.textContent = `${done} of ${total} shipped (${pct}%)`;
        progressFill.style.width = `${pct}%`;

        if (pct === 100) {
          cycleStatusPill.textContent = 'Cycle complete';
        } else if (pct >= 50) {
          cycleStatusPill.textContent = 'Ahead of pace';
        } else {
          cycleStatusPill.textContent = 'On track';
        }
      }

      function cycleTaskStatus(taskId) {
        const task = tasks.find(t => t.id === taskId);
        if (!task) return;
        const idx = STATUS_ORDER.indexOf(task.status);
        task.status = STATUS_ORDER[(idx + 1) % STATUS_ORDER.length];
        renderWorkspace();
      }

      function toggleTaskDone(taskId) {
        const task = tasks.find(t => t.id === taskId);
        if (!task) return;
        task.status = task.status === 'done' ? 'progress' : 'done';
        renderWorkspace();
      }

      function escapeHTML(str) {
        return str.replace(/[&<>"']/g, ch => ({
          '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
        })[ch]);
      }

      function renderBoard(list) {
        const cols = [
          { key: 'planned', dot: 'dot-planned', action: 'Start →' },
          { key: 'progress', dot: 'dot-progress', action: 'Ship ✓' },
          { key: 'done', dot: 'dot-done', action: 'Reopen' }
        ];

        const html = cols.map(col => {
          const items = list.filter(t => t.status === col.key);
          const cards = items.map(t => `
            <div class="task-card ${t.status === 'done' ? 'is-done' : ''}">
              <div class="task-top">
                <span class="task-id">${t.id}</span>
                ${t.priority === 'High' ? '<span class="tag tag-high">High</span>' : ''}
              </div>
              <div class="task-title">${escapeHTML(t.title)}</div>
              <div class="task-meta">
                <div class="task-tags">
                  <span class="tag ${TEAM_TAG_CLASS[t.team] || 'tag-eng'}">${t.team}</span>
                </div>
                <div class="task-actions">
                  <button class="advance-btn" type="button" data-advance="${t.id}">${col.action}</button>
                  <span class="avatar" style="background:${t.color};" title="Assigned to ${t.owner}">${t.owner}</span>
                </div>
              </div>
            </div>
          `).join('');

          return `
            <div class="board-col">
              <div class="board-col-head">
                <span class="col-title"><span class="status-dot ${col.dot}"></span>${STATUS_LABELS[col.key]}</span>
                <span class="col-count">${items.length}</span>
              </div>
              ${cards || '<div style="font-size:12.5px; color:var(--ink-muted); padding:14px 6px; text-align:center;">No tasks in this column</div>'}
            </div>
          `;
        }).join('');

        workspaceBody.innerHTML = `<div class="board-grid">${html}</div>`;
      }

      function renderList(list) {
        const rows = list.map(t => `
          <div class="list-row">
            <button class="list-check ${t.status === 'done' ? 'checked' : ''}" type="button" data-toggle-done="${t.id}" aria-label="Toggle ${escapeHTML(t.title)}">
              ${t.status === 'done' ? '<svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="3"><path d="M20 6 9 17l-5-5"/></svg>' : ''}
            </button>
            <span class="task-id list-hide-sm">${t.id}</span>

      function renderTimeline(list) {
        const barColors = {
          planned: '#6E7482',
          progress: '#2956C2',
          done: '#247852'

        const rows = list.map(t => `

            <div class="timeline-label">
              <span class="avatar" style="background:${t.color};">${t.owner}</span>
              <span style="overflow:hidden; text-overflow:ellipsis;">${escapeHTML(t.title)}</span>
            </div>
            <div class="timeline-track">
              <div class="timeline-bar" data-advance="${t.id}" style="left:${t.start}%; width:${t.span}%; background:${barColors[t.status]};" title="Click to advance status">
                <span>${t.id}</span>
                <span>${STATUS_LABELS[t.status]}</span>
              </div>
            </div>
          </div>

          <div class="timeline-wrap">

              <div class="timeline-header">

                <div>Mon 24</div>

                <div>Fri 28</div>
                <div>Tue 2</div>
                <div>Fri 5</div>
              </div>
              ${rows}
            </div>
          </div>
        `;
      }

        else if (currentView === 'list') renderList(list);

      });

      const billingSwitch = document.getElementById('billingSwitch');
      const monthlyLabel = document.getElementById('monthlyLabel');
      const annualLabel = document.getElementById('annualLabel');
      let isAnnual = true;

      function updatePricing() {
        billingSwitch.setAttribute('aria-checked', String(isAnnual));
        annualLabel.style.color = isAnnual ? 'var(--ink)' : 'var(--ink-muted)';
        monthlyLabel.style.color = isAnnual ? 'var(--ink-muted)' : 'var(--ink)';

        document.querySelectorAll('[data-annual]').forEach(el => {
          const val = isAnnual ? el.dataset.annual : el.dataset.monthly;
          el.textContent = `$${val}`;
        });

        document.querySelectorAll('.billing-note').forEach((note, idx) => {
          if (idx === 0) {
            note.textContent = isAnnual ? 'Billed annually · 14-day free trial' : 'Billed monthly · 14-day free trial';
          } else {
            note.textContent = isAnnual ? 'Billed annually · Custom invoicing' : 'Billed monthly · Custom invoicing';
          }
        });

        isAnnual = !isAnnual;
        updatePricing();

         ---------------------------------------------------------- */
      const cmdModal = document.getElementById('cmdModal');
      const cmdSearchInput = document.getElementById('cmdSearchInput');
      const cmdResults = document.getElementById('cmdResults');

      const commands = [
        { label: 'Jump to Interactive Board', hint: 'Hero demo', run: () => { document.getElementById('live-demo').scrollIntoView({ behavior: 'smooth' }); } },
        { label: 'Switch demo to Timeline (Gantt) view', hint: 'Workspace', run: () => { document.querySelector('[data-view="timeline"]').click(); document.getElementById('live-demo').scrollIntoView({ behavior: 'smooth' }); } },
        { label: 'Switch demo to List view', hint: 'Workspace', run: () => { document.querySelector('[data-view="list"]').click(); document.getElementById('live-demo').scrollIntoView({ behavior: 'smooth' }); } },
        { label: 'Explore Product Features', hint: 'Section', run: () => { document.getElementById('features').scrollIntoView({ behavior: 'smooth' }); } },
        { label: 'See Operating Workflow', hint: 'Section', run: () => { document.getElementById('workflow').scrollIntoView({ behavior: 'smooth' }); } },
        { label: 'View Pricing & Toggle Billing', hint: 'Section', run: () => { document.getElementById('pricing').scrollIntoView({ behavior: 'smooth' }); } },
        { label: 'Create Free Workspace', hint: 'Action', run: () => { openSignupModal('Create your free workspace'); } }
      ];

      function renderCommands(query = '') {
        const q = query.trim().toLowerCase();
        const filtered = commands.filter(c => c.label.toLowerCase().includes(q) || c.hint.toLowerCase().includes(q));
        cmdResults.innerHTML = filtered.length
          ? filtered.map((c, idx) => `
              <button class="cmd-option" type="button" data-cmd-idx="${idx}">
                <span>${c.label}</span>
                <small>${c.hint}</small>
              </button>
            `).join('')
          : '<div style="padding:16px; text-align:center; font-size:13.5px; color:var(--ink-muted);">No matching commands</div>';

        cmdResults.querySelectorAll('.cmd-option').forEach((btn, i) => {
          btn.addEventListener('click', () => {
            closeAllModals();
            filtered[i].run();
          });
        });
      }

      function openCmdPalette() {
        closeAllModals();
        cmdModal.classList.add('open');
        cmdSearchInput.value = '';
        renderCommands('');
        setTimeout(() => cmdSearchInput.focus(), 30);
      }

      document.getElementById('openCmdBtn').addEventListener('click', openCmdPalette);
      cmdSearchInput.addEventListener('input', () => renderCommands(cmdSearchInput.value));
      cmdModal.addEventListener('click', (e) => {
        if (e.target === cmdModal) closeAllModals();
      });

      document.addEventListener('keydown', (e) => {
        if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
          e.preventDefault();
          if (cmdModal.classList.contains('open')) closeAllModals();
          else openCmdPalette();
        } else if (e.key === 'Escape') {
          closeAllModals();
        }
      });

         8. SCROLL REVEAL
         ---------------------------------------------------------- */
      if ('IntersectionObserver' in window) {
        const io = new IntersectionObserver((entries) => {
          entries.forEach(entry => {
            if (entry.isIntersecting) {
              entry.target.classList.add('visible');
              io.unobserve(entry.target);
            }
          });
        }, { threshold: 0.12 });

        document.querySelectorAll('.reveal').forEach(el => io.observe(el));
      } else {
        document.querySelectorAll('.reveal').forEach(el => el.classList.add('visible'));
      }
    })();
  </script>
