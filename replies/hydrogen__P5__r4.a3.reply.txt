      <div class="container">
        <div class="section-header">
          <div class="section-index">02 / PRINCIPLES</div>
          <div>
            <h2>Built around how real teams actually finish things.</h2>
            <p class="section-lead">
              We removed every mechanism designed for middle-management reporting and kept the tools that help builders stay aligned.
            </p>
          </div>
        </div>

        <div class="rules-grid">
          <article class="rule-item">
            <span class="rule-num">RULE 01</span>
            <h3>Cycles close on Friday. Unfinished work doesn't auto-roll.</h3>
            <p>
              Automatic carry-over hides scope creep. At the end of a weekly cycle, anything left open goes back to the bench for a deliberate decision: finish it Monday, or drop it honestly.
            </p>
          </article>

          <article class="rule-item">
            <span class="rule-num">RULE 02</span>
            <h3>One owner per line item. Never a committee.</h3>
            <p>
              Multiple assignees mean nobody feels accountable when a deadline slips. Multiple people can comment and commit code, but one name sits next to the deliverable.
            </p>
          </article>

          <article class="rule-item">
            <span class="rule-num">RULE 03</span>
            <h3>Speed is not polish—it is the product.</h3>
            <p>
              Every navigation, filter, and status change runs optimistically in under 30ms with local cache sync. When opening a tool feels instant, people actually keep it accurate.
            </p>
          </article>

          <article class="rule-item">
            <span class="rule-num">RULE 04</span>
            <h3>Your data belongs in plain files, not a hostage database.</h3>
            <p>
              One-click export dumps your entire workspace as clean, human-readable Markdown folders and a single SQLite archive. Leave anytime without a migration script.
            </p>
          </article>
        </div>
      </div>
    </section>

        <button type="button" class="btn btn-secondary btn-sm" id="modal-done-btn">Back to page</button>
      </div>
    </form>
  </dialog>

  <script>
    (function () {
      const initialTasks = [
        {
          id: "DKT-414",
          title: "Debounce motor controller fault telemetry over CAN bus",
          area: "Firmware",
          owner: "H. Lindqvist",
          status: "shipped",
          due: "Tue, Nov 25",
          notes: "Filtered transient voltage drops under 12ms. Tested across 4 bench rigs overnight without false positives."
        },
        {
          id: "DKT-416",
          title: "Print-ready invoice PDF export with VAT breakdown",
          area: "Design",
          owner: "D. Park",
          status: "shipped",
          due: "Wed, Nov 26",
          notes: "Switched to monospace tabular numerals for line totals and fixed page break on 20+ row invoices."
        },
        {
          id: "DKT-418",
          title: "Idempotency keys on webhook retries",
          area: "Backend",
          owner: "M. Sato",
          status: "motion",
          due: "Thu, Nov 27",
          notes: "Payment webhooks double-fired twice during Tuesday's gateway blip. Backing dedup checks with existing Redis keyspace (24h TTL)."
        },
        {
          id: "DKT-419",
          title: "Zero-downtime index backfill on events_archive",
          area: "Backend",
          owner: "R. Okafor",
          status: "motion",
          due: "Thu, Nov 27",
          notes: "Running batched 5,000-row chunks during off-peak hours. Currently at 64% completion with <2% CPU overhead."
        },
        {
          id: "DKT-420",
          title: "Keyboard focus ring audit on ledger row drawer",
          area: "Design",
          owner: "D. Park",
          status: "queued",
          due: "Fri, Nov 28",
          notes: "Ensure Tab and J/K navigation stay synchronized when switching from Sheet to Columns view."
        },
        {
          id: "DKT-421",
          title: "Verify bootloader fallback on Rev-C board",
          area: "Firmware",
          owner: "H. Lindqvist",
          status: "queued",
          due: "Fri, Nov 28",
          notes: "Need to pull two Rev-C prototypes from assembly tray before Friday afternoon sign-off."
        }
      ];

      let tasks = initialTasks.map(t => ({ ...t }));
      let selectedId = "DKT-418";
      let currentView = "sheet";
      let currentFilter = "All";
      let nextNum = 422;

      const sheetEl = document.getElementById("view-sheet");
      const boardEl = document.getElementById("view-board");
      const changelogEl = document.getElementById("view-changelog");
      const progressLabel = document.getElementById("progress-label");
      const progressFill = document.getElementById("progress-fill");

      const inspId = document.getElementById("insp-id");
      const inspArea = document.getElementById("insp-area");
      const inspTitle = document.getElementById("insp-title");
      const inspStatus = document.getElementById("insp-status");
      const inspOwner = document.getElementById("insp-owner");
      const inspDue = document.getElementById("insp-due");
      const inspNotes = document.getElementById("insp-notes");

      const statusLabels = {
        queued: "Queued",
        motion: "In Motion",
        shipped: "Shipped"
      };

      const nextStatusMap = {
        queued: "motion",
        motion: "shipped",
        shipped: "queued"
      };

      function getFilteredTasks() {
        if (currentFilter === "All") return tasks;
        return tasks.filter(t => t.area === currentFilter);
      }

      function updateProgress() {
        const total = tasks.length;
        const shipped = tasks.filter(t => t.status === "shipped").length;
        const pct = total ? Math.round((shipped / total) * 100) : 0;
        progressLabel.textContent = `${shipped} / ${total} shipped`;
        progressFill.style.width = `${pct}%`;
      }

      function renderInspector() {
        const item = tasks.find(t => t.id === selectedId) || tasks[0];
        if (!item) return;
        selectedId = item.id;
        inspId.textContent = item.id;
        inspArea.textContent = item.area;
        inspTitle.textContent = item.title;
        inspStatus.value = item.status;
        inspOwner.value = item.owner;
        inspDue.textContent = item.due;
        inspNotes.textContent = item.notes;
      }

      function renderSheet() {
        const list = getFilteredTasks();
        sheetEl.innerHTML = "";
        if (!list.length) {
          sheetEl.innerHTML = `<div style="padding: 28px 18px; color: var(--ink-3); font-size: 13.5px;">No items in ${currentFilter} for Cycle 42.</div>`;
          return;
        }
        list.forEach(task => {
          const row = document.createElement("div");
          row.className = `task-row${task.id === selectedId ? " selected" : ""}${task.status === "shipped" ? " is-shipped" : ""}`;
          row.setAttribute("role", "listitem");

          const check = document.createElement("button");
          check.type = "button";
          check.className = "status-check";
          check.dataset.status = task.status;
          check.setAttribute("aria-label", `Cycle status for ${task.id}`);
          check.addEventListener("click", e => {
            e.stopPropagation();
            task.status = nextStatusMap[task.status];
            selectedId = task.id;
            renderAll();
          });

          const idSpan = document.createElement("span");
          idSpan.className = "task-id";
          idSpan.textContent = task.id;

          const titleSpan = document.createElement("span");
          titleSpan.className = "task-title";
          titleSpan.textContent = task.title;

          const ownerSpan = document.createElement("span");
          ownerSpan.className = "task-owner";
          ownerSpan.textContent = task.owner;

          const badge = document.createElement("span");
          badge.className = `status-badge badge-${task.status}`;
          badge.textContent = statusLabels[task.status];

          row.append(check, idSpan, titleSpan, ownerSpan, badge);
          row.addEventListener("click", () => {
            selectedId = task.id;
            renderAll();
          });
          sheetEl.appendChild(row);
        });
      }

      function renderBoard() {
        const list = getFilteredTasks();
        const cols = [
          { key: "queued", label: "Queued" },
          { key: "motion", label: "In Motion" },
          { key: "shipped", label: "Shipped" }
        ];
        boardEl.innerHTML = "";
        cols.forEach(col => {
          const colTasks = list.filter(t => t.status === col.key);
          const colEl = document.createElement("div");
          colEl.className = "board-col";
          colEl.innerHTML = `<div class="col-head"><span>${col.label}</span><span>${colTasks.length}</span></div>`;
          colTasks.forEach(task => {
            const card = document.createElement("div");
            card.className = `board-card${task.id === selectedId ? " selected" : ""}`;
            card.innerHTML = `
              <div class="board-card-top"><span>${task.id}</span><span>${task.area}</span></div>
              <div class="board-card-title"></div>
              <div class="board-card-foot"><span>${task.owner}</span><span style="font-family:var(--mono);font-size:11px;">${task.due}</span></div>
            `;
            card.querySelector(".board-card-title").textContent = task.title;
            card.addEventListener("click", () => {
              selectedId = task.id;
              renderAll();
            });
            colEl.appendChild(card);
          });
          boardEl.appendChild(colEl);
        });
      }

      function renderChangelog() {
        const shipped = getFilteredTasks().filter(t => t.status === "shipped");
        changelogEl.innerHTML = "";
        if (!shipped.length) {
          changelogEl.innerHTML = `<div style="color: var(--ink-3); font-size: 13.5px;">No shipped items yet for this filter. Mark a task as Shipped to see it appear here.</div>`;
          return;
        }
        shipped.forEach(task => {
          const entry = document.createElement("div");
          entry.className = "changelog-entry";
          entry.innerHTML = `
            <div class="changelog-time">${task.due}</div>
            <div>
              <div style="font-weight: 600; font-size: 14px; margin-bottom: 4px;"></div>
              <div style="font-size: 13px; color: var(--ink-2);"></div>
              <div style="font-family: var(--mono); font-size: 11px; color: var(--ink-3); margin-top: 6px;">${task.id} &middot; ${task.owner} &middot; ${task.area}</div>
            </div>
          `;
          entry.children[1].children[0].textContent = task.title;
          entry.children[1].children[1].textContent = task.notes;
          changelogEl.appendChild(entry);
        });
      }

      function renderAll() {
        updateProgress();
        renderInspector();
        sheetEl.classList.toggle("hidden", currentView !== "sheet");
        boardEl.classList.toggle("hidden", currentView !== "board");
        changelogEl.classList.toggle("hidden", currentView !== "changelog");

        if (currentView === "sheet") renderSheet();
        if (currentView === "board") renderBoard();
        if (currentView === "changelog") renderChangelog();
      }

      // View tabs
      document.querySelectorAll("[data-view]").forEach(btn => {
        btn.addEventListener("click", () => {
          currentView = btn.dataset.view;
          document.querySelectorAll("[data-view]").forEach(b => {
            b.classList.toggle("active", b === btn);
            b.setAttribute("aria-selected", String(b === btn));
          });
          renderAll();
        });
      });

      // Area filters
      document.querySelectorAll("[data-filter]").forEach(btn => {
        btn.addEventListener("click", () => {
          currentFilter = btn.dataset.filter;
          document.querySelectorAll("[data-filter]").forEach(b => b.classList.toggle("active", b === btn));
          renderAll();
        });
      });

      // Quick add form
      const quickForm = document.getElementById("quick-add-form");
      const quickInput = document.getElementById("quick-add-input");
      const quickArea = document.getElementById("quick-add-area");

      quickForm.addEventListener("submit", e => {
        e.preventDefault();
        const title = quickInput.value.trim();
        if (!title) return;
        const newId = `DKT-${nextNum++}`;
        const newTask = {
          id: newId,
          title,
          area: quickArea.value,
          owner: "M. Sato",
          status: "queued",
          due: "Fri, Nov 28",
          notes: "Added directly to Cycle 42 from the live sheet."
        };
        tasks.unshift(newTask);
        selectedId = newId;
        quickInput.value = "";
        if (currentFilter !== "All" && currentFilter !== newTask.area) {
          currentFilter = "All";
          document.querySelectorAll("[data-filter]").forEach(b => b.classList.toggle("active", b.dataset.filter === "All"));
        }
        renderAll();
      });

      // Inspector updates
      inspStatus.addEventListener("change", () => {
        const item = tasks.find(t => t.id === selectedId);
        if (item) {
          item.status = inspStatus.value;
          renderAll();
        }
      });

      inspOwner.addEventListener("change", () => {
        const item = tasks.find(t => t.id === selectedId);
        if (item) {
          item.owner = inspOwner.value;
          renderAll();
        }
      });

      // Keyboard shortcuts (1, 2, 3, C) when not typing
      document.addEventListener("keydown", e => {
        const tag = document.activeElement.tagName;
        if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
        if (e.key === "1") document.querySelector('[data-view="sheet"]').click();
        if (e.key === "2") document.querySelector('[data-view="board"]').click();
        if (e.key === "3") document.querySelector('[data-view="changelog"]').click();
        if (e.key.toLowerCase() === "c") {
          e.preventDefault();
          quickInput.focus();
        }
      });

      // Anatomy comparison toggle
      const docketBox = document.getElementById("anatomy-docket");
      const typicalBox = document.getElementById("anatomy-typical");
      document.querySelectorAll("[data-anatomy]").forEach(btn => {
        btn.addEventListener("click", () => {
          const mode = btn.dataset.anatomy;
          document.querySelectorAll("[data-anatomy]").forEach(b => b.classList.toggle("active", b === btn));
          docketBox.classList.toggle("hidden", mode !== "docket");
          typicalBox.classList.toggle("hidden", mode !== "typical");
        });
      });

      // Workspace Modal
      const modal = document.getElementById("workspace-modal");
      const wsName = document.getElementById("ws-name");
      const wsEmail = document.getElementById("ws-email");
      const slugPreview = document.getElementById("slug-preview");
      const modalFields = document.getElementById("modal-form-fields");
      const modalSuccess = document.getElementById("modal-success");

      document.querySelectorAll("[data-open-modal]").forEach(btn => {
        btn.addEventListener("click", () => {
          modalFields.classList.remove("hidden");
          modalSuccess.classList.add("hidden");
          modal.showModal();
          wsName.focus();
        });
      });

      document.getElementById("modal-close-btn").addEventListener("click", () => modal.close());
      document.getElementById("modal-done-btn").addEventListener("click", () => modal.close());
      modal.addEventListener("click", e => {
        if (e.target === modal) modal.close();
      });

      function slugify(val) {
        return val.toLowerCase().trim().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "") || "your-team";
      }

      wsName.addEventListener("input", () => {
        slugPreview.textContent = `docket.work/${slugify(wsName.value)}`;
      });

      document.getElementById("workspace-form").addEventListener("submit", e => {
        e.preventDefault();
        document.getElementById("confirm-slug").textContent = `docket.work/${slugify(wsName.value)}`;
        document.getElementById("confirm-email").textContent = wsEmail.value.trim();
        modalFields.classList.add("hidden");
        modalSuccess.classList.remove("hidden");
      });

      renderAll();
    })();
  </script>
