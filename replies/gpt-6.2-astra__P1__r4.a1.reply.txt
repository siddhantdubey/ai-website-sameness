  <script>
    const stages = ['To do', 'In progress', 'Done'];
    const seeds = [
      ['Draft launch notes', 'Maya', 'Writing', 'To do'],
      ['Review mobile layouts', 'Jules', 'Design', 'In progress'],
      ['Approve project brief', 'Sam', 'Planning', 'Done']
    ];
    const key = 'relay-launch-board-v1';
    let tasks = seeds;
    try {
      const saved = JSON.parse(localStorage.getItem(key));
      if (Array.isArray(saved) && saved.every(t => Array.isArray(t) && t.length === 4 &&
          t.every(v => typeof v === 'string') && stages.includes(t[3]))) tasks = saved;
    } catch {}
    const get = id => document.getElementById(id);
    const dialog = get('editor');
    let editing = -1;
    function element(tag, className, value) {
      const node = document.createElement(tag);
      node.className = className;
      node.textContent = value;
      return node;
    }
    function openEditor(index) {
      editing = index;
      const task = tasks[index] || ['', '', 'Task', stages[0]];
      get('editor-title').textContent = index < 0 ? 'Add task' : 'Edit task';
      get('task-name').value = task[0];
      get('owner').value = task[1];
      get('stage').value = task[3];
      dialog.showModal();
    }
    function render() {
      const board = get('board');
      board.replaceChildren();
      for (const stage of stages) {
        const column = element('section', 'column', '');
        const head = element('h3', 'column-head', '');
        head.append(element('span', 'dot', ''), stage,
          element('span', 'count', tasks.filter(task => task[3] === stage).length));
        column.append(head);
        tasks.forEach(([title, owner, tag, status], index) => {
          if (status !== stage) return;
          const card = element('button', 'task', '');
          card.dataset.index = index;
          card.append(element('span', 'tag', tag), element('strong', '', title), element('span', 'owner', owner));
          card.onclick = () => openEditor(index);
          column.append(card);
        });
        board.append(column);
      }
    }
    get('add').onclick = () => openEditor(-1);
    get('cancel').onclick = () => dialog.close();
    get('form').onsubmit = event => {
      event.preventDefault();
      const title = get('task-name').value.trim();
      const owner = get('owner').value.trim();
      if (!title || !owner) {
        const field = get(!title ? 'task-name' : 'owner');
        field.value = '';
        field.reportValidity();
        return;
      }
      const index = editing < 0 ? tasks.length : editing;
      tasks[index] = [title, owner, tasks[index]?.[2] || 'Task', get('stage').value];
      try { localStorage.setItem(key, JSON.stringify(tasks)); }
      catch { get('status').textContent = 'Changes are available for this visit only.'; }
      dialog.close();
      render();
      document.querySelector(`[data-index="${index}"]`).focus();
    };
    render();
  </script>
