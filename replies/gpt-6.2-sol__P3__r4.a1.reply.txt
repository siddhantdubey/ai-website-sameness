    function drawScene(canvas, scene) {
      const ctx = canvas.getContext('2d');
      const { width, height } = canvas;
      let state = scene.seed;

      function random() {
        state = (Math.imul(state, 1664525) + 1013904223) >>> 0;
        return state / 4294967296;
      }

      function gradient(top, bottom, colors) {
        const fill = ctx.createLinearGradient(0, top * height, 0, bottom * height);
        fill.addColorStop(0, colors[0]);
        fill.addColorStop(1, colors[1]);
        return fill;
      }

      ctx.fillStyle = gradient(0, .64, scene.sky);
      ctx.fillRect(0, 0, width, height);

      ctx.save();
      ctx.scale(width, height);
      for (let i = 0; i < 150; i++) {
        const x = random();
        const y = random() * .55;
        const spread = .025 + random() * .22;
        const haze = ctx.createRadialGradient(x, y, 0, x, y, spread);
        haze.addColorStop(0, i % 5 === 0 ? '#58626e12' : '#fff0cb16');
        haze.addColorStop(1, '#fff0cb00');
        ctx.fillStyle = haze;
        ctx.save();
        ctx.translate(0, y);
        ctx.scale(1, .15);
        ctx.translate(0, -y);
        ctx.fillRect(x - spread, y - spread, spread * 2, spread * 2);
        ctx.restore();
      }
      ctx.restore();

      if (scene.water) {
        const horizon = scene.horizon * height;
        ctx.fillStyle = gradient(scene.horizon, 1, scene.water);
        ctx.fillRect(0, horizon, width, height - horizon);
        for (let i = 0; i < 1500; i++) {
          const depth = random();
          const x = random() * width;
          const y = horizon + depth * (height - horizon);
          ctx.fillStyle = i % 4 === 0 ? '#eff1d455' : '#d7e1cd1d';
          ctx.fillRect(x, y, (3 + random() * width * .05) * (.15 + depth), .5 + depth);
        }
        ctx.save();
        ctx.scale(width, height);
        for (let i = 0; i < 7; i++) {
          const offset = i * .012;
          ctx.beginPath();
          ctx.moveTo(.45 + offset, .56);
          ctx.bezierCurveTo(.69 + offset, .68, .35 + offset, .7, .31 + offset, 1);
          ctx.strokeStyle = i % 2 ? '#ebf1db42' : '#dfe8d56b';
          ctx.lineWidth = .0015 + random() * .002;
          ctx.stroke();
        }
        ctx.restore();
      }

      for (const ridge of scene.ridges) {
        const first = ridge.points[0];
        const last = ridge.points[ridge.points.length - 1];
        ctx.beginPath();
        ctx.moveTo(first[0] * width, first[1] * height);

        for (let i = 1; i < ridge.points.length; i++) {
          const [x0, y0] = ridge.points[i - 1];
          const [x1, y1] = ridge.points[i];
          const steps = Math.max(3, Math.ceil(Math.hypot(x1 - x0, y1 - y0) * 155));
          for (let j = 1; j <= steps; j++) {
            const t = j / steps;
            const roughness = Math.sin(Math.PI * t) * ridge.rough;
            const y = y0 + (y1 - y0) * t + (random() - .5) * roughness;
            ctx.lineTo((x0 + (x1 - x0) * t) * width, y * height);
          }
        }
        ctx.lineTo(last[0] * width, height);
        ctx.lineTo(first[0] * width, height);
        ctx.closePath();
        ctx.fillStyle = gradient(.39, 1, ridge.colors);
        ctx.fill();

        ctx.save();
        ctx.clip();
        for (let i = 0; i < 4700; i++) {
          const x = random() * width;
          const y = (.36 + random() * .64) * height;
          ctx.fillStyle = i % 3 ? '#181c1b13' : '#fff4d320';
          ctx.fillRect(x, y, 1 + random() * width * .015, .6 + random() * 2);
        }
        ctx.restore();
      }

      const pixels = ctx.getImageData(0, 0, width, height);
      for (let i = 0; i < pixels.data.length; i += 4) {
        const grain = (random() - .5) * 17;
        pixels.data[i] += grain;
        pixels.data[i + 1] += grain;
        pixels.data[i + 2] += grain;
      }
      ctx.putImageData(pixels, 0, 0);
    }

    for (const canvas of document.querySelectorAll('canvas[data-scene]')) {
      drawScene(canvas, scenes[canvas.dataset.scene]);
    }

    const dialog = document.querySelector('#lightbox');
    const preview = document.querySelector('#preview');
    const title = document.querySelector('#preview-title');

    for (const button of document.querySelectorAll('button[data-title]')) {
      button.addEventListener('click', () => {
        const original = button.querySelector('canvas');
        preview.width = original.width;
        preview.height = original.height;
        preview.setAttribute('aria-label', original.getAttribute('aria-label'));
        preview.getContext('2d').drawImage(original, 0, 0);
        title.textContent = button.dataset.title;
        dialog.showModal();
      });
    }

    dialog.addEventListener('click', (event) => {
      if (event.target !== dialog) return;
      const bounds = dialog.getBoundingClientRect();
      if (event.clientX < bounds.left || event.clientX > bounds.right ||
          event.clientY < bounds.top || event.clientY > bounds.bottom) {
        dialog.close();
      }
    });
  </script>
