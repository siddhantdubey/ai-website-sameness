  <script>
    // Initial Sandbox Issues Data
    let tasks = [
      { id: 'KNX-201', title: 'Implement zero-copy WebSocket frame serializer', status: 'todo', priority: 'URGENT', points: 8, owner: 'EV', tag: 'Rust Core' },
      { id: 'KNX-205', title: 'Add keyboard shortcut chord G+V for Cycle Telemetry', status: 'todo', priority: 'NORMAL', points: 3, owner: 'MC', tag: 'Frontend' },
      { id: 'KNX-198', title: 'CRDT state reconciliation for offline IndexedDB cache', status: 'progress', priority: 'URGENT', points: 13, owner: 'MC', tag: 'Sync Engine' },
      { id: 'KNX-202', title: 'Automated GitHub PR preview environment linker', status: 'progress', priority: 'HIGH', points: 5, owner: 'AP', tag: 'Integrations' },
      { id: 'KNX-190', title: 'Sub-millisecond fuzzy search index in WebAssembly', status: 'done', priority: 'HIGH', points: 8, owner: 'EV', tag: 'Performance' },
      { id: 'KNX-194', title: 'SOC2 Type II automated audit log exporter', status: 'done', priority: 'NORMAL', points: 5, owner: 'AP', tag: 'Security' },
      { id: 'KNX-196', title: 'Monte Carlo burnup confidence interval shader', status: 'done', priority: 'HIGH', points: 8, owner: 'MC', tag: 'Analytics' }
    ];

    let nextIssueNum = 206;

    function getPriorityBadge(priority) {
      if (priority === 'URGENT') {
        return `<span class="px-1.5 py-0.5 rounded bg-rose-500/20 text-rose-300 border border-rose-500/30 font-mono text-[10px]">URGENT</span>`;
      }
      if (priority === 'HIGH') {
        return `<span class="px-1.5 py-0.5 rounded bg-amber-500/20 text-amber-300 border border-amber-500/30 font-mono text-[10px]">HIGH</span>`;
      }
      return `<span class="px-1.5 py-0.5 rounded bg-white/5 text-gray-400 border border-white/10 font-mono text-[10px]">NORMAL</span>`;
    }

    function advanceTaskStatus(id) {
      const order = ['todo', 'progress', 'done'];
      tasks = tasks.map(t => {
        if (t.id === id) {
          const nextIdx = (order.indexOf(t.status) + 1) % order.length;
          return { ...t, status: order[nextIdx] };
        }
        return t;
      });
      renderWorkspace();
    }

    function renderWorkspace() {
      const filter = document.getElementById('priority-filter').value;
      const cols = { todo: [], progress: [], done: [] };

      tasks.forEach(task => {
        if (filter !== 'ALL' && task.priority !== filter) return;
        cols[task.status].push(task);
      });

      ['todo', 'progress', 'done'].forEach(status => {
        const container = document.getElementById(`col-${status}`);
        const countEl = document.getElementById(`count-${status}`);
        countEl.textContent = cols[status].length;

        container.innerHTML = cols[status].map(task => {
          const actionLabel = status === 'todo' ? 'Start →' : status === 'progress' ? 'Deploy ✓' : 'Reopen ↺';
          const statusBtnColor = status === 'done'
            ? 'bg-emerald-500/15 text-emerald-300 border-emerald-500/30 hover:bg-emerald-500/25'
            : 'bg-indigo-500/15 text-indigo-300 border-indigo-500/30 hover:bg-indigo-500/30';

          return `
            <div class="p-3.5 rounded-xl bg-obsidian-950/90 border border-white/10 hover:border-indigo-500/40 transition group">
              <div class="flex items-center justify-between text-xs mb-2">
                <span class="font-mono text-gray-400">${task.id}</span>
                <div class="flex items-center space-x-1.5">
                  ${getPriorityBadge(task.priority)}
                  <span class="font-mono text-[10px] px-1.5 py-0.5 rounded bg-white/5 text-gray-300">${task.points} pts</span>
                </div>
              </div>
              <p class="text-sm font-medium text-gray-100 leading-snug">${task.title}</p>
              <div class="mt-3 pt-2.5 border-t border-white/5 flex items-center justify-between">
                <div class="flex items-center space-x-2">
                  <span class="w-5 h-5 rounded-full bg-indigo-600/40 border border-indigo-400/40 text-[10px] font-mono flex items-center justify-center text-white">${task.owner}</span>
                  <span class="text-[11px] font-mono text-gray-400">${task.tag}</span>
                </div>
                <button onclick="advanceTaskStatus('${task.id}')" class="text-[11px] font-mono px-2 py-0.5 rounded border transition ${statusBtnColor}">
                  ${actionLabel}
                </button>
              </div>
            </div>
          `;
        }).join('');
      });

      // Update live cycle velocity bar
      const totalPts = tasks.reduce((acc, t) => acc + Number(t.points), 0);
      const donePts = tasks.filter(t => t.status === 'done').reduce((acc, t) => acc + Number(t.points), 0);
      const pct = totalPts > 0 ? Math.round((donePts / totalPts) * 100) : 0;

      document.getElementById('metric-velocity').textContent = `${donePts} / ${totalPts} pts`;
      document.getElementById('metric-progress-bar').style.width = `${pct}%`;
      document.getElementById('metric-percent').textContent = `${pct}%`;
    }

    function switchWorkspaceView(view) {
      ['board', 'timeline', 'analytics'].forEach(v => {
        const el = document.getElementById(`view-${v}`);
        const tab = document.getElementById(`tab-${v}`);
        if (v === view) {
          el.classList.remove('hidden');
          tab.className = 'px-3 py-1.5 rounded-md bg-indigo-600 text-white shadow transition flex items-center space-x-1.5';
        } else {
          el.classList.add('hidden');
          tab.className = 'px-3 py-1.5 rounded-md text-gray-400 hover:text-white transition flex items-center space-x-1.5';
        }
      });
    }

    function openTaskModal(defaultStatus = 'progress') {
      document.getElementById('new-task-status').value = defaultStatus;
      const modal = document.getElementById('task-modal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
      setTimeout(() => document.getElementById('new-task-title').focus(), 50);
    }

    function closeTaskModal() {
      const modal = document.getElementById('task-modal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    function createNewTask(e) {
      e.preventDefault();
      const title = document.getElementById('new-task-title').value.trim();
      const status = document.getElementById('new-task-status').value;
      const priority = document.getElementById('new-task-priority').value;
      const points = Number(document.getElementById('new-task-points').value) || 3;

      if (!title) return;

      tasks.unshift({
        id: `KNX-${nextIssueNum++}`,
        title,
        status,
        priority,
        points,
        owner: 'YOU',
        tag: 'Cycle 24'
      });

      document.getElementById('new-task-title').value = '';
      closeTaskModal();
      switchWorkspaceView('board');
      renderWorkspace();
      document.getElementById('interactive-demo').scrollIntoView({ behavior: 'smooth' });
    }

    // Command Palette (Cmd+K / Ctrl+K)
    function toggleCommandModal() {
      const modal = document.getElementById('cmd-modal');
      if (modal.classList.contains('hidden')) {
        modal.classList.remove('hidden');
        modal.classList.add('flex');
        setTimeout(() => document.getElementById('cmd-input').focus(), 50);
      } else {
        modal.classList.add('hidden');
        modal.classList.remove('flex');
      }
    }

    window.addEventListener('keydown', (e) => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
        e.preventDefault();
        toggleCommandModal();
      } else if (e.key === 'Escape') {
        document.getElementById('cmd-modal').classList.add('hidden');
        document.getElementById('cmd-modal').classList.remove('flex');
        closeTaskModal();
      }
    });

    // ROI Calculator
    function updateROICalculator(headcount) {
      const count = Number(headcount);
      document.getElementById('team-size-label').textContent = `${count} Engineers`;
      const hoursSaved = count * 156; // 3 hrs/wk * 52 wks
      const dollarsSaved = hoursSaved * 85;
      document.getElementById('roi-hours').textContent = `${hoursSaved.toLocaleString()} hrs`;
      document.getElementById('roi-dollars').textContent = `$${dollarsSaved.toLocaleString()}`;
    }

    // Pricing Billing Toggle
    function setBilling(mode) {
      const btnMonthly = document.getElementById('btn-monthly');
      const btnAnnual = document.getElementById('btn-annual');
      if (mode === 'annual') {
        btnAnnual.className = 'px-4 py-2 rounded-lg text-xs font-medium bg-indigo-600 text-white shadow transition flex items-center space-x-1.5';
        btnMonthly.className = 'px-4 py-2 rounded-lg text-xs font-medium text-gray-400 hover:text-white transition';
        document.getElementById('price-pro').textContent = '$12';
        document.getElementById('price-ent').textContent = '$28';
      } else {
        btnMonthly.className = 'px-4 py-2 rounded-lg text-xs font-medium bg-indigo-600 text-white shadow transition';
        btnAnnual.className = 'px-4 py-2 rounded-lg text-xs font-medium text-gray-400 hover:text-white transition flex items-center space-x-1.5';
        document.getElementById('price-pro').textContent = '$15';
        document.getElementById('price-ent').textContent = '$35';
      }
    }

    // Initialize workspace on load
    renderWorkspace();
  </script>
