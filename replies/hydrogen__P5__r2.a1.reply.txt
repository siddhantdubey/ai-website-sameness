      const modal = document.getElementById("workspaceModal");
      const modalTitle = document.getElementById("modalTitle");
      const modalBody = document.getElementById("modalBody");
      const initialModalHtml = modalBody.innerHTML;

      function bindModalForm() {
        const wsName = document.getElementById("wsName");
        const wsSlug = document.getElementById("wsSlug");
        const form = document.getElementById("workspaceForm");
        const cancelBtn = document.getElementById("cancelModalBtn");

        if (wsName && wsSlug) {
          wsName.addEventListener("input", function () {
            const slug = wsName.value
              .toLowerCase()
              .trim()
              .replace(/[^a-z0-9]+/g, "-")
              .replace(/^-|-$/g, "");
            wsSlug.textContent = slug || "foundry-robotics";
          });
        }
        if (cancelBtn) {
          cancelBtn.addEventListener("click", closeModal);
        }
        if (form) {
          form.addEventListener("submit", function (e) {
            e.preventDefault();
            const slug = (document.getElementById("wsSlug") || {}).textContent || "team";
            const email = (document.getElementById("wsEmail") || {}).value || "";
            modalBody.innerHTML = `
              <div style="padding:8px 0;">
                <p class="mono" style="font-size:11.5px;color:var(--green);margin:0 0 8px;">WORKSPACE RESERVED</p>
                <p style="margin:0 0 12px;font-size:14.5px;">
                  We sent a magic sign-in link to <strong>${escapeHtml(email)}</strong> for <span class="mono">docket.app/${escapeHtml(slug)}</span>.
                </p>
                <button type="button" class="btn btn-primary" id="doneModalBtn">Back to page</button>
              </div>
            `;
            document.getElementById("doneModalBtn").addEventListener("click", closeModal);
          });
        }
      }

      function openModal(mode, plan) {
        modalBody.innerHTML = initialModalHtml;
        if (mode === "signin") {
          modalTitle.textContent = "Sign in to Docket";
          const btn = document.getElementById("submitModalBtn");
          if (btn) btn.textContent = "Send sign-in link";
        } else {
          modalTitle.textContent = plan ? `Start ${plan} workspace` : "Create your Docket workspace";
        }
        bindModalForm();
        modal.classList.add("is-open");
        const firstInput = modal.querySelector("input");
        if (firstInput) firstInput.focus();
      }

      function closeModal() {
        modal.classList.remove("is-open");
      }

      document.addEventListener("click", function (e) {
        const trigger = e.target.closest("[data-open-modal]");
        if (trigger) {
          e.preventDefault();
          openModal(trigger.getAttribute("data-open-modal"), trigger.getAttribute("data-plan"));
        }
      });

      document.getElementById("closeModalBtn").addEventListener("click", closeModal);
      modal.addEventListener("click", function (e) {
        if (e.target === modal) closeModal();
      });
      document.addEventListener("keydown", function (e) {
        if (e.key === "Escape" && modal.classList.contains("is-open")) {
          closeModal();
        }
      });
    })();
