      function updateProgress() {
        const doneCount = tasks.filter(t => t.status === 'done').length;
        const pct = Math.round((doneCount / tasks.length) * 100);
        wsProgressLabel.textContent = `${doneCount} / ${tasks.length} done`;

      function cycleStatus(task) {
        if (task.status === 'blocked') {
          task.status = 'progress';
          task.overdue = false;
        } else {
          const idx = STATUS_ORDER.indexOf(task.status);
          task.status = STATUS_ORDER[(idx + 1) % STATUS_ORDER.length];
        }
        renderWorkspace();
      }

      function renderInspector() {
        const task = tasks.find(t => t.id === selectedTaskId) || tasks[0];
        if (!task) return;
        const meta = STATUS_META[task.status];

        const statusButtonsHtml = Object.keys(STATUS_META).map(key => {
          const active = task.status === key ? 'active' : '';
          return `<button type="button" class="insp-status-btn ${active}" data-set-status="${key}">${STATUS_META[key].label}</button>`;
        }).join('');

        const depsHtml = task.deps.length
          ? task.deps.map(depId => {
              const dt = tasks.find(x => x.id === depId);
              if (!dt) return '';
              return `<button type="button" class="insp-dep-btn" data-jump-task="${dt.id}">
                <span><strong class="mono">${dt.id}</strong> ${dt.title}</span>
                <span class="mono" style="font-size:10.5px;color:var(--ink-3)">↗</span>
              </button>`;
            }).join('')
          : `<div class="mono" style="font-size:11.5px;color:var(--ink-3)">No upstream blockers</div>`;

        wsInspector.innerHTML = `
          <div class="insp-top">
            <span>${task.id} · ${task.milestone}</span>
            <span class="status-chip ${meta.cls}"><span class="status-dot"></span>${meta.label}</span>

          <h3 class="insp-title">${task.title}</h3>
          <div class="insp-meta-grid">

        currentView = view;

        renderWorkspace();

        if (e.key === '2') setView('timeline');
