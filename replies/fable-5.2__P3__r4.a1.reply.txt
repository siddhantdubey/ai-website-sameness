(function () {
  var root = document.documentElement;

  /* Light table toggle */
  var lamp = document.getElementById('lamp');
  var saved = null;
  try { saved = localStorage.getItem('theme'); } catch (e) {}
  if (saved === 'light' || saved === 'dark') root.dataset.theme = saved;
  function isDark() {
    if (root.dataset.theme) return root.dataset.theme === 'dark';
    return window.matchMedia('(prefers-color-scheme: dark)').matches;
  }
  function syncLamp() {
    var on = !isDark();
    lamp.setAttribute('aria-pressed', String(on));
    lamp.textContent = 'Light table ' + (on ? 'on' : 'off');
  }
  lamp.addEventListener('click', function () {
    root.dataset.theme = isDark() ? 'light' : 'dark';
    try { localStorage.setItem('theme', root.dataset.theme); } catch (e) {}
    syncLamp();
  });
  syncLamp();

  /* Lightbox */
  var frames = Array.prototype.slice.call(document.querySelectorAll('.frame .photo'));
  var lb = document.getElementById('lb');
  var use = document.getElementById('lb-use');
  var svg = document.getElementById('lb-svg');
  var elTitle = document.getElementById('lb-title');
  var elMeta = document.getElementById('lb-meta');
  var elRoll = document.getElementById('lb-roll');
  var elCount = document.getElementById('lb-count');
  var current = 0;

  function pad(n) { return (n < 10 ? '0' : '') + n; }
  function show(i) {
    current = (i + frames.length) % frames.length;
    var f = frames[current];
    use.setAttribute('href', '#' + f.dataset.src);
    svg.setAttribute('aria-label', f.getAttribute('aria-label').replace(/^Open photograph: /, ''));
    elTitle.textContent = f.dataset.title;
    elMeta.textContent = f.dataset.meta;
    elRoll.textContent = f.closest('.roll').dataset.roll;
    elCount.textContent = pad(current + 1) + ' / ' + pad(frames.length);
  }
  frames.forEach(function (f, i) {
    f.addEventListener('click', function () {
      show(i);
      if (lb.showModal) lb.showModal(); else lb.setAttribute('open', '');
    });
  });
  document.getElementById('lb-prev').addEventListener('click', function () { show(current - 1); });
  document.getElementById('lb-next').addEventListener('click', function () { show(current + 1); });
  document.getElementById('lb-close').addEventListener('click', function () { lb.close(); });
  lb.addEventListener('click', function (e) { if (e.target === lb) lb.close(); });
  lb.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowLeft') show(current - 1);
    if (e.key === 'ArrowRight') show(current + 1);
  });
  lb.addEventListener('close', function () { frames[current].focus(); });

  /* Copy email */

  var address = document.getElementById('mail').textContent;

  });

The portfolio is a single HTML file with no external requests, designed as a film contact sheet on a light table. I have not opened it in a browser, so it is untested.

- **Photographer:** an invented one, Noor Halvorsen in Bergen. The bio, clients, rates and the `hello@noorhalvorsen.example` address are all placeholders to replace.
- **Photographs:** the twelve frames are illustrated SVG stand-ins with a grain overlay, not real photos. To use real ones, replace each `<svg><use href="#pN"/></svg>` with an `<img>`; for a single file they would need to be base64-embedded.
- **Work section:** three film strips of four frames each, with red grease-pencil circles on the picks. On phones each strip scrolls sideways.
- **Viewer:** clicking a frame opens a full-screen view with captions, a frame counter and previous/next buttons (arrow keys and Esc also work).
- **Light table toggle:** this is the light/dark theme switch; it follows the system setting by default and remembers the choice.
- **Contact:** a large mailto link and a copy-address button. There is no form, since a form needs a backend.
